    63→// ── Animation variants ──
    64→const containerVariants = {
    65→  hidden: { opacity: 0 },
    66→  visible: {

    70→};
    71→

    76→

    78→interface ApiQuote {
    79→  id: string;
    80→  title: string;
    81→  description: string | null;
    82→  status: string;
    83→  createdAt: string;
    84→  proposals: {
    85→    id: string;
    86→    title: string;
    87→    description: string | null;
    88→    totalAmount: number | null;
    89→    status: string;
    90→    validUntil: string | null;
    91→    createdAt: string;
    92→  }[];
    93→}
    94→
    95→interface ApiProposal {
    96→  id: string;
    97→  title: string;
    98→  description: string | null;
    99→  totalAmount: number | null;
   100→  status: string;
   101→  validUntil: string | null;
   102→  createdAt: string;
   103→  quote: {
   104→    id: string;
   105→    title: string;
   106→    userId: string;
   107→  };
   108→}
   109→
   110→interface ApiPayment {
   111→  id: string;
   112→  amount: number;
   113→  method: string;
   114→  status: string;
   115→  createdAt: string;
   116→  proposal: {
   117→    id: string;
   118→    title: string;
   119→    description: string | null;
   120→    totalAmount: number | null;
   121→    status: string;
   122→    createdAt: string;
   123→    quote: {
   124→      id: string;
   125→      title: string;
   126→      status: string;
   127→    };
   128→  } | null;
   129→}
   130→
   131→interface ApiInvoice {
   132→  id: string;
   133→  number: string;
   134→  totalAmount: number;
   135→  status: string;
   136→  dueDate: string | null;
   137→  createdAt: string;
   138→  proposal: {
   139→    id: string;
   140→    title: string;
   141→    quote: {
   142→      id: string;
   143→      title: string;
   144→      userId: string;
   145→    };
   146→  };
   147→  items: {
   148→    id: string;
   149→    description: string;
   150→    quantity: number;
   151→    unitPrice: number;
   152→    total: number;

   154→}
   155→
   156→export function FinancialSection() {
   157→  const { t, formatCurrency, formatDate } = useLanguage();
   158→  const user = useAuthStore((s) => s.user);
   159→
   160→  const [activeTab, setActiveTab] = useState('quotes');
   161→  const [quoteDialogOpen, setQuoteDialogOpen] = useState(false);
   162→  const [newQuoteService, setNewQuoteService] = useState('');
   163→  const [newQuoteDescription, setNewQuoteDescription] = useState('');
   164→  const [newQuoteBudget, setNewQuoteBudget] = useState('');
   165→
   166→  const [quotes, setQuotes] = useState<ApiQuote[]>([]);
   167→  const [payments, setPayments] = useState<ApiPayment[]>([]);
   168→  const [invoices, setInvoices] = useState<ApiInvoice[]>([]);
   169→  const [loading, setLoading] = useState(true);
   170→  const [error, setError] = useState<string | null>(null);
   171→  const [submittingQuote, setSubmittingQuote] = useState(false);
   172→
   173→  useEffect(() => {
   174→    if (!user?.id) return;
   175→
   176→    Promise.all([
   177→      fetch(`/api/quotes?userId=${user.id}`).then((res) => {
   178→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   179→        return res.json();
   180→      }),
   181→      fetch(`/api/payments?userId=${user.id}`).then((res) => {
   182→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   183→        return res.json();
   184→      }),
   185→      fetch(`/api/invoices?userId=${user.id}`).then((res) => {
   186→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   187→        return res.json();
   188→      }),
   189→    ])
   190→      .then(([quotesJson, paymentsJson, invoicesJson]) => {
   191→        if (quotesJson.success) setQuotes(quotesJson.data || []);
   192→        if (paymentsJson.success) setPayments(paymentsJson.data || []);
   193→        if (invoicesJson.success) setInvoices(invoicesJson.data || []);
   194→      })
   195→      .catch((err) => {
   196→        setError(err.message || 'Network error');
   197→      })
   198→      .finally(() => {
   199→        setLoading(false);
   200→      });
   201→  }, [user?.id]);
   202→
   203→  // Derive proposals from quotes
   204→  const proposals: ApiProposal[] = quotes.flatMap((q) =>
   205→    q.proposals.map((p) => ({
   206→      ...p,
   207→      quote: { id: q.id, title: q.title, userId: q.id },
   208→    }))
   209→  );
   210→
   211→  const statusBadge = (status: string) => {
   212→    switch (status) {
   213→      case 'approved':
   214→      case 'accepted':
   215→      case 'confirmed':
   216→      case 'paid':
   217→      case 'resolved':
   218→      case 'closed':
   219→        return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('common.approved')}</Badge>;
   220→      case 'pending':
   221→      case 'sent':
   222→      case 'open':
   223→      case 'in_progress':
   224→      case 'issued':
   225→      case 'unpaid':
   226→        return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">{t('common.pending')}</Badge>;
   227→      case 'rejected':
   228→      case 'failed':
   229→        return <Badge className="bg-red-100 text-red-700 border-red-200">{t('common.rejected')}</Badge>;
   230→      case 'overdue':
   231→        return <Badge className="bg-red-100 text-red-800 border-red-200">{t('financial.invoiceOverdue')}</Badge>;
   232→      default:
   233→        return <Badge variant="outline">{status}</Badge>;
   234→    }
   235→  };
   236→
   237→  const methodBadge = (method: string) => {
   238→    switch (method) {
   239→      case 'mpesa':
   240→        return (
   241→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   242→            <Phone className="h-3 w-3 mr-1" />
   243→            {t('financial.mpesa')}
   244→          </Badge>
   245→        );
   246→      case 'transfer':
   247→        return (
   248→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   249→            <Building2 className="h-3 w-3 mr-1" />
   250→            {t('financial.transfer')}
   251→          </Badge>
   252→        );
   253→      case 'deposit':
   254→        return (
   255→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   256→            <BanknoteIcon className="h-3 w-3 mr-1" />
   257→            Deposit
   258→          </Badge>
   259→        );
   260→      default:
   261→        return <Badge variant="outline">{method}</Badge>;
   262→    }
   263→  };
   264→
   265→  const handleSubmitQuote = async () => {
   266→    if (!user?.id || !newQuoteService) return;
   267→
   268→    setSubmittingQuote(true);
   269→    try {
   270→      const res = await fetch('/api/quotes', {
   271→        method: 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify({
   274→          userId: user.id,
   275→          title: newQuoteService,
   276→          description: newQuoteDescription,
   277→        }),
   278→      });
   279→
   280→      const json = await res.json();
   281→      if (json.success) {
   282→        // Refresh quotes list
   283→        const quotesRes = await fetch(`/api/quotes?userId=${user.id}`);
   284→        const quotesJson = await quotesRes.json();
   285→        if (quotesJson.success) setQuotes(quotesJson.data || []);
   286→
   287→        setQuoteDialogOpen(false);
   288→        setNewQuoteService('');
   289→        setNewQuoteDescription('');
   290→        setNewQuoteBudget('');
   291→      }
   292→    } catch (err) {
   293→      console.error('Failed to submit quote:', err);
   294→    } finally {
   295→      setSubmittingQuote(false);
   296→    }
   297→  };
   298→
   299→  // Summary stats from real data
   300→  const totalQuotesValue = proposals.reduce((s, p) => s + (p.totalAmount || 0), 0);
   301→  const totalPaymentsValue = payments.reduce((s, p) => s + p.amount, 0);
   302→  const totalInvoicesValue = invoices.reduce((s, i) => s + i.totalAmount, 0);
   303→  const overdueInvoices = invoices.filter((i) => i.status === 'overdue').length;
   304→  const acceptedProposals = proposals.filter((p) => p.status === 'accepted').length;
   305→  const confirmedPayments = payments.filter((p) => p.status === 'confirmed').length;
   306→
   307→  // ── Loading skeletons ──
   308→  if (loading) {
   309→    return (
   310→      <motion.div
   311→        variants={containerVariants}
   312→        initial="hidden"
   313→        animate="visible"
   314→        className="space-y-6"
   315→      >
   316→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   317→          {[1, 2, 3, 4].map((i) => (
   318→            <Card key={i} className="border-l-4 border-l-emerald-500">
   319→              <CardContent className="p-4 sm:p-5">
   320→                <Skeleton className="h-4 w-20 mb-2" />
   321→                <Skeleton className="h-8 w-24 mb-1" />
   322→                <Skeleton className="h-3 w-16" />
   323→              </CardContent>
   324→            </Card>
   325→          ))}
   326→        </div>
   327→
   328→        <Card>
   329→          <CardContent className="p-6">
   330→            <Skeleton className="h-10 w-full mb-4" />
   331→            <div className="space-y-3">
   332→              {[1, 2, 3, 4, 5].map((i) => (
   333→                <Skeleton key={i} className="h-14 rounded-lg" />
   334→              ))}
   335→            </div>
   336→          </CardContent>
   337→        </Card>
   338→      </motion.div>
   339→    );
   340→  }
   341→
   342→  // ── Error state ──
   343→  if (error) {
   344→    return (
   345→      <motion.div
   346→        variants={containerVariants}
   347→        initial="hidden"
   348→        animate="visible"
   349→        className="space-y-6"
   350→      >
   351→        <Card className="border-l-4 border-l-red-500">
   352→          <CardContent className="p-6">
   353→            <div className="flex items-center gap-3">
   354→              <AlertCircle className="h-6 w-6 text-red-500" />
   355→              <div>
   356→                <h3 className="font-semibold text-red-700">Failed to load financial data</h3>
   357→                <p className="text-sm text-muted-foreground">{error}</p>
   358→              </div>
   359→            </div>
   360→            <Button
   361→              className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   362→              onClick={() => {
   363→                setError(null);
   364→                setLoading(true);
   365→                Promise.all([
   366→                  fetch(`/api/quotes?userId=${user?.id}`).then((r) => r.json()),
   367→                  fetch(`/api/payments?userId=${user?.id}`).then((r) => r.json()),
   368→                  fetch(`/api/invoices?userId=${user?.id}`).then((r) => r.json()),
   369→                ])
   370→                  .then(([q, p, i]) => {
   371→                    if (q.success) setQuotes(q.data || []);
   372→                    if (p.success) setPayments(p.data || []);
   373→                    if (i.success) setInvoices(i.data || []);
   374→                  })
   375→                  .catch((err) => setError(err.message))
   376→                  .finally(() => setLoading(false));
   377→              }}
   378→            >
   379→              {t('common.retry') || 'Retry'}
   380→            </Button>
   381→          </CardContent>
   382→        </Card>
   383→      </motion.div>
   384→    );
   385→  }
   386→
   387→  return (
   388→    <motion.div
   389→      variants={containerVariants}
   390→      initial="hidden"
   391→      animate="visible"
   392→      className="space-y-6"
   393→    >
   394→      {/* ── Summary Stats ── */}
   395→      <motion.div variants={itemVariants} className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   396→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   397→          <CardContent className="p-4 sm:p-5">
   398→            <div className="flex items-start justify-between">
   399→              <div>
   400→                <p className="text-sm text-muted-foreground">{t('dashboard.quotes')}</p>
   401→                <p className="text-2xl font-bold">{quotes.length}</p>
   402→                <p className="text-xs text-emerald-600 mt-1">
   403→                  {totalQuotesValue > 0 ? `Value: ${formatCurrency(totalQuotesValue)}` : 'No proposals yet'}
   404→                </p>
   405→              </div>
   406→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   407→                <FileText className="h-5 w-5" />
   408→              </div>
   409→            </div>
   410→          </CardContent>
   411→        </Card>
   412→
   413→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   414→          <CardContent className="p-4 sm:p-5">
   415→            <div className="flex items-start justify-between">
   416→              <div>
   417→                <p className="text-sm text-muted-foreground">{t('financial.proposal')}</p>
   418→                <p className="text-2xl font-bold">{proposals.length}</p>
   419→                <p className="text-xs text-emerald-600 mt-1">{acceptedProposals} accepted</p>
   420→              </div>
   421→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   422→                <Send className="h-5 w-5" />
   423→              </div>
   424→            </div>
   425→          </CardContent>
   426→        </Card>
   427→
   428→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   429→          <CardContent className="p-4 sm:p-5">
   430→            <div className="flex items-start justify-between">
   431→              <div>
   432→                <p className="text-sm text-muted-foreground">{t('financial.payment')}</p>
   433→                <p className="text-2xl font-bold">{formatCurrency(totalPaymentsValue)}</p>
   434→                <p className="text-xs text-emerald-600 mt-1">{confirmedPayments} confirmed</p>
   435→              </div>
   436→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   437→                <CreditCard className="h-5 w-5" />
   438→              </div>
   439→            </div>
   440→          </CardContent>
   441→        </Card>
   442→
   443→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   444→          <CardContent className="p-4 sm:p-5">
   445→            <div className="flex items-start justify-between">
   446→              <div>
   447→                <p className="text-sm text-muted-foreground">{t('financial.invoice')}</p>
   448→                <p className="text-2xl font-bold">{invoices.length}</p>
   449→                <p className="text-xs text-red-600 mt-1">{overdueInvoices} overdue</p>
   450→              </div>
   451→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   452→                <Receipt className="h-5 w-5" />
   453→              </div>
   454→            </div>
   455→          </CardContent>
   456→        </Card>
   457→      </motion.div>
   458→
   459→      {/* ── Main Tabs ── */}
   460→      <motion.div variants={itemVariants}>
   461→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   462→          <TabsList className="bg-emerald-50 w-full sm:w-auto">
   463→            <TabsTrigger value="quotes" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   464→              {t('dashboard.quotes')}
   465→            </TabsTrigger>
   466→            <TabsTrigger value="proposals" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   467→              {t('financial.proposal')}
   468→            </TabsTrigger>
   469→            <TabsTrigger value="payments" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   470→              {t('financial.payment')}
   471→            </TabsTrigger>
   472→            <TabsTrigger value="invoices" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   473→              {t('financial.invoice')}
   474→            </TabsTrigger>
   475→          </TabsList>
   476→
   477→          {/* ── Quotes Tab ── */}
   478→          <TabsContent value="quotes" className="space-y-4 mt-4">
   479→            {/* Request New Quote */}
   480→            <Card>
   481→              <CardHeader>
   482→                <div className="flex items-center justify-between">
   483→                  <CardTitle className="text-lg flex items-center gap-2">
   484→                    <Plus className="h-5 w-5 text-emerald-600" />
   485→                    {t('financial.quoteRequestTitle')}
   486→                  </CardTitle>
   487→                </div>
   488→                <CardDescription>{t('financial.quoteRequestDesc')}</CardDescription>
   489→              </CardHeader>
   490→              <CardContent>
   491→                <Dialog open={quoteDialogOpen} onOpenChange={setQuoteDialogOpen}>
   492→                  <DialogTrigger asChild>
   493→                    <Button className="bg-emerald-600 hover:bg-emerald-700 text-white font-semibold">
   494→                      <Plus className="h-4 w-4 mr-2" />
   495→                      {t('dashboard.quoteRequest')}
   496→                    </Button>
   497→                  </DialogTrigger>
   498→                  <DialogContent className="sm:max-w-lg">
   499→                    <DialogHeader>
   500→                      <DialogTitle>{t('financial.quoteRequestTitle')}</DialogTitle>
   501→                      <DialogDescription>{t('financial.quoteRequestDesc')}</DialogDescription>
   502→                    </DialogHeader>
   503→                    <div className="space-y-4">
   504→                      <div className="space-y-2">
   505→                        <Label>{t('services.title')}</Label>
   506→                        <Select value={newQuoteService} onValueChange={setNewQuoteService}>
   507→                          <SelectTrigger className="w-full">
   508→                            <SelectValue placeholder="Select a service" />
   509→                          </SelectTrigger>
   510→                          <SelectContent>
   511→                            <SelectItem value="web-dev">{t('services.webDev')}</SelectItem>
   512→                            <SelectItem value="mobile-dev">{t('services.mobileDev')}</SelectItem>
   513→                            <SelectItem value="cloud">{t('services.cloud')}</SelectItem>
   514→                            <SelectItem value="ai">{t('services.ai')}</SelectItem>
   515→                            <SelectItem value="consulting">{t('services.consulting')}</SelectItem>
   516→                            <SelectItem value="design">{t('services.design')}</SelectItem>
   517→                            <SelectItem value="seo">{t('services.seo')}</SelectItem>
   518→                            <SelectItem value="maintenance">{t('services.maintenance')}</SelectItem>
   519→                          </SelectContent>
   520→                        </Select>
   521→                      </div>
   522→                      <div className="space-y-2">
   523→                        <Label>{t('financial.description')}</Label>
   524→                        <Textarea
   525→                          value={newQuoteDescription}
   526→                          onChange={(e) => setNewQuoteDescription(e.target.value)}
   527→                          placeholder="Describe your requirements..."
   528→                        />
   529→                      </div>
   530→                      <div className="space-y-2">
   531→                        <Label>{t('financial.amount')} ({t('financial.currency')})</Label>
   532→                        <Input
   533→                          type="number"
   534→                          value={newQuoteBudget}
   535→                          onChange={(e) => setNewQuoteBudget(e.target.value)}
   536→                          placeholder="Estimated budget in MT"
   537→                        />
   538→                      </div>
   539→                    </div>
   540→                    <DialogFooter className="mt-4">
   541→                      <Button variant="outline" onClick={() => setQuoteDialogOpen(false)}>
   542→                        {t('common.cancel')}
   543→                      </Button>
   544→                      <Button
   545→                        className="bg-emerald-600 hover:bg-emerald-700 text-white"
   546→                        onClick={handleSubmitQuote}
   547→                        disabled={submittingQuote || !newQuoteService}
   548→                      >
   549→                        <Send className="h-4 w-4 mr-2" />
   550→                        {submittingQuote ? 'Submitting...' : t('common.confirm')}
   551→                      </Button>
   552→                    </DialogFooter>
   553→                  </DialogContent>
   554→                </Dialog>
   555→              </CardContent>
   556→            </Card>
   557→
   558→            {/* Existing Quotes List */}
   559→            <Card>
   560→              <CardHeader>
   561→                <CardTitle className="text-lg flex items-center gap-2">
   562→                  <FileText className="h-5 w-5 text-emerald-600" />
