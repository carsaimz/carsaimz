    42→    42→// ──────────────────────────────────────────────
    43→    43→// Auth Store (Firebase Auth — client-side authentication)
    44→    44→//
    45→    45→// Flow:
    46→    46→// 1. Client authenticates with Firebase Auth directly (email/password, Google, phone, anonymous)
    47→    47→// 2. Firebase Auth returns a UserCredential with an ID token
    48→    48→// 3. Client sends ID token to /api/auth/login, /api/auth/social, /api/auth/anonymous, or /api/auth/verify
    49→    49→// 4. Server verifies token, returns Firestore profile (role, isActive, etc.)
    50→    50→// 5. Store saves the combined profile
    51→    51→//
    52→    52→// Phone Auth Flow:
    53→    53→// 1. Client calls loginWithPhone(phoneNumber, recaptchaContainerId)
    54→    54→// 2. Firebase sends SMS OTP to the phone number
    55→    55→// 3. User enters the OTP code
    56→    56→// 4. Client calls verifyPhoneCode(verificationId, otpCode)
    57→    57→// 5. Firebase verifies OTP and returns credential with ID token
    58→    58→// 6. Client sends ID token to /api/auth/social (phone treated as social provider)
    59→    59→// 7. Server verifies token and returns Firestore profile
    60→    60→// ──────────────────────────────────────────────

   108→   108→function mapRole(roleData: any): UserRole {
   109→   109→  if (typeof roleData === 'string') return roleData as UserRole

   114→   114→/**
   115→   115→ * Verify authentication with server API route.
   116→   116→ * If server is unavailable (static export / Capacitor), falls back to
   117→   117→ * client-side Firestore profile lookup/creation.
   118→   118→ */
   119→   119→async function verifyWithServer(idToken: string, endpoint: string): Promise<{ success: boolean; user?: User; error?: string }> {
   120→   120→  try {
   121→   121→    const res = await fetch(buildApiUrl(endpoint), {
   122→   122→      method: 'POST',
   123→   123→      headers: { 'Content-Type': 'application/json' },
   124→   124→      body: JSON.stringify({ idToken }),
   125→   125→    })
   126→   126→
   127→   127→    // If the server returned a valid response (not 404), use it
   128→   128→    if (res.ok) {
   129→   129→      const data = await res.json()
   130→   130→
   131→   131→      if (data.user) {
   132→   132→        const userRole = mapRole(data.user.role)
   133→   133→        const user: User = {
   134→   134→          id: data.user.id,
   135→   135→          name: data.user.name,
   136→   136→          email: data.user.email,
   137→   137→          role: userRole,
   138→   138→          avatar: data.user.avatar || null,
   139→   139→          phone: data.user.phone || null,
   140→   140→          company: data.user.company || null,
   141→   141→          bio: data.user.bio || null,
   142→   142→          address: data.user.address || null,
   143→   143→          authProvider: data.user.authProvider as AuthProvider || 'unknown',
   144→   144→          isAnonymous: data.user.isAnonymous || false,
   145→   145→          emailVerified: data.user.emailVerified || false,
   146→   146→        }
   147→   147→        return { success: true, user }
   148→   148→      } else {
   149→   149→        return { success: false, error: data.error || 'Autenticação falhou.' }
   150→   150→      }
   151→   151→    }
   152→   152→
   153→   153→    // Server returned 404 or other error — fall back to client-side Firestore
   154→   154→    console.warn('[Auth] Server API unavailable, falling back to client-side Firestore')
   155→   155→    return await verifyWithClientFirestore(idToken)
   156→   156→  } catch (err) {
   157→   157→    // Network error — server not available (static export / Capacitor)
   158→   158→    console.warn('[Auth] Server API not reachable, falling back to client-side Firestore:', err)
   159→   159→    return await verifyWithClientFirestore(idToken)
   160→   160→  }
   161→   161→}
   162→   162→
   163→   163→/**
   164→   164→ * Client-side Firestore fallback for authentication.
   165→   165→ * Used when the server API routes are not available (static export / Capacitor).
   166→   166→ * Reads the user profile from Firestore directly, or creates one if it doesn't exist.
   167→   167→ */
   168→   168→async function verifyWithClientFirestore(idToken: string): Promise<{ success: boolean; user?: User; error?: string }> {
   169→   169→  try {
   170→   170→    const { auth, firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   171→   171→
   172→   172→    if (!isFirebaseConfigured() || !firestoreClient) {
   173→   173→      return { success: false, error: 'Firebase não configurado.' }
   174→   174→    }
   175→   175→
   176→   176→    // Decode the ID token to get the Firebase user info
   177→   177→    // We need to get the current Firebase Auth user
   178→   178→    const currentUser = auth.currentUser
   179→   179→    if (!currentUser) {
   180→   180→      return { success: false, error: 'Utilizador não autenticado.' }
   181→   181→    }
   182→   182→
   183→   183→    const uid = currentUser.uid
   184→   184→    const { doc, getDoc, setDoc } = await import('firebase/firestore')
   185→   185→
   186→   186→    // Try to read existing profile from Firestore
   187→   187→    const userRef = doc(firestoreClient, 'users', uid)
   188→   188→    const userSnap = await getDoc(userRef)
   189→   189→
   190→   190→    if (userSnap.exists()) {
   191→   191→      // Profile exists — return it
   192→   192→      const profile = userSnap.data()
   193→   193→      const userRole = mapRole(profile.role)
   194→   194→      const user: User = {
   195→   195→        id: uid,
   196→   196→        name: profile.name || currentUser.displayName || 'Utilizador',
   197→   197→        email: profile.email || currentUser.email || null,
   198→   198→        role: userRole,
   199→   199→        avatar: profile.avatar || currentUser.photoURL || null,
   200→   200→        phone: profile.phone || currentUser.phoneNumber || null,
   201→   201→        company: profile.company || null,
   202→   202→        bio: profile.bio || null,
   203→   203→        address: profile.address || null,
   204→   204→        authProvider: (currentUser.providerData[0]?.providerId as AuthProvider) || 'unknown',
   205→   205→        isAnonymous: currentUser.isAnonymous,
   206→   206→        emailVerified: currentUser.emailVerified,
   207→   207→      }
   208→   208→      return { success: true, user }
   209→   209→    }
   210→   210→
   211→   211→    // Profile doesn't exist — create it (first-time social/anonymous login)
   212→   212→    const providerId = currentUser.providerData[0]?.providerId || 'unknown'
   213→   213→    let authProvider: AuthProvider = 'unknown'
   214→   214→    if (providerId === 'google.com') authProvider = 'google.com'
   215→   215→    else if (providerId === 'phone') authProvider = 'phone'
   216→   216→    else if (providerId === 'anonymous') authProvider = 'anonymous'
   217→   217→    else if (providerId === 'password') authProvider = 'email'
   218→   218→
   219→   219→    const newProfile = {
   220→   220→      name: currentUser.displayName || 'Utilizador',
   221→   221→      email: currentUser.email || null,
   222→   222→      phone: currentUser.phoneNumber || null,
   223→   223→      avatar: currentUser.photoURL || null,
   224→   224→      company: null,
   225→   225→      bio: null,
   226→   226→      address: null,
   227→   227→      role: 'user',
   228→   228→      isActive: true,
   229→   229→      emailVerified: currentUser.emailVerified,
   230→   230→      authProvider,
   231→   231→      isAnonymous: currentUser.isAnonymous,
   232→   232→      createdAt: new Date(),
   233→   233→      updatedAt: new Date(),
   234→   234→    }
   235→   235→
   236→   236→    await setDoc(userRef, newProfile)
   237→   237→
   238→   238→    const user: User = {
   239→   239→      id: uid,
   240→   240→      name: newProfile.name,
   241→   241→      email: newProfile.email,
   242→   242→      role: 'user',
   243→   243→      avatar: newProfile.avatar,
   244→   244→      phone: newProfile.phone,
   245→   245→      company: null,
   246→   246→      bio: null,
   247→   247→      address: null,
   248→   248→      authProvider,
   249→   249→      isAnonymous: currentUser.isAnonymous,
   250→   250→      emailVerified: currentUser.emailVerified,
   251→   251→    }
   252→   252→    return { success: true, user }
   253→   253→  } catch (err) {
   254→   254→    console.error('[Auth] Client-side Firestore fallback error:', err)
   255→   255→    return { success: false, error: 'Erro de autenticação. Tente novamente.' }
   256→   256→  }
   257→   257→}

   278→   278→      isUser: false,
   279→   279→      isSuperAdmin: false,
   280→   280→      isLoading: false,
   281→   281→      hasHydrated: false,

   284→   284→      idToken: null,
   285→   285→      phoneVerificationId: null,
   286→   286→
   287→   287→      // ── Login with pre-obtained ID token ──
   288→   288→      loginWithToken: async (idToken: string): Promise<AuthResult> => {
   289→   289→        set({ isLoading: true, lastLoginError: null })
   290→   290→        const result = await verifyWithServer(idToken, '/api/auth/login')
   291→   291→

   298→   298→          return { success: false, error: result.error }
   299→   299→        }

   301→   301→
   302→   302→      // ── Email/password login via Firebase Auth client ──
   303→   303→      loginWithEmailPassword: async (email: string, password: string): Promise<AuthResult> => {
   304→   304→        set({ isLoading: true, lastLoginError: null })
   305→   305→
   306→   306→        try {
   307→   307→          const { signInWithEmailAndPassword, auth } = await getFirebaseAuth()
   308→   308→          const credential = await signInWithEmailAndPassword(auth, email, password)
   309→   309→          const idToken = await credential.user.getIdToken()
   310→   310→
   311→   311→          const result = await verifyWithServer(idToken, '/api/auth/login')
   312→   312→
   313→   313→          if (result.success && result.user) {
   314→   314→            setUserInStore(set, result.user)
   315→   315→            set({ idToken })
   316→   316→            return { success: true, user: result.user }
   317→   317→          } else {
   318→   318→            set({ isLoading: false, lastLoginError: result.error })
   319→   319→            return { success: false, error: result.error }
   320→   320→          }
   321→   321→        } catch (err: any) {
   322→   322→          console.error('Email login error:', err)
   323→   323→          let errorMsg = 'Credenciais inválidas.'
   324→   324→
   325→   325→          if (err.code === 'auth/user-not-found' || err.code === 'auth/wrong-password' || err.code === 'auth/invalid-credential') {
   326→   326→            errorMsg = 'Email ou senha incorretos.'
   327→   327→          } else if (err.code === 'auth/too-many-requests') {
   328→   328→            errorMsg = 'Muitas tentativas. Tente novamente mais tarde.'
   329→   329→          } else if (err.code === 'auth/user-disabled') {
   330→   330→            errorMsg = 'Conta desactivada. Contacte o suporte.'
   331→   331→          } else if (err.code === 'auth/invalid-email') {
   332→   332→            errorMsg = 'Email inválido.'
   333→   333→          } else if (err.message?.includes('not configured')) {
   334→   334→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   335→   335→          }
   336→   336→
   337→   337→          set({ isLoading: false, lastLoginError: errorMsg })

   340→   340→      },

   422→   422→

   424→   424→          const credential = PhoneAuthProvider.credential(verificationId, verificationCode)
   425→   425→          const userCredential = await signInWithCredential(auth, credential)
   426→   426→          const idToken = await userCredential.user.getIdToken()
   427→   427→
   428→   428→          // Verify with server to get Firestore profile
   429→   429→          const result = await verifyWithServer(idToken, '/api/auth/social')
   430→   430→
   431→   431→          if (result.success && result.user) {
   432→   432→            setUserInStore(set, result.user)

   444→   444→            errorMsg = 'Código SMS incorreto. Verifique e tente novamente.'
   445→   445→          } else if (err.code === 'auth/invalid-verification-id') {
   446→   446→            errorMsg = 'Sessão de verificação expirada. Tente novamente.'
   447→   447→          } else if (err.code === 'auth/code-expired') {
   448→   448→            errorMsg = 'Código expirado. Solicite um novo código SMS.'
   449→   449→          }
   450→   450→
   451→   451→          set({ isLoading: false, lastLoginError: errorMsg, phoneVerificationId: null })
   452→   452→          return { success: false, error: errorMsg }
   453→   453→        }
   454→   454→      },
   455→   455→
   456→   456→      // ── Anonymous Sign-In ──
   457→   457→      loginAnonymously: async (): Promise<AuthResult> => {
   458→   458→        set({ isLoading: true, lastLoginError: null })
   459→   459→
   460→   460→        try {
   461→   461→          const { signInAnonymously, auth } = await getFirebaseAuth()
   462→   462→          const credential = await signInAnonymously(auth)
   463→   463→          const idToken = await credential.user.getIdToken()
   464→   464→
   465→   465→          const result = await verifyWithServer(idToken, '/api/auth/anonymous')
   466→   466→
   467→   467→          if (result.success && result.user) {
   468→   468→            setUserInStore(set, result.user)
   469→   469→            set({ idToken })
   470→   470→            return { success: true, user: result.user }
   471→   471→          } else {
   472→   472→            set({ isLoading: false, lastLoginError: result.error })
   473→   473→            return { success: false, error: result.error }
   474→   474→          }
   475→   475→        } catch (err: any) {
   476→   476→          console.error('Anonymous login error:', err)
   477→   477→          set({ isLoading: false, lastLoginError: 'Falha no login como visitante.' })
   478→   478→          return { success: false, error: 'Falha no login como visitante.' }
   479→   479→        }
   480→   480→      },
   481→   481→
   482→   482→      // ── Email/password registration ──
   483→   483→      register: async (name: string, email: string, password: string, phone?: string): Promise<AuthResult> => {
   484→   484→        set({ isLoading: true, lastRegisterError: null })
   485→   485→
   486→   486→        try {
   487→   487→          const { createUserWithEmailAndPassword, auth, updateProfile } = await getFirebaseAuth()
   488→   488→          const credential = await createUserWithEmailAndPassword(auth, email, password)
   489→   489→
   490→   490→          // Update display name in Firebase Auth
   491→   491→          await updateProfile(credential.user, { displayName: name })
   492→   492→
   493→   493→          const idToken = await credential.user.getIdToken()
   494→   494→
   495→   495→          // Send to server to create Firestore profile
   496→   496→          const res = await fetch(buildApiUrl('/api/auth/register'), {
   497→   497→            method: 'POST',
   498→   498→            headers: { 'Content-Type': 'application/json' },
   499→   499→            body: JSON.stringify({ name, email, phone }),
   500→   500→          })
   501→   501→
   502→   502→          const data = await res.json()
   503→   503→
   504→   504→          if (data.user) {
   505→   505→            const userRole = mapRole(data.user.role)
   506→   506→            const user: User = {
   507→   507→              id: data.user.id || credential.user.uid,
   508→   508→              name: data.user.name || name,
   509→   509→              email: data.user.email || email,
   510→   510→              role: userRole,
   511→   511→              avatar: null,
   512→   512→              phone: data.user.phone || phone || null,
   513→   513→              authProvider: 'email',
   514→   514→              isAnonymous: false,
   515→   515→              emailVerified: false,
   516→   516→            }
   517→   517→            setUserInStore(set, user)
   518→   518→            set({ idToken, lastRegisterError: null })
   519→   519→            return { success: true, user }
   520→   520→          } else {
   521→   521→            // Firebase Auth user was created but Firestore profile failed
   522→   522→            // Still set the user with basic info
   523→   523→            const user: User = {
   524→   524→              id: credential.user.uid,
   525→   525→              name: name,
   526→   526→              email: email,
   527→   527→              role: 'user',
   528→   528→              avatar: null,
   529→   529→              authProvider: 'email',
   530→   530→              isAnonymous: false,
   531→   531→              emailVerified: false,
   532→   532→            }
   533→   533→            setUserInStore(set, user)
   534→   534→            set({ idToken })
   535→   535→            return { success: true, user }
   536→   536→          }
   537→   537→        } catch (err: any) {
   538→   538→          console.error('Register error:', err)
   539→   539→          let errorMsg = 'Falha ao criar conta.'
   540→   540→
   541→   541→          if (err.code === 'auth/email-already-in-use') {
   542→   542→            errorMsg = 'Este email já está registado.'
   543→   543→          } else if (err.code === 'auth/weak-password') {
   544→   544→            errorMsg = 'Senha demasiado fraca. Use pelo menos 6 caracteres.'
   545→   545→          } else if (err.code === 'auth/invalid-email') {
   546→   546→            errorMsg = 'Email inválido.'
   547→   547→          } else if (err.message?.includes('not configured')) {
   548→   548→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   549→   549→          }
   550→   550→
   551→   551→          set({ isLoading: false, lastRegisterError: errorMsg })
   552→   552→          return { success: false, error: errorMsg }
   553→   553→        }
   554→   554→      },
   555→   555→
   556→   556→      logout: async () => {
   557→   557→        try {
   558→   558→          const { signOut, auth } = await getFirebaseAuth()
   559→   559→          await signOut(auth)
   560→   560→        } catch (err) {
   561→   561→          console.warn('Firebase signOut failed (non-critical):', err)
   562→   562→        }
   563→   563→
   564→   564→        set({
   565→   565→          user: null,
   566→   566→          isAuthenticated: false,
   567→   567→          isAdmin: false,
   568→   568→          isPartner: false,
   569→   569→          isUser: false,
   570→   570→          isSuperAdmin: false,
   571→   571→          idToken: null,
   572→   572→          phoneVerificationId: null,
   573→   573→          lastLoginError: null,
   574→   574→          lastRegisterError: null,
   575→   575→        })
   576→   576→      },
   577→   577→
   578→   578→      setUser: (user: User) => {
   579→   579→        setUserInStore(set, user)
   580→   580→      },
   581→   581→
   582→   582→      updateAvatar: (avatar: string) => {
   583→   583→        const currentUser = get().user
   584→   584→        if (currentUser) {
   585→   585→          set({ user: { ...currentUser, avatar } })
   586→   586→        }
   587→   587→      },
   588→   588→
   589→   589→      setIdToken: (token: string) => {
   590→   590→        set({ idToken: token })
   591→   591→      },
   592→   592→    }),
   593→   593→    {
   594→   594→      name: 'carsai-auth',
   595→   595→      partialize: (state) => ({
   596→   596→        user: state.user,
   597→   597→        isAuthenticated: state.isAuthenticated,

   600→   600→        isUser: state.isUser,

   604→   604→      onRehydrateStorage: () => (state) => {

   606→   606→          state.hasHydrated = true
   607→   607→        }
   608→   608→      },
   609→   609→    }
   610→   610→  )

   617→   617→interface NotificationState {
   618→   618→  notifications: Notification[]
   619→   619→  unreadCount: number
   620→   620→  addNotification: (type: NotificationType, title: string, message: string) => void
   621→   621→  markAsRead: (id: string) => void
   622→   622→  markAllAsRead: () => void
   623→   623→  removeNotification: (id: string) => void
   624→   624→  clearAll: () => void
   625→   625→}
   626→   626→
   627→   627→export const useNotificationStore = create<NotificationState>((set, get) => ({
   628→   628→  notifications: [],
   629→   629→  unreadCount: 0,
   630→   630→
   631→   631→  addNotification: (type: NotificationType, title: string, message: string) => {
   632→   632→    const id = `notif-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
   633→   633→    const newNotification: Notification = {
   634→   634→      id,
   635→   635→      type,
   636→   636→      title,
   637→   637→      message,
   638→   638→      read: false,
   639→   639→      createdAt: new Date(),
   640→   640→    }
   641→   641→    const updated = [newNotification, ...get().notifications]
   642→   642→    set({
   643→   643→      notifications: updated,
   644→   644→      unreadCount: updated.filter((n) => !n.read).length,
   645→   645→    })
   646→   646→  },
   647→   647→
   648→   648→  markAsRead: (id: string) => {
   649→   649→    const updated = get().notifications.map((n) =>
   650→   650→      n.id === id ? { ...n, read: true } : n
   651→   651→    )
   652→   652→    set({
   653→   653→      notifications: updated,
   654→   654→      unreadCount: updated.filter((n) => !n.read).length,
   655→   655→    })
   656→   656→  },
   657→   657→
   658→   658→  markAllAsRead: () => {
   659→   659→    const updated = get().notifications.map((n) => ({ ...n, read: true }))
   660→   660→    set({
   661→   661→      notifications: updated,
   662→   662→      unreadCount: 0,
   663→   663→    })
   664→   664→  },
   665→   665→
   666→   666→  removeNotification: (id: string) => {
   667→   667→    const updated = get().notifications.filter((n) => n.id !== id)
   668→   668→    set({
   669→   669→      notifications: updated,
   670→   670→      unreadCount: updated.filter((n) => !n.read).length,
   671→   671→    })
   672→   672→  },
   673→   673→
   674→   674→  clearAll: () => {
   675→   675→    set({ notifications: [], unreadCount: 0 })
   676→   676→  },
   677→   677→}))
