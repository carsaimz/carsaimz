    97→function slugify(text: string): string {
    98→  return text
    99→    .toLowerCase()
   100→    .normalize('NFD')
   101→    .replace(/[\u0300-\u036f]/g, '')
   102→    .replace(/[^\w\s-]/g, '')
   103→    .replace(/\s+/g, '-')
   104→    .replace(/-+/g, '-')
   105→    .trim();
   106→}
   107→
   108→function parseI18n(jsonStr: string | null | undefined): Record<string, string> {
   109→  if (!jsonStr) return {};
   110→  try {
   111→    return JSON.parse(jsonStr);
   112→  } catch {
   113→    return {};
   114→  }
   115→}
   116→
   117→function stringifyI18n(obj: Record<string, string>): string | null {
   118→  const filtered = Object.fromEntries(
   119→    Object.entries(obj).filter(([_, v]) => v && v.trim())
   120→  );
   121→  if (Object.keys(filtered).length === 0) return null;
   122→  return JSON.stringify(filtered);
   123→}

   125→function getApiEndpoint(type: ContentType): string {
   126→  return `/api/admin/${type}`;
   127→}
