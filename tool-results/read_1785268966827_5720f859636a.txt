    51→// ──────────────────────────────────────────────
    52→// Auth Store (Firebase Auth — client-side authentication)
    53→//
    54→// Flow:
    55→// 1. Client authenticates with Firebase Auth directly (email/password, Google, phone, anonymous)
    56→// 2. Firebase Auth returns a UserCredential with an ID token
    57→// 3. Client sends ID token to /api/auth/login, /api/auth/social, /api/auth/anonymous, or /api/auth/verify
    58→// 4. Server verifies token, returns Firestore profile (role, isActive, etc.)
    59→// 5. Store saves the combined profile
    60→//
    61→// Phone Auth Flow:
    62→// 1. Client calls loginWithPhone(phoneNumber, recaptchaContainerId)
    63→// 2. Firebase sends SMS OTP to the phone number
    64→// 3. User enters the OTP code
    65→// 4. Client calls verifyPhoneCode(verificationId, otpCode)
    66→// 5. Firebase verifies OTP and returns credential with ID token
    67→// 6. Client sends ID token to /api/auth/social (phone treated as social provider)
    68→// 7. Server verifies token and returns Firestore profile
    69→// ──────────────────────────────────────────────

   105→function setUserInStore(set: (partial: Partial<AuthState>) => void, user: User) {
   106→  const userRole = user.role
   107→  set({
   108→    user,
   109→    isAuthenticated: true,
   110→    isAdmin: userRole === 'admin' || userRole === 'super_admin',
   111→    isPartner: userRole === 'partner',
   112→    isUser: userRole === 'user',
   113→    isSuperAdmin: userRole === 'super_admin',
   114→    isLoading: false,
   115→  })
   116→}
   117→
   118→function mapRole(roleData: any): UserRole {
   119→  if (typeof roleData === 'string') return roleData as UserRole
   120→  if (roleData && typeof roleData === 'object' && roleData.name) return roleData.name as UserRole
   121→  return 'user'
   122→}

   124→/**
   125→ * Verify authentication with server API route.
   126→ * If server is unavailable (static export / Capacitor), falls back to
   127→ * client-side Firestore profile lookup/creation.
   128→ */
   129→async function verifyWithServer(idToken: string, endpoint: string): Promise<{ success: boolean; user?: User; error?: string }> {
   130→  try {
   131→    const res = await fetch(buildApiUrl(endpoint), {
   132→      method: 'POST',
   133→      headers: { 'Content-Type': 'application/json' },
   134→      body: JSON.stringify({ idToken }),
   135→    })
   136→
   137→    // If the server returned a valid response (not 404), use it
   138→    if (res.ok) {
   139→      const data = await res.json()
   140→
   141→      if (data.user) {
   142→        const userRole = mapRole(data.user.role)
   143→        const user: User = {
   144→          id: data.user.id,
   145→          name: data.user.name,
   146→          email: data.user.email,
   147→          role: userRole,
   148→          avatar: data.user.avatar || null,
   149→          phone: data.user.phone || null,
   150→          company: data.user.company || null,
   151→          bio: data.user.bio || null,
   152→          address: data.user.address || null,
   153→          authProvider: data.user.authProvider as AuthProvider || 'unknown',
   154→          isAnonymous: data.user.isAnonymous || false,
   155→          emailVerified: data.user.emailVerified || false,
   156→        }
   157→        return { success: true, user }
   158→      } else {
   159→        return { success: false, error: data.error || 'Autenticação falhou.' }
   160→      }
   161→    }
   162→
   163→    // Server returned 404 or other error — fall back to client-side Firestore
   164→    console.warn('[Auth] Server API unavailable, falling back to client-side Firestore')
   165→    return await verifyWithClientFirestore(idToken)
   166→  } catch (err) {
   167→    // Network error — server not available (static export / Capacitor)
   168→    console.warn('[Auth] Server API not reachable, falling back to client-side Firestore:', err)
   169→    return await verifyWithClientFirestore(idToken)
   170→  }
   171→}

   173→/**
   174→ * Client-side Firestore fallback for authentication.
   175→ * Used when the server API routes are not available (static export / Capacitor).
   176→ * Reads the user profile from Firestore directly, or creates one if it doesn't exist.
   177→ */
   178→async function verifyWithClientFirestore(idToken: string): Promise<{ success: boolean; user?: User; error?: string }> {
   179→  try {
   180→    const { auth, firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   181→
   182→    if (!isFirebaseConfigured() || !firestoreClient) {
   183→      return { success: false, error: 'Firebase não configurado.' }
   184→    }
   185→
   186→    // Decode the ID token to get the Firebase user info
   187→    // We need to get the current Firebase Auth user
   188→    const currentUser = auth.currentUser
   189→    if (!currentUser) {
   190→      return { success: false, error: 'Utilizador não autenticado.' }
   191→    }
   192→
   193→    const uid = currentUser.uid
   194→    const { doc, getDoc, setDoc } = await import('firebase/firestore')
   195→
   196→    // Try to read existing profile from Firestore
   197→    const userRef = doc(firestoreClient, 'users', uid)
   198→    const userSnap = await getDoc(userRef)
   199→
   200→    if (userSnap.exists()) {
   201→      // Profile exists — return it
   202→      const profile = userSnap.data()
   203→      const userRole = mapRole(profile.role)
   204→      const user: User = {
   205→        id: uid,
   206→        name: profile.name || currentUser.displayName || 'Utilizador',
   207→        email: profile.email || currentUser.email || null,
   208→        role: userRole,
   209→        avatar: profile.avatar || currentUser.photoURL || null,
   210→        phone: profile.phone || currentUser.phoneNumber || null,
   211→        company: profile.company || null,
   212→        bio: profile.bio || null,
   213→        address: profile.address || null,
   214→        authProvider: (currentUser.providerData[0]?.providerId as AuthProvider) || 'unknown',
   215→        isAnonymous: currentUser.isAnonymous,
   216→        emailVerified: currentUser.emailVerified,
   217→      }
   218→      return { success: true, user }
   219→    }
   220→
   221→    // Profile doesn't exist — create it (first-time social/anonymous login)
   222→    const providerId = currentUser.providerData[0]?.providerId || 'unknown'
   223→    let authProvider: AuthProvider = 'unknown'
   224→    if (providerId === 'google.com') authProvider = 'google.com'
   225→    else if (providerId === 'phone') authProvider = 'phone'
   226→    else if (providerId === 'anonymous') authProvider = 'anonymous'
   227→    else if (providerId === 'password') authProvider = 'email'
   228→
   229→    const newProfile = {
   230→      name: currentUser.displayName || 'Utilizador',
   231→      email: currentUser.email || null,
   232→      phone: currentUser.phoneNumber || null,
   233→      avatar: currentUser.photoURL || null,
   234→      company: null,
   235→      bio: null,
   236→      address: null,
   237→      role: 'user',
   238→      isActive: true,
   239→      emailVerified: currentUser.emailVerified,
   240→      authProvider,
   241→      isAnonymous: currentUser.isAnonymous,
   242→      createdAt: new Date(),
   243→      updatedAt: new Date(),
   244→    }
   245→
   246→    await setDoc(userRef, newProfile)
   247→
   248→    const user: User = {
   249→      id: uid,
   250→      name: newProfile.name,
   251→      email: newProfile.email,
   252→      role: 'user',
   253→      avatar: newProfile.avatar,
   254→      phone: newProfile.phone,
   255→      company: null,
   256→      bio: null,
   257→      address: null,
   258→      authProvider,
   259→      isAnonymous: currentUser.isAnonymous,
   260→      emailVerified: currentUser.emailVerified,
   261→    }
   262→    return { success: true, user }
   263→  } catch (err) {
   264→    console.error('[Auth] Client-side Firestore fallback error:', err)
   265→    return { success: false, error: 'Erro de autenticação. Tente novamente.' }
   266→  }
   267→}

   281→export const useAuthStore = create<AuthState>()(
   282→  persist(
   283→    (set, get) => ({
   284→      user: null,
   285→      isAuthenticated: false,
   286→      isAdmin: false,
   287→      isPartner: false,
   288→      isUser: false,
   289→      isSuperAdmin: false,

   293→      lastRegisterError: null,
   294→      idToken: null,
   295→      phoneVerificationId: null,
   296→
   297→      // ── Login with pre-obtained ID token ──
   298→      loginWithToken: async (idToken: string): Promise<AuthResult> => {
   299→        set({ isLoading: true, lastLoginError: null })
   300→        const result = await verifyWithServer(idToken, '/api/auth/login')
   301→
   302→        if (result.success && result.user) {
   303→          setUserInStore(set, result.user)
   304→          set({ idToken })
   305→          return { success: true, user: result.user }
   306→        } else {
   307→          set({ isLoading: false, lastLoginError: result.error })
   308→          return { success: false, error: result.error }
   309→        }
   310→      },
   311→
   312→      // ── Email/password login via Firebase Auth client ──
   313→      loginWithEmailPassword: async (email: string, password: string): Promise<AuthResult> => {
   314→        set({ isLoading: true, lastLoginError: null })
   315→
   316→        try {
   317→          let idToken: string
   318→
   319→          if (shouldUseNativeAuth()) {
   320→            // ── Native path: @capacitor-firebase/authentication ──
   321→            const nativeResult = await nativeSignInWithEmailPassword(email, password)
   322→            idToken = nativeResult.idToken
   323→          } else {
   324→            // ── Web path: Firebase Web SDK ──
   325→            const { signInWithEmailAndPassword, auth } = await getFirebaseAuth()
   326→            const credential = await signInWithEmailAndPassword(auth, email, password)
   327→            idToken = await credential.user.getIdToken()
   328→          }
   329→
   330→          const result = await verifyWithServer(idToken, '/api/auth/login')
   331→
   332→          if (result.success && result.user) {
   333→            setUserInStore(set, result.user)
   334→            set({ idToken })
   335→            return { success: true, user: result.user }
   336→          } else {
   337→            set({ isLoading: false, lastLoginError: result.error })
   338→            return { success: false, error: result.error }
   339→          }
   340→        } catch (err: any) {
   341→          console.error('Email login error:', err)
   342→          let errorMsg = 'Credenciais inválidas.'
   343→
   344→          if (err.code === 'auth/user-not-found' || err.code === 'auth/wrong-password' || err.code === 'auth/invalid-credential') {
   345→            errorMsg = 'Email ou senha incorretos.'
   346→          } else if (err.code === 'auth/too-many-requests') {
   347→            errorMsg = 'Muitas tentativas. Tente novamente mais tarde.'
   348→          } else if (err.code === 'auth/user-disabled') {
   349→            errorMsg = 'Conta desactivada. Contacte o suporte.'
   350→          } else if (err.code === 'auth/invalid-email') {
   351→            errorMsg = 'Email inválido.'
   352→          } else if (err.message?.includes('not configured')) {
   353→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   354→          }
   355→
   356→          set({ isLoading: false, lastLoginError: errorMsg })
   357→          return { success: false, error: errorMsg }
   358→        }
   359→      },
   360→
   361→      // ── Google Sign-In ──
   362→      // Web: Uses signInWithRedirect (more reliable than popup — avoids popup blockers)
   363→      // Native: Uses @capacitor-firebase/authentication GoogleSignIn
   364→      loginWithGoogle: async (): Promise<AuthResult> => {
   365→        set({ isLoading: true, lastLoginError: null })
   366→
   367→        try {
   368→          let idToken: string
   369→
   370→          if (shouldUseNativeAuth()) {
   371→            // ── Native path: @capacitor-firebase/authentication ──
   372→            // Uses Google Sign-In via Android system account picker / iOS ASAuthorization
   373→            const nativeResult = await nativeSignInWithGoogle()
   374→            idToken = nativeResult.idToken
   375→          } else {
   376→            // ── Web path: Firebase Web SDK signInWithRedirect ──
   377→            // Using redirect instead of popup because:
   378→            // 1. Popup blockers don't interfere
   379→            // 2. Works in Capacitor WebView (popup doesn't close immediately)
   380→            // 3. More reliable cross-browser compatibility
   381→            const { signInWithRedirect, auth, googleProvider } = await getFirebaseAuth()
   382→            await signInWithRedirect(auth, googleProvider)
   383→            // After redirect, the page will reload and getRedirectResult() will
   384→            // return the credential. The auth-context.tsx handles this on mount.
   385→            // Return a special "redirecting" result so the UI knows what's happening.
   386→            return { success: true }
   387→          }
   388→
   389→          const result = await verifyWithServer(idToken, '/api/auth/social')
   390→
   391→          if (result.success && result.user) {
   392→            setUserInStore(set, result.user)

   403→          if (err.code === 'auth/popup-closed-by-user') {
   404→            errorMsg = 'Popup fechado. Tente novamente.'
   405→          } else if (err.code === 'auth/popup-blocked') {
   406→            errorMsg = 'Popup bloqueado pelo navegador. Permita popups e tente novamente.'

   408→            errorMsg = 'Operação cancelada.'

   410→            errorMsg = 'Domínio não autorizado. Adicione este domínio no Firebase Console.'

   412→            errorMsg = 'Operação de redirecionamento em curso. Aguarde.'

   496→            errorMsg = 'Muitas tentativas. Tente novamente mais tarde.'
   497→          }

   504→      // ── Phone Authentication (Step 2: Verify OTP code) ──
   505→      verifyPhoneCode: async (verificationId: string, verificationCode: string): Promise<AuthResult> => {

   509→          const fb = await getFirebaseAuth()
   510→          const { auth, PhoneAuthProvider, signInWithCredential } = fb
   511→
   512→          // Create credential from verification ID + OTP code
   513→          const credential = PhoneAuthProvider.credential(verificationId, verificationCode)
   514→          const userCredential = await signInWithCredential(auth, credential)
   515→          const idToken = await userCredential.user.getIdToken()
   516→
   517→          // Verify with server to get Firestore profile
   518→          const result = await verifyWithServer(idToken, '/api/auth/social')
   519→
   520→          if (result.success && result.user) {
   521→            setUserInStore(set, result.user)
   522→            set({ idToken, phoneVerificationId: null })
   523→            return { success: true, user: result.user }
   524→          } else {
   525→            set({ isLoading: false, lastLoginError: result.error, phoneVerificationId: null })
   526→            return { success: false, error: result.error }
   527→          }
   528→        } catch (err: any) {
   529→          console.error('Phone verify error:', err)
   530→          let errorMsg = 'Código de verificação inválido.'
   531→
   532→          if (err.code === 'auth/invalid-verification-code') {
   533→            errorMsg = 'Código SMS incorreto. Verifique e tente novamente.'
   534→          } else if (err.code === 'auth/invalid-verification-id') {
   535→            errorMsg = 'Sessão de verificação expirada. Tente novamente.'
   536→          } else if (err.code === 'auth/code-expired') {
   537→            errorMsg = 'Código expirado. Solicite um novo código SMS.'
   538→          }
   539→
   540→          set({ isLoading: false, lastLoginError: errorMsg, phoneVerificationId: null })
   541→          return { success: false, error: errorMsg }
   542→        }
   543→      },
   544→
   545→      // ── Anonymous Sign-In ──
   546→      loginAnonymously: async (): Promise<AuthResult> => {
   547→        set({ isLoading: true, lastLoginError: null })
   548→
   549→        try {
   550→          let idToken: string
   551→
   552→          if (shouldUseNativeAuth()) {
   553→            // ── Native path: @capacitor-firebase/authentication ──
   554→            const nativeResult = await nativeSignInAnonymously()
   555→            idToken = nativeResult.idToken
   556→          } else {
   557→            // ── Web path: Firebase Web SDK ──
   558→            const { signInAnonymously, auth } = await getFirebaseAuth()
   559→            const credential = await signInAnonymously(auth)
   560→            idToken = await credential.user.getIdToken()
   561→          }
   562→
   563→          const result = await verifyWithServer(idToken, '/api/auth/anonymous')
   564→
   565→          if (result.success && result.user) {
   566→            setUserInStore(set, result.user)
   567→            set({ idToken })
   568→            return { success: true, user: result.user }
   569→          } else {
   570→            set({ isLoading: false, lastLoginError: result.error })
   571→            return { success: false, error: result.error }
   572→          }
   573→        } catch (err: any) {
   574→          console.error('Anonymous login error:', err)
   575→          set({ isLoading: false, lastLoginError: 'Falha no login como visitante.' })
   576→          return { success: false, error: 'Falha no login como visitante.' }
   577→        }
   578→      },
   579→
   580→      // ── Email/password registration ──
   581→      register: async (name: string, email: string, password: string, phone?: string): Promise<AuthResult> => {
   582→        set({ isLoading: true, lastRegisterError: null })
   583→
   584→        try {
   585→          const { createUserWithEmailAndPassword, auth, updateProfile } = await getFirebaseAuth()
   586→          const credential = await createUserWithEmailAndPassword(auth, email, password)
   587→
   588→          // Update display name in Firebase Auth
   589→          await updateProfile(credential.user, { displayName: name })
   590→
   591→          const idToken = await credential.user.getIdToken()
   592→
   593→          // Send to server to create Firestore profile
   594→          const res = await fetch(buildApiUrl('/api/auth/register'), {
   595→            method: 'POST',
   596→            headers: { 'Content-Type': 'application/json' },
   597→            body: JSON.stringify({ name, email, phone }),
   598→          })
   599→
   600→          const data = await res.json()
   601→
   602→          if (data.user) {
   603→            const userRole = mapRole(data.user.role)
   604→            const user: User = {
   605→              id: data.user.id || credential.user.uid,
   606→              name: data.user.name || name,
   607→              email: data.user.email || email,
   608→              role: userRole,
   609→              avatar: null,
   610→              phone: data.user.phone || phone || null,
   611→              authProvider: 'email',
   612→              isAnonymous: false,
   613→              emailVerified: false,
   614→            }
   615→            setUserInStore(set, user)
   616→            set({ idToken, lastRegisterError: null })
   617→            return { success: true, user }
   618→          } else {
   619→            // Firebase Auth user was created but Firestore profile failed
   620→            // Still set the user with basic info
   621→            const user: User = {
   622→              id: credential.user.uid,
   623→              name: name,
   624→              email: email,
   625→              role: 'user',
   626→              avatar: null,
   627→              authProvider: 'email',
   628→              isAnonymous: false,
   629→              emailVerified: false,
   630→            }
   631→            setUserInStore(set, user)
   632→            set({ idToken })
   633→            return { success: true, user }
   634→          }
   635→        } catch (err: any) {
   636→          console.error('Register error:', err)
   637→          let errorMsg = 'Falha ao criar conta.'
   638→
   639→          if (err.code === 'auth/email-already-in-use') {
   640→            errorMsg = 'Este email já está registado.'
   641→          } else if (err.code === 'auth/weak-password') {
   642→            errorMsg = 'Senha demasiado fraca. Use pelo menos 6 caracteres.'
   643→          } else if (err.code === 'auth/invalid-email') {
   644→            errorMsg = 'Email inválido.'
   645→          } else if (err.message?.includes('not configured')) {
   646→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   647→          }
   648→
   649→          set({ isLoading: false, lastRegisterError: errorMsg })
   650→          return { success: false, error: errorMsg }
   651→        }
   652→      },
   653→
   654→      logout: async () => {
   655→        try {
   656→          if (shouldUseNativeAuth()) {
   657→            await nativeSignOut()
   658→          } else {
   659→            const { signOut, auth } = await getFirebaseAuth()
   660→            await signOut(auth)
   661→          }
   662→        } catch (err) {
   663→          console.warn('Firebase signOut failed (non-critical):', err)
   664→        }
   665→
   666→        set({
   667→          user: null,
   668→          isAuthenticated: false,
   669→          isAdmin: false,
   670→          isPartner: false,
   671→          isUser: false,
   672→          isSuperAdmin: false,
   673→          idToken: null,
   674→          phoneVerificationId: null,
   675→          lastLoginError: null,
   676→          lastRegisterError: null,
   677→        })
   678→      },
   679→
   680→      setUser: (user: User) => {
   681→        setUserInStore(set, user)
   682→      },
   683→
   684→      updateAvatar: (avatar: string) => {
   685→        const currentUser = get().user
   686→        if (currentUser) {
   687→          set({ user: { ...currentUser, avatar } })
   688→        }
   689→      },
   690→
   691→      setIdToken: (token: string) => {
   692→        set({ idToken: token })
   693→      },
   694→    }),
   695→    {
   696→      name: 'carsai-auth',
   697→      partialize: (state) => ({
   698→        user: state.user,
   699→        isAuthenticated: state.isAuthenticated,
   700→        isAdmin: state.isAdmin,
   701→        isPartner: state.isPartner,
   702→        isUser: state.isUser,
   703→        isSuperAdmin: state.isSuperAdmin,
   704→        idToken: state.idToken,
   705→      }),
   706→      onRehydrateStorage: () => (state) => {
   707→        if (state) {
   708→          state.hasHydrated = true
   709→        }
   710→      },
   711→    }
   712→  )

   716→// Notification Store
   717→// ──────────────────────────────────────────────

   719→interface NotificationState {
   720→  notifications: Notification[]
   721→  unreadCount: number
   722→  addNotification: (type: NotificationType, title: string, message: string) => void
   723→  markAsRead: (id: string) => void
   724→  markAllAsRead: () => void
   725→  removeNotification: (id: string) => void
   726→  clearAll: () => void
   727→}
   728→
   729→export const useNotificationStore = create<NotificationState>((set, get) => ({
   730→  notifications: [],
   731→  unreadCount: 0,
   732→
   733→  addNotification: (type: NotificationType, title: string, message: string) => {
   734→    const id = `notif-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
   735→    const newNotification: Notification = {
   736→      id,
   737→      type,
   738→      title,
   739→      message,
   740→      read: false,
   741→      createdAt: new Date(),
   742→    }
   743→    const updated = [newNotification, ...get().notifications]
   744→    set({
   745→      notifications: updated,
   746→      unreadCount: updated.filter((n) => !n.read).length,
   747→    })
   748→  },
   749→
   750→  markAsRead: (id: string) => {
   751→    const updated = get().notifications.map((n) =>
   752→      n.id === id ? { ...n, read: true } : n
   753→    )
   754→    set({
   755→      notifications: updated,
   756→      unreadCount: updated.filter((n) => !n.read).length,
   757→    })
   758→  },
   759→
   760→  markAllAsRead: () => {
   761→    const updated = get().notifications.map((n) => ({ ...n, read: true }))
   762→    set({
   763→      notifications: updated,
   764→      unreadCount: 0,
   765→    })
   766→  },
   767→
   768→  removeNotification: (id: string) => {
   769→    const updated = get().notifications.filter((n) => n.id !== id)
   770→    set({
   771→      notifications: updated,
   772→      unreadCount: updated.filter((n) => !n.read).length,
   773→    })
   774→  },
   775→
   776→  clearAll: () => {
   777→    set({ notifications: [], unreadCount: 0 })
   778→  },
   779→}))
