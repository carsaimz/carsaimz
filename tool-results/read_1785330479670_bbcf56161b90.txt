    52→// ──────────────────────────────────────────────
    53→// Auth Store (Firebase Auth — client-side authentication)
    54→//
    55→// Flow:
    56→// 1. Client authenticates with Firebase Auth directly (email/password, Google, phone, anonymous)
    57→// 2. Firebase Auth returns a UserCredential with an ID token
    58→// 3. Client sends ID token to /api/auth/login, /api/auth/social, /api/auth/anonymous, or /api/auth/verify
    59→// 4. Server verifies token, returns Firestore profile (role, isActive, etc.)
    60→// 5. Store saves the combined profile
    61→//
    62→// Phone Auth Flow:
    63→// 1. Client calls loginWithPhone(phoneNumber, recaptchaContainerId)
    64→// 2. Firebase sends SMS OTP to the phone number
    65→// 3. User enters the OTP code
    66→// 4. Client calls verifyPhoneCode(verificationId, otpCode)
    67→// 5. Firebase verifies OTP and returns credential with ID token
    68→// 6. Client sends ID token to /api/auth/social (phone treated as social provider)
    69→// 7. Server verifies token and returns Firestore profile
    70→// ──────────────────────────────────────────────
    71→

    73→  success: boolean
    74→  error?: string

   119→function mapRole(roleData: any): UserRole {
   120→  if (typeof roleData === 'string') return roleData as UserRole
   121→  if (roleData && typeof roleData === 'object' && roleData.name) return roleData.name as UserRole
   122→  return 'user'

   125→/**
   126→ * Verify authentication with server API route.
   127→ * Uses apiFetch (instead of raw fetch) to handle:
   128→ *   - Capacitor CORS bypass (external URL + CORS middleware)
   129→ *   - HTML response detection/retry (static export fallback)
   130→ *   - Proper error handling without "Unexpected token '<'" crashes
   131→ *
   132→ * If server is unavailable, falls back to client-side Firestore or
   133→ * native auth result data.
   134→ *
   135→ * @param nativeResult Optional native auth result with user info,
   136→ *   used when the Web SDK's auth.currentUser is not yet synced.
   137→ */
   138→async function verifyWithServer(
   139→  idToken: string,
   140→  endpoint: string,
   141→  nativeResult?: NativeAuthResult
   142→): Promise<{ success: boolean; user?: User; error?: string }> {
   143→  try {
   144→    // Use apiFetch instead of raw fetch — it handles CORS, HTML detection,
   145→    // and automatic retry with external URL for Capacitor apps.
   146→    // Pass the relative path (e.g., '/api/auth/login') — apiFetch handles
   147→    // the full URL resolution internally (don't use buildApiUrl here).
   148→    const res = await apiFetch(endpoint, {
   149→      method: 'POST',
   150→      headers: { 'Content-Type': 'application/json' },
   151→      body: JSON.stringify({ idToken }),
   152→    })
   153→
   154→    // Check if response is JSON before parsing
   155→    const contentType = res.headers.get('content-type') || ''
   156→    if (!contentType.includes('application/json')) {
   157→      console.warn('[Auth] Server returned non-JSON response:', contentType)
   158→      // Not JSON — fall back
   159→      return await fallbackVerify(idToken, nativeResult)
   160→    }
   161→
   162→    if (res.ok) {
   163→      const data = await safeJson(res)
   164→      if (!data) return await fallbackVerify(idToken, nativeResult)
   165→
   166→      if (data.user) {
   167→        const userRole = mapRole(data.user.role)
   168→        const user: User = {
   169→          id: data.user.id,
   170→          name: data.user.name,
   171→          email: data.user.email,
   172→          role: userRole,
   173→          avatar: data.user.avatar || null,
   174→          phone: data.user.phone || null,
   175→          company: data.user.company || null,
   176→          bio: data.user.bio || null,
   177→          address: data.user.address || null,
   178→          authProvider: data.user.authProvider as AuthProvider || 'unknown',
   179→          isAnonymous: data.user.isAnonymous || false,
   180→          emailVerified: data.user.emailVerified || false,
   181→        }
   182→        return { success: true, user }
   183→      } else {
   184→        return { success: false, error: data.error || 'Autenticação falhou.' }
   185→      }
   186→    }
   187→
   188→    // Server returned non-OK status — try to parse error
   189→    let errorMsg = 'Autenticação falhou.'
   190→    try {
   191→      const errData = await safeJson(res)
   192→      if (errData) errorMsg = errData.error || errorMsg
   193→    } catch {}
   194→
   195→    // If the error is not a permanent auth error, try client-side fallback
   196→    if (res.status !== 401 && res.status !== 403) {
   197→      console.warn('[Auth] Server API error, falling back to client-side:', errorMsg)
   198→      return await fallbackVerify(idToken, nativeResult)
   199→    }
   200→
   201→    return { success: false, error: errorMsg }
   202→  } catch (err) {
   203→    // Network error — server not available (static export / Capacitor / CORS blocked)
   204→    console.warn('[Auth] Server API not reachable, falling back:', err)
   205→    return await fallbackVerify(idToken, nativeResult)
   206→  }
   207→}

   209→/**
   210→ * Smart fallback: tries client-side Firestore first (if auth.currentUser available),
   211→ * then uses native auth result data if provided.
   212→ * This prevents the "Utilizador não autenticado" error on mobile.
   213→ */
   214→async function fallbackVerify(
   215→  idToken: string,
   216→  nativeResult?: NativeAuthResult
   217→): Promise<{ success: boolean; user?: User; error?: string }> {
   218→  // Try client-side Firestore (needs auth.currentUser)
   219→  const clientResult = await verifyWithClientFirestore(idToken)
   220→  if (clientResult.success) return clientResult
   221→
   222→  // If client-side Firestore failed AND we have native auth result,
   223→  // use the native result data directly
   224→  if (nativeResult) {
   225→    return verifyWithNativeResult(nativeResult)
   226→  }
   227→
   228→  return clientResult
   229→}

   231→/**
   232→ * Create user profile from native auth result data.
   233→ * Used when both the server API and client-side Firestore are unavailable,
   234→ * but we have the native auth result with user info.
   235→ *
   236→ * This prevents the "Utilizador não autenticado" error on mobile.
   237→ */
   238→async function verifyWithNativeResult(nativeResult: NativeAuthResult): Promise<{ success: boolean; user?: User; error?: string }> {
   239→  try {
   240→    // Try to create/update the Firestore profile using client-side Firestore
   241→    // even if auth.currentUser is not yet synced
   242→    const { firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   243→
   244→    if (isFirebaseConfigured() && firestoreClient) {
   245→      const { doc, getDoc, setDoc } = await import('firebase/firestore')
   246→      const uid = nativeResult.uid
   247→
   248→      // Try to read existing profile
   249→      const userRef = doc(firestoreClient, 'users', uid)
   250→      const userSnap = await getDoc(userRef)
   251→
   252→      if (userSnap.exists()) {
   253→        const profile = userSnap.data()
   254→        const user: User = {
   255→          id: uid,
   256→          name: profile.name || nativeResult.displayName || 'Utilizador',
   257→          email: profile.email || nativeResult.email || null,
   258→          role: mapRole(profile.role),
   259→          avatar: profile.avatar || nativeResult.photoUrl || null,
   260→          phone: profile.phone || nativeResult.phoneNumber || null,
   261→          authProvider: nativeResult.providerId as AuthProvider || 'unknown',
   262→          isAnonymous: nativeResult.isAnonymous,
   263→          emailVerified: nativeResult.emailVerified,
   264→        }
   265→        return { success: true, user }
   266→      }
   267→
   268→      // Profile doesn't exist — create it from native auth result
   269→      const authProvider = nativeResult.providerId as AuthProvider
   270→      await setDoc(userRef, {
   271→        name: nativeResult.displayName || nativeResult.email?.split('@')[0] || 'Utilizador',
   272→        email: nativeResult.email || null,
   273→        phone: nativeResult.phoneNumber || null,
   274→        avatar: nativeResult.photoUrl || null,
   275→        company: null,
   276→        bio: null,
   277→        address: null,
   278→        role: 'user',
   279→        isActive: true,
   280→        emailVerified: nativeResult.emailVerified,
   281→        authProvider,
   282→        isAnonymous: nativeResult.isAnonymous,
   283→        createdAt: new Date(),
   284→        updatedAt: new Date(),
   285→      })
   286→
   287→      const user: User = {
   288→        id: uid,
   289→        name: nativeResult.displayName || 'Utilizador',
   290→        email: nativeResult.email,
   291→        role: 'user',
   292→        avatar: nativeResult.photoUrl || null,
   293→        phone: nativeResult.phoneNumber || null,
   294→        authProvider,
   295→        isAnonymous: nativeResult.isAnonymous,
   296→        emailVerified: nativeResult.emailVerified,
   297→      }
   298→      return { success: true, user }
   299→    }
   300→
   301→    // Firestore not available — use native result data as last resort
   302→    const user: User = {
   303→      id: nativeResult.uid,
   304→      name: nativeResult.displayName || 'Utilizador',
   305→      email: nativeResult.email,
   306→      role: 'user',
   307→      avatar: nativeResult.photoUrl || null,
   308→      phone: nativeResult.phoneNumber || null,
   309→      authProvider: nativeResult.providerId as AuthProvider || 'unknown',
   310→      isAnonymous: nativeResult.isAnonymous,
   311→      emailVerified: nativeResult.emailVerified,
   312→    }
   313→    return { success: true, user }
   314→  } catch (err) {
   315→    console.error('[Auth] Native result fallback error:', err)
   316→    // Last resort: use native result data directly without Firestore
   317→    const user: User = {
   318→      id: nativeResult.uid,
   319→      name: nativeResult.displayName || 'Utilizador',
   320→      email: nativeResult.email,
   321→      role: 'user',
   322→      avatar: nativeResult.photoUrl || null,
   323→      phone: nativeResult.phoneNumber || null,
   324→      authProvider: nativeResult.providerId as AuthProvider || 'unknown',
   325→      isAnonymous: nativeResult.isAnonymous,
   326→      emailVerified: nativeResult.emailVerified,
   327→    }
   328→    return { success: true, user }
   329→  }
   330→}

   332→/**
   333→ * Client-side Firestore fallback for authentication.
   334→ * Used when the server API routes are not available (static export / Capacitor).
   335→ * Reads the user profile from Firestore directly, or creates one if it doesn't exist.
   336→ */
   337→async function verifyWithClientFirestore(idToken: string): Promise<{ success: boolean; user?: User; error?: string }> {
   338→  try {
   339→    const { auth, firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   340→
   341→    if (!isFirebaseConfigured() || !firestoreClient) {
   342→      return { success: false, error: 'Firebase não configurado.' }
   343→    }
   344→
   345→    // Decode the ID token to get the Firebase user info
   346→    // We need to get the current Firebase Auth user
   347→    const currentUser = auth.currentUser
   348→    if (!currentUser) {
   349→      return { success: false, error: 'Utilizador não autenticado.' }
   350→    }
   351→
   352→    const uid = currentUser.uid
   353→    const { doc, getDoc, setDoc } = await import('firebase/firestore')
   354→
   355→    // Try to read existing profile from Firestore
   356→    const userRef = doc(firestoreClient, 'users', uid)
   357→    const userSnap = await getDoc(userRef)
   358→
   359→    if (userSnap.exists()) {
   360→      // Profile exists — return it
   361→      const profile = userSnap.data()
   362→      const userRole = mapRole(profile.role)
   363→      const user: User = {
   364→        id: uid,
   365→        name: profile.name || currentUser.displayName || 'Utilizador',
   366→        email: profile.email || currentUser.email || null,
   367→        role: userRole,
   368→        avatar: profile.avatar || currentUser.photoURL || null,
   369→        phone: profile.phone || currentUser.phoneNumber || null,
   370→        company: profile.company || null,
   371→        bio: profile.bio || null,
   372→        address: profile.address || null,
   373→        authProvider: (currentUser.providerData[0]?.providerId as AuthProvider) || 'unknown',
   374→        isAnonymous: currentUser.isAnonymous,
   375→        emailVerified: currentUser.emailVerified,
   376→      }
   377→      return { success: true, user }
   378→    }
   379→
   380→    // Profile doesn't exist — create it (first-time social/anonymous login)
   381→    const providerId = currentUser.providerData[0]?.providerId || 'unknown'
   382→    let authProvider: AuthProvider = 'unknown'
   383→    if (providerId === 'google.com') authProvider = 'google.com'
   384→    else if (providerId === 'github.com') authProvider = 'github.com'
   385→    else if (providerId === 'phone') authProvider = 'phone'
   386→    else if (providerId === 'anonymous') authProvider = 'anonymous'
   387→    else if (providerId === 'password') authProvider = 'email'
   388→
   389→    const newProfile = {
   390→      name: currentUser.displayName || 'Utilizador',
   391→      email: currentUser.email || null,
   392→      phone: currentUser.phoneNumber || null,
   393→      avatar: currentUser.photoURL || null,
   394→      company: null,
   395→      bio: null,
   396→      address: null,
   397→      role: 'user',
   398→      isActive: true,
   399→      emailVerified: currentUser.emailVerified,
   400→      authProvider,
   401→      isAnonymous: currentUser.isAnonymous,
   402→      createdAt: new Date(),
   403→      updatedAt: new Date(),
   404→    }
   405→
   406→    await setDoc(userRef, newProfile)
   407→
   408→    const user: User = {
   409→      id: uid,
   410→      name: newProfile.name,
   411→      email: newProfile.email,
   412→      role: 'user',
   413→      avatar: newProfile.avatar,
   414→      phone: newProfile.phone,
   415→      company: null,
   416→      bio: null,
   417→      address: null,
   418→      authProvider,
   419→      isAnonymous: currentUser.isAnonymous,
   420→      emailVerified: currentUser.emailVerified,
   421→    }
   422→    return { success: true, user }
   423→  } catch (err) {
   424→    console.error('[Auth] Client-side Firestore fallback error:', err)
   425→    return { success: false, error: 'Erro de autenticação. Tente novamente.' }
   426→  }
   427→}

   441→export const useAuthStore = create<AuthState>()(
   442→  persist(
   443→    (set, get) => ({
   444→      user: null,
   445→      isAuthenticated: false,
   446→      isAdmin: false,
   447→      isPartner: false,
   448→      isUser: false,
   449→      isSuperAdmin: false,
   450→      isLoading: false,
   451→      hasHydrated: false,
   452→      lastLoginError: null,
   453→      lastRegisterError: null,
   454→      idToken: null,
   455→      phoneVerificationId: null,
   456→
   457→      // ── Login with pre-obtained ID token ──
   458→      loginWithToken: async (idToken: string): Promise<AuthResult> => {
   459→        set({ isLoading: true, lastLoginError: null })
   460→        const result = await verifyWithServer(idToken, '/api/auth/login')
   461→
   462→        if (result.success && result.user) {
   463→          setUserInStore(set, result.user)
   464→          set({ idToken })
   465→          return { success: true, user: result.user }
   466→        } else {
   467→          set({ isLoading: false, lastLoginError: result.error })
   468→          return { success: false, error: result.error }
   469→        }
   470→      },
   471→
   472→      // ── Email/password login via Firebase Auth client ──
   473→      loginWithEmailPassword: async (email: string, password: string): Promise<AuthResult> => {
   474→        set({ isLoading: true, lastLoginError: null })
   475→
   476→        try {
   477→          let idToken: string
   478→          let nativeAuthResult: NativeAuthResult | undefined
   479→
   480→          if (shouldUseNativeAuth()) {
   481→            // ── Native path: @capacitor-firebase/authentication ──
   482→            const nativeResult = await nativeSignInWithEmailPassword(email, password)
   483→            idToken = nativeResult.idToken
   484→            nativeAuthResult = nativeResult // Pass to fallback if server fails
   485→          } else {
   486→            // ── Web path: Firebase Web SDK ──
   487→            const { signInWithEmailAndPassword, auth } = await getFirebaseAuth()
   488→            const credential = await signInWithEmailAndPassword(auth, email, password)
   489→            idToken = await credential.user.getIdToken()
   490→          }
   491→
   492→          const result = await verifyWithServer(idToken, '/api/auth/login', nativeAuthResult)
   493→
   494→          if (result.success && result.user) {
   495→            setUserInStore(set, result.user)
   496→            set({ idToken })
   497→            return { success: true, user: result.user }
   498→          } else {
   499→            set({ isLoading: false, lastLoginError: result.error })
   500→            return { success: false, error: result.error }
   501→          }
   502→        } catch (err: any) {
   503→          console.error('Email login error:', err)
   504→          let errorMsg = 'Credenciais inválidas.'
   505→
   506→          if (err.code === 'auth/user-not-found' || err.code === 'auth/wrong-password' || err.code === 'auth/invalid-credential') {
   507→            errorMsg = 'Email ou senha incorretos.'
   508→          } else if (err.code === 'auth/too-many-requests') {
   509→            errorMsg = 'Muitas tentativas. Tente novamente mais tarde.'
   510→          } else if (err.code === 'auth/user-disabled') {
   511→            errorMsg = 'Conta desactivada. Contacte o suporte.'
   512→          } else if (err.code === 'auth/invalid-email') {
   513→            errorMsg = 'Email inválido.'
   514→          } else if (err.message?.includes('not configured')) {
   515→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   516→          }
   517→
   518→          set({ isLoading: false, lastLoginError: errorMsg })
   519→          return { success: false, error: errorMsg }
   520→        }
   521→      },
   522→
   523→      // ── Google Sign-In ──
   524→      // Web: Uses signInWithRedirect (more reliable than popup — avoids popup blockers)
   525→      // Native: Uses @capacitor-firebase/authentication GoogleSignIn
   526→      loginWithGoogle: async (): Promise<AuthResult> => {
   527→        set({ isLoading: true, lastLoginError: null })
   528→
   529→        try {
   530→          let idToken: string
   531→          let nativeAuthResult: NativeAuthResult | undefined
   532→          const isNative = shouldUseNativeAuth()
   533→          console.log('[Auth] loginWithGoogle — isNative:', isNative)
   534→
   535→          if (isNative) {
   536→            // ── Native path: @capacitor-firebase/authentication ──
   537→            // Uses Google Sign-In via Android system account picker / iOS ASAuthorization
   538→            console.log('[Auth] Calling nativeSignInWithGoogle...')
   539→            const nativeResult = await nativeSignInWithGoogle()
   540→            idToken = nativeResult.idToken
   541→            nativeAuthResult = nativeResult
   542→            console.log('[Auth] Native Google sign-in succeeded, got ID token')
   543→          } else {
   544→            // ── Web path: Firebase Web SDK signInWithRedirect ──
   545→            // Using redirect instead of popup because:
   546→            // 1. Popup blockers don't interfere
   547→            // 2. Works in Capacitor WebView (popup doesn't close immediately)
   548→            // 3. More reliable cross-browser compatibility
   549→            const { signInWithRedirect, auth, googleProvider } = await getFirebaseAuth()
   550→            await signInWithRedirect(auth, googleProvider)
   551→            // After redirect, the page will reload and getRedirectResult() will
   552→            // return the credential. The auth-context.tsx handles this on mount.
   553→            // Return a special "redirecting" result so the UI knows what's happening.
   554→            return { success: true }
   555→          }
   556→

   756→      // ── Email/password registration ──
   757→      register: async (name: string, email: string, password: string, phone?: string): Promise<AuthResult> => {
   758→        set({ isLoading: true, lastRegisterError: null })
   759→
   760→        try {
   761→          const { createUserWithEmailAndPassword, auth, updateProfile } = await getFirebaseAuth()
   762→          const credential = await createUserWithEmailAndPassword(auth, email, password)
   763→
   764→          // Update display name in Firebase Auth
   765→          await updateProfile(credential.user, { displayName: name })
   766→
   767→          const idToken = await credential.user.getIdToken()
   768→
   769→          // Try server API first to create Firestore profile
   770→          let serverProfileCreated = false
   771→          try {
   772→            const res = await apiFetch('/api/auth/register', {
   773→              method: 'POST',
   774→              headers: { 'Content-Type': 'application/json' },
   775→              body: JSON.stringify({ name, email, phone }),
   776→            })
   777→
   778→            // Check if response is JSON (not HTML from static export)
   779→            const contentType = res.headers.get('content-type') || ''
   780→            if (contentType.includes('application/json')) {
   781→              const data = await safeJson(res)
   782→              if (!data) return { success: false, error: 'Server returned non-JSON response' }
   783→
   784→              if (data.user) {
   785→                const userRole = mapRole(data.user.role)
   786→                const user: User = {
   787→                  id: data.user.id || credential.user.uid,
   788→                  name: data.user.name || name,
   789→                  email: data.user.email || email,
   790→                  role: userRole,
   791→                  avatar: null,
   792→                  phone: data.user.phone || phone || null,
   793→                  authProvider: 'email',
   794→                  isAnonymous: false,
   795→                  emailVerified: false,
   796→                }
   797→                setUserInStore(set, user)
   798→                set({ idToken, lastRegisterError: null })
   799→                return { success: true, user }
   800→              }
   801→            }
   802→            // If not JSON or no user data, fall through to client-side Firestore
   803→          } catch (apiErr) {
   804→            console.warn('[Auth] Server API unavailable for registration, falling back to client-side Firestore:', apiErr)
   805→          }
   806→
   807→          // Fallback: Create Firestore profile directly using client SDK
   808→          if (!serverProfileCreated) {
   809→            try {
   810→              const { firestoreClient } = await import('@/lib/firebase-client')
   811→              if (firestoreClient) {
   812→                const { doc, setDoc, getDoc, getDocs, collection, query, where } = await import('firebase/firestore')
   813→                const uid = credential.user.uid
   814→
   815→                // Find the "user" role
   816→                let roleId: string | null = null
   817→                try {
   818→                  const rolesRef = collection(firestoreClient, 'roles')
   819→                  const q = query(rolesRef, where('name', '==', 'user'))
   820→                  const roleSnap = await getDocs(q)
   821→                  if (!roleSnap.empty) {
   822→                    roleId = roleSnap.docs[0].id
   823→                  }
   824→                } catch (e) {
   825→                  console.warn('[Auth] Could not find user role:', e)
   826→                }
   827→
   828→                // Create user profile in Firestore
   829→                const userRef = doc(firestoreClient, 'users', uid)
   830→                await setDoc(userRef, {
   831→                  name: name || email.split('@')[0],
   832→                  email: email.toLowerCase().trim(),
   833→                  phone: phone || null,
   834→                  company: null,
   835→                  avatar: null,
   836→                  bio: null,
   837→                  address: null,
   838→                  roleId: roleId,
   839→                  isActive: true,
   840→                  emailVerified: false,
   841→                  authProvider: 'email',
   842→                  createdAt: new Date(),
   843→                  updatedAt: new Date(),
   844→                })
   845→
   846→                const user: User = {
   847→                  id: uid,
   848→                  name: name,
   849→                  email: email,
   850→                  role: 'user',
   851→                  avatar: null,
   852→                  phone: phone || null,
   853→                  authProvider: 'email',
   854→                  isAnonymous: false,
   855→                  emailVerified: false,
   856→                }
   857→                setUserInStore(set, user)
   858→                set({ idToken, lastRegisterError: null })
   859→                return { success: true, user }
   860→              }
   861→            } catch (fsErr) {
   862→              console.error('[Auth] Client-side Firestore profile creation failed:', fsErr)
   863→            }
   864→          }
   865→
   866→          // Last resort: Firebase Auth user was created but Firestore profile failed
   867→          // Still set the user with basic info so the UI works
   868→          const user: User = {
   869→            id: credential.user.uid,
   870→            name: name,
   871→            email: email,
   872→            role: 'user',
   873→            avatar: null,
   874→            authProvider: 'email',
   875→            isAnonymous: false,
   876→            emailVerified: false,
   877→          }
   878→          setUserInStore(set, user)
   879→          set({ idToken })
   880→          return { success: true, user }
   881→        } catch (err: any) {
   882→          console.error('Register error:', err)
   883→          let errorMsg = 'Falha ao criar conta.'
   884→
   885→          if (err.code === 'auth/email-already-in-use') {
   886→            errorMsg = 'Este email já está registado.'
   887→          } else if (err.code === 'auth/weak-password') {
   888→            errorMsg = 'Senha demasiado fraca. Use pelo menos 6 caracteres.'
   889→          } else if (err.code === 'auth/invalid-email') {
   890→            errorMsg = 'Email inválido.'
   891→          } else if (err.message?.includes('not configured')) {
   892→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   893→          }
   894→
   895→          set({ isLoading: false, lastRegisterError: errorMsg })
   896→          return { success: false, error: errorMsg }
   897→        }
   898→      },
   899→
   900→      logout: async () => {
   901→        try {
   902→          if (shouldUseNativeAuth()) {
   903→            await nativeSignOut()
   904→          } else {
   905→            const { signOut, auth } = await getFirebaseAuth()

   912→        set({
   913→          user: null,
   914→          isAuthenticated: false,
   915→          isAdmin: false,
   916→          isPartner: false,
   917→          isUser: false,
   918→          isSuperAdmin: false,

   920→          phoneVerificationId: null,

   924→      },
   925→
   926→      setUser: (user: User) => {

   929→
   930→      updateAvatar: (avatar: string) => {

   935→      },
   936→
   937→      setIdToken: (token: string) => {

   941→    {
   942→      name: 'carsai-auth',

   965→interface NotificationState {
   966→  notifications: Notification[]
   967→  unreadCount: number
   968→  addNotification: (type: NotificationType, title: string, message: string) => void
   969→  markAsRead: (id: string) => void
   970→  markAllAsRead: () => void
   971→  removeNotification: (id: string) => void
   972→  clearAll: () => void
   973→}
   974→
   975→export const useNotificationStore = create<NotificationState>((set, get) => ({
   976→  notifications: [],
   977→  unreadCount: 0,
   978→
   979→  addNotification: (type: NotificationType, title: string, message: string) => {
   980→    const id = `notif-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
   981→    const newNotification: Notification = {
   982→      id,
   983→      type,
   984→      title,
   985→      message,
   986→      read: false,
   987→      createdAt: new Date(),
   988→    }
   989→    const updated = [newNotification, ...get().notifications]
   990→    set({
   991→      notifications: updated,
   992→      unreadCount: updated.filter((n) => !n.read).length,
   993→    })
   994→  },
   995→
   996→  markAsRead: (id: string) => {
   997→    const updated = get().notifications.map((n) =>
   998→      n.id === id ? { ...n, read: true } : n
   999→    )
  1000→    set({
  1001→      notifications: updated,
  1002→      unreadCount: updated.filter((n) => !n.read).length,
  1003→    })
  1004→  },
  1005→
  1006→  markAllAsRead: () => {
  1007→    const updated = get().notifications.map((n) => ({ ...n, read: true }))
  1008→    set({
  1009→      notifications: updated,
  1010→      unreadCount: 0,
  1011→    })
  1012→  },
  1013→
  1014→  removeNotification: (id: string) => {
  1015→    const updated = get().notifications.filter((n) => n.id !== id)
  1016→    set({
  1017→      notifications: updated,
  1018→      unreadCount: updated.filter((n) => !n.read).length,
  1019→    })
  1020→  },
  1021→
  1022→  clearAll: () => {
  1023→    set({ notifications: [], unreadCount: 0 })
  1024→  },
  1025→}))
