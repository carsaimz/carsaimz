     5→     5→import {

     9→     9→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    10→    10→import { Badge } from '@/components/ui/badge';

    14→    14→import { Label } from '@/components/ui/label';
    15→    15→import { Separator } from '@/components/ui/separator';
    16→    16→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    17→    17→import { Avatar, AvatarFallback } from '@/components/ui/avatar';

    25→    25→import {

    27→    27→  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
    28→    28→  AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger,
    29→    29→} from '@/components/ui/alert-dialog';
    30→    30→import { useLanguage } from '@/contexts/language-context';
    31→    31→import { useAuthStore } from '@/lib/store';
    32→    32→import { apiFetch, safeJson } from '@/lib/api-fetch';
    33→    33→import { useToast } from '@/hooks/use-toast';
    34→    34→

    37→    37→
    38→    38→interface UserData {
    39→    39→  id: string;
    40→    40→  name: string;
    41→    41→  email: string;
    42→    42→  phone?: string;
    43→    43→  role: string;
    44→    44→  isActive?: boolean;
    45→    45→  createdAt: string;
    46→    46→}
    47→    47→
    48→    48→interface CreateUserData {
    49→    49→  name: string;
    50→    50→  email: string;
    51→    51→  password: string;
    52→    52→  role: string;
    53→    53→  phone: string;
    54→    54→}
    55→    55→
    56→    56→interface EditUserData {
    57→    57→  name: string;
    58→    58→  email: string;
    59→    59→  phone: string;
    60→    60→  role: string;
    61→    61→  isActive: boolean;
    62→    62→  bio: string;
    63→    63→  company: string;
    64→    64→  address: string;
    65→    65→}
    66→    66→
    67→    67→const PAGE_SIZE = 10;
    68→    68→
    69→    69→export function AdminUsers() {
    70→    70→  const { t, formatDate } = useLanguage();
    71→    71→  const { user: currentUser } = useAuthStore();
    72→    72→  const { toast } = useToast();
    73→    73→
    74→    74→  // ── State ──
    75→    75→  const [users, setUsers] = useState<UserData[]>([]);
    76→    76→  const [loading, setLoading] = useState(true);
    77→    77→  const [error, setError] = useState<string | null>(null);
    78→    78→  const [searchQuery, setSearchQuery] = useState('');
    79→    79→  const [currentPage, setCurrentPage] = useState(1);
    80→    80→  const [totalPages, setTotalPages] = useState(1);
    81→    81→  const [total, setTotal] = useState(0);
    82→    82→
    83→    83→  // ── Create dialog ──
    84→    84→  const [createOpen, setCreateOpen] = useState(false);
    85→    85→  const [createData, setCreateData] = useState<CreateUserData>({
    86→    86→    name: '', email: '', password: '', role: 'user', phone: '',
    87→    87→  });
    88→    88→  const [creating, setCreating] = useState(false);
    89→    89→
    90→    90→  // ── Edit dialog ──
    91→    91→  const [editOpen, setEditOpen] = useState(false);
    92→    92→  const [editUser, setEditUser] = useState<UserData | null>(null);
    93→    93→  const [editData, setEditData] = useState<EditUserData>({
    94→    94→    name: '', email: '', phone: '', role: 'user', isActive: true, bio: '', company: '', address: '',
    95→    95→  });
    96→    96→  const [editing, setEditing] = useState(false);
    97→    97→
    98→    98→  // ── Deactivate ──
    99→    99→  const [deactivatingId, setDeactivatingId] = useState<string | null>(null);
   100→   100→
   101→   101→  // ── Fetch users ──
   102→   102→  const fetchUsers = useCallback(async (page: number, search: string) => {
   103→   103→    setLoading(true);
   104→   104→    setError(null);
   105→   105→    try {
   106→   106→      const params = new URLSearchParams({
   107→   107→        limit: String(PAGE_SIZE),
   108→   108→        page: String(page),
   109→   109→      });
   110→   110→      if (search) params.set('search', search);
   111→   111→
   112→   112→      const res = await apiFetch(`/api/admin/users?${params.toString()}`);
   113→   113→      const data = await safeJson(res);
   114→   114→      if (!data) { setError(t('common.serverNonJson')); return; }
   115→   115→      if (data.success) {
   116→   116→        setUsers(data.data || []);
   117→   117→        setTotal(data.meta?.total || 0);
   118→   118→        setTotalPages(data.meta?.totalPages || 1);
   119→   119→      } else {
   120→   120→        setError(data.message || t('admin.usersLoadFailed'));
   121→   121→      }
   122→   122→    } catch (err: any) {
   123→   123→      setError(err.message);
   124→   124→    } finally {
   125→   125→      setLoading(false);
   126→   126→    }
   127→   127→  }, []);
   128→   128→
   129→   129→  useEffect(() => {
   130→   130→    fetchUsers(currentPage, searchQuery);
   131→   131→  }, [currentPage, searchQuery, fetchUsers, currentUser?.id]);
   132→   132→
   133→   133→  // ── Search debounce ──
   134→   134→  const [searchInput, setSearchInput] = useState('');
   135→   135→  useEffect(() => {
   136→   136→    const timer = setTimeout(() => {
   137→   137→      setSearchQuery(searchInput);
   138→   138→      setCurrentPage(1);
   139→   139→    }, 300);
   140→   140→    return () => clearTimeout(timer);
   141→   141→  }, [searchInput]);
   142→   142→
   143→   143→  // ── Create user ──
   144→   144→  const handleCreate = async () => {
   145→   145→    setCreating(true);
   146→   146→    try {
   147→   147→      const res = await apiFetch('/api/admin/users', {
   148→   148→        method: 'POST',
   149→   149→        headers: { 'Content-Type': 'application/json' },
   150→   150→        body: JSON.stringify(createData),
   151→   151→      });
   152→   152→      const data = await safeJson(res);
   153→   153→      if (!data) { toast({ title: t('admin.error'), description: t('common.serverNonJson'), variant: 'destructive' }); return; }
   154→   154→      if (data.success) {
   155→   155→        toast({ title: t('admin.userCreated'), description: t('admin.createUserDesc') });
   156→   156→        setCreateOpen(false);
   157→   157→        setCreateData({ name: '', email: '', password: '', role: 'user', phone: '' });
   158→   158→        fetchUsers(currentPage, searchQuery);
   159→   159→      } else {
   160→   160→        toast({ title: t('admin.error'), description: data.message, variant: 'destructive' });
   161→   161→      }
   162→   162→    } catch (err: any) {
   163→   163→      toast({ title: t('admin.error'), description: err.message, variant: 'destructive' });
   164→   164→    } finally {
   165→   165→      setCreating(false);
   166→   166→    }
   167→   167→  };
   168→   168→
   169→   169→  // ── Edit user ──
   170→   170→  const handleEdit = async () => {
   171→   171→    if (!editUser) return;
   172→   172→    setEditing(true);
   173→   173→    try {
   174→   174→      const res = await apiFetch(`/api/admin/users?id=${editUser.id}`, {
   175→   175→        method: 'PUT',
   176→   176→        headers: { 'Content-Type': 'application/json' },
   177→   177→        body: JSON.stringify(editData),
   178→   178→      });
   179→   179→      const data = await safeJson(res);
   180→   180→      if (!data) { toast({ title: t('admin.error'), description: t('common.serverNonJson'), variant: 'destructive' }); return; }
   181→   181→      if (data.success) {
   182→   182→        toast({ title: t('admin.userUpdated'), description: t('admin.editUserDesc') });
   183→   183→        setEditOpen(false);
   184→   184→        setEditUser(null);
   185→   185→        fetchUsers(currentPage, searchQuery);
   186→   186→      } else {
   187→   187→        toast({ title: t('admin.error'), description: data.message, variant: 'destructive' });
   188→   188→      }
   189→   189→    } catch (err: any) {
   190→   190→      toast({ title: t('admin.error'), description: err.message, variant: 'destructive' });
   191→   191→    } finally {
   192→   192→      setEditing(false);
   193→   193→    }
   194→   194→  };
   195→   195→
   196→   196→  // ── Deactivate user ──
   197→   197→  const handleDeactivate = async (userId: string) => {
   198→   198→    setDeactivatingId(userId);
   199→   199→    try {
   200→   200→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   201→   201→        method: 'DELETE',
   202→   202→      });
   203→   203→      const data = await safeJson(res);
   204→   204→      if (!data) { toast({ title: t('admin.error'), description: t('common.serverNonJson'), variant: 'destructive' }); return; }
   205→   205→      if (data.success) {
   206→   206→        toast({ title: t('admin.userDeactivated') });
   207→   207→        fetchUsers(currentPage, searchQuery);
   208→   208→      } else {
   209→   209→        toast({ title: t('admin.error'), description: data.message, variant: 'destructive' });
   210→   210→      }
   211→   211→    } catch (err: any) {
   212→   212→      toast({ title: t('admin.error'), description: err.message, variant: 'destructive' });
   213→   213→    } finally {
   214→   214→      setDeactivatingId(null);
   215→   215→    }
   216→   216→  };
   217→   217→
   218→   218→  // ── Activate user (re-enable via PUT) ──
   219→   219→  const handleActivate = async (userId: string) => {
   220→   220→    setDeactivatingId(userId);
   221→   221→    try {
   222→   222→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   223→   223→        method: 'PUT',
   224→   224→        headers: { 'Content-Type': 'application/json' },
   225→   225→        body: JSON.stringify({ isActive: true }),
   226→   226→      });
   227→   227→      const data = await safeJson(res);
   228→   228→      if (!data) { toast({ title: t('admin.error'), description: t('common.serverNonJson'), variant: 'destructive' }); return; }
   229→   229→      if (data.success) {
   230→   230→        toast({ title: t('admin.userUpdated') });
   231→   231→        fetchUsers(currentPage, searchQuery);
   232→   232→      } else {
   233→   233→        toast({ title: t('admin.error'), description: data.message, variant: 'destructive' });
   234→   234→      }
   235→   235→    } catch (err: any) {
   236→   236→      toast({ title: t('admin.error'), description: err.message, variant: 'destructive' });
   237→   237→    } finally {
   238→   238→      setDeactivatingId(null);
   239→   239→    }
   240→   240→  };
   241→   241→
   242→   242→  // ── Role badge ──
   243→   243→  const roleBadge = (role: string) => {
   244→   244→    switch (role) {
   245→   245→      case 'super_admin': return <Badge className="bg-purple-100 text-purple-700 border-purple-200">{t('admin.roleSuperAdmin')}</Badge>;
   246→   246→      case 'admin': return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('admin.roleAdmin')}</Badge>;
   247→   247→      case 'partner': return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">{t('admin.rolePartner')}</Badge>;
   248→   248→      case 'user': return <Badge className="bg-blue-100 text-blue-700 border-blue-200">{t('admin.roleUser')}</Badge>;
   249→   249→      default: return <Badge variant="outline">{role}</Badge>;
   250→   250→    }
   251→   251→  };
   252→   252→
   253→   253→  // ── Status badge ──
   254→   254→  const statusBadge = (isActive?: boolean) => {
   255→   255→    if (isActive === false) {
   256→   256→      return <Badge className="bg-red-100 text-red-700 border-red-200">{t('admin.userInactive')}</Badge>;
   257→   257→    }
   258→   258→    return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('admin.userActive')}</Badge>;
   259→   259→  };
   260→   260→
   261→   261→  // ── Open edit dialog ──
   262→   262→  const openEdit = (u: UserData) => {
   263→   263→    setEditUser(u);
   264→   264→    setEditData({
   265→   265→      name: u.name,
   266→   266→      email: u.email,
   267→   267→      phone: u.phone || '',
   268→   268→      role: u.role,
   269→   269→      isActive: u.isActive !== false,
   270→   270→      bio: (u as any).bio || '',
   271→   271→      company: (u as any).company || '',
   272→   272→      address: (u as any).address || '',
   273→   273→    });
   274→   274→    setEditOpen(true);
   275→   275→  };
   276→   276→
   277→   277→  // ── Error state ──
   278→   278→  if (error && !loading) {
   279→   279→    return (
   280→   280→      <motion.div variants={containerVariants} initial="hidden" animate="visible">
   281→   281→        <Card className="border-l-4 border-l-red-500">
   282→   282→          <CardContent className="p-6">
   283→   283→            <div className="flex items-center gap-3">
   284→   284→              <AlertCircle className="h-6 w-6 text-red-500" />
   285→   285→              <p>{error}</p>
   286→   286→            </div>
   287→   287→            <Button className="mt-4" onClick={() => fetchUsers(currentPage, searchQuery)}>
   288→   288→              <RefreshCw className="mr-2 h-4 w-4" />
   289→   289→              {t('common.retry')}
   290→   290→            </Button>
   291→   291→          </CardContent>
   292→   292→        </Card>
   293→   293→      </motion.div>
   294→   294→    );
   295→   295→  }
   296→   296→
   297→   297→  return (
   298→   298→    <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-6">
   299→   299→      {/* ── Header ── */}
   300→   300→      <motion.div variants={itemVariants}>
   301→   301→        <div className="flex items-center justify-between">
   302→   302→          <div>
   303→   303→            <h2 className="text-2xl font-bold flex items-center gap-2">
   304→   304→              <Users className="h-6 w-6 text-emerald-600" />
   305→   305→              {t('admin.users')}
   306→   306→            </h2>
   307→   307→            <p className="text-muted-foreground mt-1">
   308→   308→              {total} {t('admin.usersTotal')}
   309→   309→            </p>
   310→   310→          </div>
   311→   311→
   312→   312→          {/* ── Create user button ── */}
   313→   313→          <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   314→   314→            <DialogTrigger asChild>
   315→   315→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">
   316→   316→                <UserPlus className="h-4 w-4" />
   317→   317→                {t('admin.createUser')}
   318→   318→              </Button>
   319→   319→            </DialogTrigger>
   320→   320→            <DialogContent className="sm:max-w-[425px]">
   321→   321→              <DialogHeader>
   322→   322→                <DialogTitle>{t('admin.createUser')}</DialogTitle>
   323→   323→                <DialogDescription>{t('admin.createUserDesc')}</DialogDescription>
   324→   324→              </DialogHeader>
   325→   325→              <div className="grid gap-4 py-4">
   326→   326→                <div className="grid gap-2">
   327→   327→                  <Label htmlFor="create-name">{t('admin.userName')}</Label>
   328→   328→                  <Input
   329→   329→                    id="create-name"
   330→   330→                    value={createData.name}
   331→   331→                    onChange={(e) => setCreateData(prev => ({ ...prev, name: e.target.value }))}
   332→   332→                    className="focus-visible:ring-emerald-500"
   333→   333→                  />
   334→   334→                </div>
   335→   335→                <div className="grid gap-2">
   336→   336→                  <Label htmlFor="create-email">{t('admin.userEmail')}</Label>
   337→   337→                  <Input
   338→   338→                    id="create-email"
   339→   339→                    type="email"
   340→   340→                    value={createData.email}
   341→   341→                    onChange={(e) => setCreateData(prev => ({ ...prev, email: e.target.value }))}
   342→   342→                    className="focus-visible:ring-emerald-500"
   343→   343→                  />
   344→   344→                </div>
   345→   345→                <div className="grid gap-2">
   346→   346→                  <Label htmlFor="create-password">{t('admin.userPassword')}</Label>
   347→   347→                  <Input
   348→   348→                    id="create-password"
   349→   349→                    type="password"
   350→   350→                    value={createData.password}
   351→   351→                    onChange={(e) => setCreateData(prev => ({ ...prev, password: e.target.value }))}
   352→   352→                    className="focus-visible:ring-emerald-500"
   353→   353→                  />
   354→   354→                </div>
   355→   355→                <div className="grid gap-2">
   356→   356→                  <Label htmlFor="create-phone">{t('admin.userPhone')}</Label>
   357→   357→                  <Input
   358→   358→                    id="create-phone"
   359→   359→                    value={createData.phone}
   360→   360→                    onChange={(e) => setCreateData(prev => ({ ...prev, phone: e.target.value }))}
   361→   361→                    className="focus-visible:ring-emerald-500"
   362→   362→                  />
   363→   363→                </div>
   364→   364→                <div className="grid gap-2">
   365→   365→                  <Label>{t('admin.userRole')}</Label>
   366→   366→                  <Select
   367→   367→                    value={createData.role}
   368→   368→                    onValueChange={(val) => setCreateData(prev => ({ ...prev, role: val }))}
   369→   369→                  >
   370→   370→                    <SelectTrigger className="focus:ring-emerald-500">
   371→   371→                      <SelectValue />
   372→   372→                    </SelectTrigger>
   373→   373→                    <SelectContent>
   374→   374→                      <SelectItem value="user">{t('admin.roleUser')}</SelectItem>
   375→   375→                      <SelectItem value="partner">{t('admin.rolePartner')}</SelectItem>
   376→   376→                      <SelectItem value="admin">{t('admin.roleAdmin')}</SelectItem>
   377→   377→                      <SelectItem value="super_admin">{t('admin.roleSuperAdmin')}</SelectItem>
   378→   378→                    </SelectContent>
   379→   379→                  </Select>
   380→   380→                </div>
   381→   381→              </div>
   382→   382→              <DialogFooter>
   383→   383→                <Button variant="outline" onClick={() => setCreateOpen(false)}>
   384→   384→                  {t('admin.cancel')}
   385→   385→                </Button>
   386→   386→                <Button
   387→   387→                  className="bg-emerald-600 hover:bg-emerald-700 text-white"
   388→   388→                  onClick={handleCreate}
   389→   389→                  disabled={creating || !createData.name || !createData.email || !createData.password}
   390→   390→                >
   391→   391→                  {creating ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Plus className="h-4 w-4 mr-2" />}
   392→   392→                  {t('admin.createUser')}
   393→   393→                </Button>
   394→   394→              </DialogFooter>
   395→   395→            </DialogContent>
   396→   396→          </Dialog>
   397→   397→        </div>
   398→   398→      </motion.div>
   399→   399→
   400→   400→      {/* ── Search ── */}
   401→   401→      <motion.div variants={itemVariants}>
   402→   402→        <div className="relative">
   403→   403→          <Search className="absolute left-3 top-1/2 h-4 w-4 text-muted-foreground" />
   404→   404→          <Input
   405→   405→            placeholder={t('admin.searchUsers')}
   406→   406→            value={searchInput}
   407→   407→            onChange={(e) => setSearchInput(e.target.value)}
   408→   408→            className="pl-10 focus-visible:ring-emerald-500"
   409→   409→          />
   410→   410→        </div>
   411→   411→      </motion.div>
   412→   412→
   413→   413→      {/* ── Users table ── */}
   414→   414→      <motion.div variants={itemVariants}>
   415→   415→        <Card>
   416→   416→          <CardContent className="p-0">
   417→   417→            {loading ? (
   418→   418→              <div className="p-6 space-y-4">
   419→   419→                {[1, 2, 3].map(i => (
   420→   420→                  <Skeleton key={i} className="h-12 w-full rounded" />
   421→   421→                ))}
   422→   422→              </div>
   423→   423→            ) : users.length === 0 ? (
   424→   424→              <div className="text-center py-12">
   425→   425→                <Users className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   426→   426→                <p className="text-muted-foreground">{t('common.noData')}</p>
   427→   427→              </div>
   428→   428→            ) : (
   429→   429→              <Table>
   430→   430→                <TableHeader>
   431→   431→                  <TableRow className="bg-emerald-50/50">
   432→   432→                    <TableHead>{t('admin.userName')}</TableHead>
   433→   433→                    <TableHead>{t('admin.userEmail')}</TableHead>
   434→   434→                    <TableHead>{t('admin.userPhone')}</TableHead>
   435→   435→                    <TableHead>{t('admin.userRole')}</TableHead>
   436→   436→                    <TableHead>{t('admin.userStatus')}</TableHead>
   437→   437→                    <TableHead>{t('admin.date')}</TableHead>
   438→   438→                    <TableHead className="text-right">{t('admin.actions')}</TableHead>
   439→   439→                  </TableRow>
   440→   440→                </TableHeader>
   441→   441→                <TableBody>
   442→   442→                  {users.map((u) => (
