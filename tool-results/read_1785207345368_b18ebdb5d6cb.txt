    57→function loadMessages(): ChatMessage[] {
    58→  try {
    59→    const stored = localStorage.getItem(STORAGE_KEY);
    60→    if (stored) {
    61→      const parsed = JSON.parse(stored);
    62→      return parsed.slice(-MAX_MESSAGES);
    63→    }
    64→  } catch {
    65→    // Ignore parse errors
    66→  }
    67→  return [];
    68→}
    69→
    70→function saveMessages(messages: ChatMessage[]) {
    71→  try {
    72→    localStorage.setItem(
    73→      STORAGE_KEY,
    74→      JSON.stringify(messages.slice(-MAX_MESSAGES))
    75→    );
    76→  } catch {
    77→    // Ignore storage errors
    78→  }
    79→}
    80→
    81→function clearMessages() {
    82→  try {
    83→    localStorage.removeItem(STORAGE_KEY);
    84→    localStorage.removeItem(SESSION_KEY);
    85→  } catch {
    86→    // Ignore
    87→  }
    88→}
    89→
    90→function getOrCreateSessionId(): string {
    91→  try {
    92→    const existing = localStorage.getItem(SESSION_KEY);
    93→    if (existing) return existing;
    94→    const newId = `session-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    95→    localStorage.setItem(SESSION_KEY, newId);
    96→    return newId;
    97→  } catch {
    98→    return `session-${Date.now()}`;
    99→  }
   100→}
   101→
   102→function loadPosition(): { x: number; y: number } {
   103→  try {
   104→    const stored = localStorage.getItem(POSITION_KEY);
   105→    if (stored) {
   106→      const parsed = JSON.parse(stored);
   107→      if (typeof parsed.x === 'number' && typeof parsed.y === 'number') {
   108→        return parsed;
   109→      }
   110→    }
   111→  } catch {
   112→    // Ignore
   113→  }
   114→  // Default position: bottom-right area
   115→  return { x: 0, y: 0 };
   116→}
   117→
   118→function savePosition(x: number, y: number) {
   119→  try {
   120→    localStorage.setItem(POSITION_KEY, JSON.stringify({ x, y }));
   121→  } catch {
   122→    // Ignore
   123→  }
   124→}

   130→function useDraggable(isEnabled: boolean) {
   131→  const isDragging = useRef(false);
   132→  const dragStart = useRef({ x: 0, y: 0 });
   133→  const containerRef = useRef<HTMLDivElement>(null);
   134→
   135→  // Motion values for smooth drag performance - start at 0, set from effect
   136→  const motionX = useMotionValue(0);
   137→  const motionY = useMotionValue(0);
   138→
   139→  // Store current position for pointer tracking (not accessed during render)
   140→  const currentPosition = useRef({ x: 0, y: 0 });
   141→
   142→  // Load saved position after mount
   143→  useEffect(() => {
   144→    const saved = loadPosition();
   145→    currentPosition.current = saved;
   146→    motionX.set(saved.x);
   147→    motionY.set(saved.y);
   148→  }, [motionX, motionY]);
   149→
   150→  const onPointerDown = useCallback(
   151→    (e: ReactPointerEvent<HTMLDivElement>) => {
   152→      if (!isEnabled) return;
   153→      // Only drag from header, not from buttons
   154→      const target = e.target as HTMLElement;
   155→      if (
   156→        target.closest('button') ||
   157→        target.closest('[data-no-drag]')
   158→      ) {
   159→        return;
   160→      }
   161→
   162→      isDragging.current = true;
   163→      dragStart.current = {
   164→        x: e.clientX - currentPosition.current.x,
   165→        y: e.clientY - currentPosition.current.y,
   166→      };
   167→
   168→      // Capture pointer for reliable tracking
   169→      (e.target as HTMLElement).setPointerCapture(e.pointerId);
   170→    },
   171→    [isEnabled]
   172→  );
   173→
   174→  const onPointerMove = useCallback(
   175→    (e: ReactPointerEvent<HTMLDivElement>) => {
   176→      if (!isDragging.current) return;
   177→
   178→      const newX = e.clientX - dragStart.current.x;
   179→      const newY = e.clientY - dragStart.current.y;
   180→
   181→      // Constrain to viewport bounds
   182→      const container = containerRef.current;
   183→      if (container) {
   184→        const rect = container.getBoundingClientRect();
   185→        const vw = window.innerWidth;
   186→        const vh = window.innerHeight;
   187→
   188→        const constrainedX = Math.max(
   189→          0,
   190→          Math.min(vw - rect.width, newX)
   191→        );
   192→        const constrainedY = Math.max(
   193→          0,
   194→          Math.min(vh - rect.height, newY)
   195→        );
   196→
   197→        currentPosition.current = { x: constrainedX, y: constrainedY };
   198→        motionX.set(constrainedX);
   199→        motionY.set(constrainedY);
   200→      }
   201→    },
   202→    [motionX, motionY]
   203→  );
   204→
   205→  const onPointerUp = useCallback(() => {
   206→    if (isDragging.current) {
   207→      isDragging.current = false;
   208→      savePosition(currentPosition.current.x, currentPosition.current.y);
   209→    }
   210→  }, []);
   211→
   212→  // Reset position when transitioning to fullscreen or when disabled
   213→  useEffect(() => {
   214→    if (!isEnabled) {
   215→      const saved = loadPosition();
   216→      currentPosition.current = saved;
   217→      motionX.set(saved.x);
   218→      motionY.set(saved.y);
   219→    }
   220→  }, [isEnabled, motionX, motionY]);
   221→
   222→  return {
   223→    containerRef,
   224→    motionX,
   225→    motionY,
   226→    onPointerDown,
   227→    onPointerMove,
   228→    onPointerUp,
   229→    isDragging: isDragging,
   230→  };
   231→}

   385→// ──────────────────────────────────────────────
   386→// Enhanced AI Chat Assistant Component
   387→// ──────────────────────────────────────────────
   388→
   389→export function AiChatAssistant() {
   390→  const { t } = useLanguage();
   391→  const [windowState, setWindowState] = useState<WindowState>('closed');
   392→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   393→  const [input, setInput] = useState('');
   394→  const [isLoading, setIsLoading] = useState(false);
   395→  const [sessionId, setSessionId] = useState('');
   396→  const [hasError, setHasError] = useState(false);
   397→  const [hasUnread, setHasUnread] = useState(false);
   398→  const scrollRef = useRef<HTMLDivElement>(null);
   399→  const inputRef = useRef<HTMLInputElement>(null);
   400→  const prevMessageCountRef = useRef(0);
   401→  const prevWindowStateRef = useRef<WindowState>('closed');
   402→
   403→  // ── Dragging (enabled only in 'normal' state on desktop) ──
   404→  const isDesktop = useRef(true);
   405→
   406→  useEffect(() => {
   407→    const checkDesktop = () => {
   408→      isDesktop.current = window.innerWidth >= 1024;
   409→    };
   410→    checkDesktop();
   411→    window.addEventListener('resize', checkDesktop);
   412→    return () => window.removeEventListener('resize', checkDesktop);
   413→  }, []);
   414→
   415→  const draggable = useDraggable(
   416→    windowState === 'normal' && isDesktop.current
   417→  );
   418→
   419→  // ── Quick questions ──
   420→  const quickQuestions = [
   421→    t('chat.quickQuestion1'),
   422→    t('chat.quickQuestion2'),
   423→    t('chat.quickQuestion3'),
   424→    t('chat.quickQuestion4'),
   425→  ];
   426→
   427→  // ── Initialize session and load memory ──
   428→  useEffect(() => {
   429→    setSessionId(getOrCreateSessionId());
   430→    const stored = loadMessages();
   431→    if (stored.length > 0) {
   432→      setMessages(stored);
   433→    }
   434→  }, []);
   435→
   436→  // ── Save messages to localStorage on change ──
   437→  useEffect(() => {
   438→    if (messages.length > 0) {
   439→      saveMessages(messages);
   440→    }
   441→  }, [messages]);
   442→
   443→  // ── Track unread notifications ──
   444→  useEffect(() => {
   445→    const prevState = prevWindowStateRef.current;
   446→    const prevCount = prevMessageCountRef.current;
   447→
   448→    // If new assistant messages arrived while window is closed or minimized
   449→    if (
   450→      (prevState === 'closed' || prevState === 'minimized') &&
   451→      messages.length > prevCount &&
   452→      messages[messages.length - 1]?.role === 'assistant'
   453→    ) {
   454→      setHasUnread(true);
   455→    }
   456→
   457→    prevMessageCountRef.current = messages.length;
   458→  }, [messages]);
   459→
   460→  // ── Clear unread when window is opened normally or fullscreen ──
   461→  useEffect(() => {
   462→    if (windowState === 'normal' || windowState === 'fullscreen') {
   463→      setHasUnread(false);
   464→    }
   465→    prevWindowStateRef.current = windowState;
   466→  }, [windowState]);
   467→
   468→  // ── Auto-scroll to bottom ──
   469→  const scrollToBottom = useCallback(() => {
   470→    if (scrollRef.current) {
   471→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   472→    }
   473→  }, []);
   474→
   475→  useEffect(() => {
   476→    scrollToBottom();
   477→  }, [messages, isLoading, scrollToBottom]);
   478→
   479→  // ── Focus input on open and after response received ──
   480→  const prevLoadingRef = useRef(false);
   481→  const prevWindowStateRef2 = useRef<WindowState>('closed');
   482→  useEffect(() => {
   483→    const isOpen = windowState === 'normal' || windowState === 'fullscreen';
   484→    const justOpened = prevWindowStateRef2.current !== windowState && isOpen;
   485→    const justFinishedLoading = prevLoadingRef.current && !isLoading;
   486→
   487→    // Update refs after computing conditions
   488→    prevWindowStateRef2.current = windowState;
   489→    prevLoadingRef.current = isLoading;
   490→
   491→    // Focus on open OR when response arrives (loading finishes)
   492→    if (isOpen && (justOpened || justFinishedLoading)) {
   493→      // Use double requestAnimationFrame for reliable focus on mobile
   494→      requestAnimationFrame(() => {
   495→        requestAnimationFrame(() => {
   496→          inputRef.current?.focus();
   497→        });
   498→      });
   499→    }
   500→  }, [windowState, isLoading]);
   501→
   502→  // ── Initialize greeting if no messages ──
   503→  useEffect(() => {
   504→    if (
   505→      (windowState === 'normal' || windowState === 'fullscreen') &&
   506→      messages.length === 0
   507→    ) {
   508→      const greeting: ChatMessage = {
   509→        id: 'greeting',
   510→        role: 'assistant',
   511→        content: t('chat.greeting'),
   512→        timestamp: Date.now(),
   513→      };
   514→      setMessages([greeting]);
   515→    }
   516→  }, [windowState, messages.length, t]);
   517→
   518→  // ── ESC key closes the chat ──
   519→  useEffect(() => {
   520→    const handleKeyDown = (e: KeyboardEvent) => {
   521→      if (e.key === 'Escape') {
   522→        if (windowState === 'fullscreen') {
   523→          setWindowState('normal');
   524→        } else if (windowState === 'normal' || windowState === 'minimized') {
   525→          setWindowState('closed');
   526→        }
   527→      }
   528→    };
   529→    window.addEventListener('keydown', handleKeyDown);
   530→    return () => window.removeEventListener('keydown', handleKeyDown);
   531→  }, [windowState]);
   532→
   533→  // ── Send message to API ──
   534→  const sendMessage = async (content: string) => {
   535→    if (!content.trim() || isLoading) return;
   536→
   537→    const userMessage: ChatMessage = {
   538→      id: `user-${Date.now()}`,
   539→      role: 'user',
   540→      content: content.trim(),
   541→      timestamp: Date.now(),
   542→    };
   543→
   544→    setMessages((prev) => [...prev, userMessage]);
   545→    setInput('');
   546→    setIsLoading(true);
   547→    setHasError(false);
   548→
   549→    try {
   550→      // Include up to 10 recent messages for context/memory
   551→      const contextMessages = messages.slice(-10).map((m) => ({
   552→        role: m.role,
   553→        content: m.content,
   554→      }));
   555→
   556→      const response = await fetch(buildApiUrl('/api/chat'), {
   557→        method: 'POST',
   558→        headers: { 'Content-Type': 'application/json' },
   559→        body: JSON.stringify({
   560→          message: content.trim(),
   561→          context: contextMessages,
   562→          sessionId,
   563→        }),
   564→      });
   565→
   566→      const data = await response.json();
   567→
   568→      if (data.success && data.response) {
   569→        const assistantMessage: ChatMessage = {
   570→          id: `assistant-${Date.now()}`,
   571→          role: 'assistant',
   572→          content: data.response,
   573→          timestamp: Date.now(),
   574→        };
   575→        setMessages((prev) => [...prev, assistantMessage]);
   576→
   577→        // Update session ID if server provides one
   578→        if (data.sessionId) {
   579→          setSessionId(data.sessionId);
   580→          try {
   581→            localStorage.setItem(SESSION_KEY, data.sessionId);
   582→          } catch {
   583→            /* ignore */
   584→          }
   585→        }
   586→      } else {
   587→        setHasError(true);
   588→        const errorMessage: ChatMessage = {
   589→          id: `error-${Date.now()}`,
   590→          role: 'assistant',
   591→          content: t('chat.error'),
   592→          timestamp: Date.now(),
   593→        };
   594→        setMessages((prev) => [...prev, errorMessage]);
   595→      }
   596→    } catch {
   597→      setHasError(true);
   598→      const errorMessage: ChatMessage = {
   599→        id: `error-${Date.now()}`,
   600→        role: 'assistant',
   601→        content: t('chat.error'),
   602→        timestamp: Date.now(),
   603→      };
   604→      setMessages((prev) => [...prev, errorMessage]);
   605→    } finally {
   606→      setIsLoading(false);
   607→    }
   608→  };
   609→
   610→  // ── Handle submit ──
   611→  const handleSubmit = (e: React.FormEvent) => {
   612→    e.preventDefault();
   613→    sendMessage(input);
   614→  };
   615→
   616→  // ── Handle quick question click ──
   617→  const handleQuickQuestion = (question: string) => {
   618→    sendMessage(question);
   619→  };
   620→
   621→  // ── Clear chat history ──
   622→  const handleClearHistory = () => {
   623→    setMessages([]);
   624→    clearMessages();
   625→    setSessionId(getOrCreateSessionId());
   626→    // Add fresh greeting
   627→    const greeting: ChatMessage = {
   628→      id: 'greeting-new',
   629→      role: 'assistant',
   630→      content: t('chat.greeting'),
   631→      timestamp: Date.now(),
   632→    };
   633→    setMessages([greeting]);
   634→  };
   635→
   636→  // ── Retry last failed message ──
   637→  const handleRetry = () => {
   638→    const lastUserMsg = [...messages]
   639→      .reverse()
   640→      .find((m) => m.role === 'user');
   641→    if (lastUserMsg) {
   642→      // Remove error messages after the last user message
   643→      setMessages(
   644→        (prev) => prev.filter((m) => m.timestamp <= lastUserMsg.timestamp)
   645→      );
   646→      sendMessage(lastUserMsg.content);
   647→    }
   648→  };
   649→
   650→  // ── Format timestamp ──
   651→  const formatTime = (timestamp: number) => {
   652→    const date = new Date(timestamp);
   653→    return date.toLocaleTimeString([], {
   654→      hour: '2-digit',
   655→      minute: '2-digit',
   656→    });
   657→  };
   658→
   659→  // ── Window state transitions ──
   660→  const openNormal = () => setWindowState('normal');
   661→  const minimize = () => setWindowState('minimized');
