    53→    53→// Auth Store (Firebase Auth — client-side authentication)
    54→    54→//
    55→    55→// Flow:
    56→    56→// 1. Client authenticates with Firebase Auth directly (email/password, Google, phone, anonymous)
    57→    57→// 2. Firebase Auth returns a UserCredential with an ID token
    58→    58→// 3. Client sends ID token to /api/auth/login, /api/auth/social, /api/auth/anonymous, or /api/auth/verify
    59→    59→// 4. Server verifies token, returns Firestore profile (role, isActive, etc.)
    60→    60→// 5. Store saves the combined profile
    61→    61→//
    62→    62→// Phone Auth Flow:
    63→    63→// 1. Client calls loginWithPhone(phoneNumber, recaptchaContainerId)
    64→    64→// 2. Firebase sends SMS OTP to the phone number
    65→    65→// 3. User enters the OTP code
    66→    66→// 4. Client calls verifyPhoneCode(verificationId, otpCode)
    67→    67→// 5. Firebase verifies OTP and returns credential with ID token
    68→    68→// 6. Client sends ID token to /api/auth/social (phone treated as social provider)
    69→    69→// 7. Server verifies token and returns Firestore profile

   125→   125→/**
   126→   126→ * Verify authentication with server API route.
   127→   127→ * Uses apiFetch (instead of raw fetch) to handle:
   128→   128→ *   - Capacitor CORS bypass (external URL + CORS middleware)
   129→   129→ *   - HTML response detection/retry (static export fallback)
   130→   130→ *   - Proper error handling without "Unexpected token '<'" crashes
   131→   131→ *
   132→   132→ * If server is unavailable, falls back to client-side Firestore or
   133→   133→ * native auth result data.
   134→   134→ *
   135→   135→ * @param nativeResult Optional native auth result with user info,
   136→   136→ *   used when the Web SDK's auth.currentUser is not yet synced.
   137→   137→ */
   138→   138→async function verifyWithServer(
   139→   139→  idToken: string,
   140→   140→  endpoint: string,
   141→   141→  nativeResult?: NativeAuthResult
   142→   142→): Promise<{ success: boolean; user?: User; error?: string }> {
   143→   143→  try {
   144→   144→    // Use apiFetch instead of raw fetch — it handles CORS, HTML detection,
   145→   145→    // and automatic retry with external URL for Capacitor apps.
   146→   146→    // Pass the relative path (e.g., '/api/auth/login') — apiFetch handles
   147→   147→    // the full URL resolution internally (don't use buildApiUrl here).
   148→   148→    const res = await apiFetch(endpoint, {
   149→   149→      method: 'POST',
   150→   150→      headers: { 'Content-Type': 'application/json' },
   151→   151→      body: JSON.stringify({ idToken }),
   152→   152→    })
   153→   153→
   154→   154→    // Check if response is JSON before parsing
   155→   155→    const contentType = res.headers.get('content-type') || ''
   156→   156→    if (!contentType.includes('application/json')) {
   157→   157→      console.warn('[Auth] Server returned non-JSON response:', contentType)
   158→   158→      // Not JSON — fall back
   159→   159→      return await fallbackVerify(idToken, nativeResult)
   160→   160→    }
   161→   161→
   162→   162→    if (res.ok) {
   163→   163→      const data = await res.json()
   164→   164→
   165→   165→      if (data.user) {
   166→   166→        const userRole = mapRole(data.user.role)
   167→   167→        const user: User = {
   168→   168→          id: data.user.id,
   169→   169→          name: data.user.name,
   170→   170→          email: data.user.email,
   171→   171→          role: userRole,
   172→   172→          avatar: data.user.avatar || null,
   173→   173→          phone: data.user.phone || null,
   174→   174→          company: data.user.company || null,
   175→   175→          bio: data.user.bio || null,
   176→   176→          address: data.user.address || null,
   177→   177→          authProvider: data.user.authProvider as AuthProvider || 'unknown',
   178→   178→          isAnonymous: data.user.isAnonymous || false,
   179→   179→          emailVerified: data.user.emailVerified || false,
   180→   180→        }
   181→   181→        return { success: true, user }
   182→   182→      } else {
   183→   183→        return { success: false, error: data.error || 'Autenticação falhou.' }
   184→   184→      }
   185→   185→    }
   186→   186→
   187→   187→    // Server returned non-OK status — try to parse error
   188→   188→    let errorMsg = 'Autenticação falhou.'
   189→   189→    try {
   190→   190→      const errData = await res.json()
   191→   191→      errorMsg = errData.error || errorMsg
   192→   192→    } catch {}
   193→   193→
   194→   194→    // If the error is not a permanent auth error, try client-side fallback
   195→   195→    if (res.status !== 401 && res.status !== 403) {
   196→   196→      console.warn('[Auth] Server API error, falling back to client-side:', errorMsg)
   197→   197→      return await fallbackVerify(idToken, nativeResult)
   198→   198→    }
   199→   199→
   200→   200→    return { success: false, error: errorMsg }
   201→   201→  } catch (err) {
   202→   202→    // Network error — server not available (static export / Capacitor / CORS blocked)
   203→   203→    console.warn('[Auth] Server API not reachable, falling back:', err)
   204→   204→    return await fallbackVerify(idToken, nativeResult)
   205→   205→  }
   206→   206→}

   208→   208→/**
   209→   209→ * Smart fallback: tries client-side Firestore first (if auth.currentUser available),
   210→   210→ * then uses native auth result data if provided.
   211→   211→ * This prevents the "Utilizador não autenticado" error on mobile.
   212→   212→ */
   213→   213→async function fallbackVerify(
   214→   214→  idToken: string,
   215→   215→  nativeResult?: NativeAuthResult
   216→   216→): Promise<{ success: boolean; user?: User; error?: string }> {
   217→   217→  // Try client-side Firestore (needs auth.currentUser)
   218→   218→  const clientResult = await verifyWithClientFirestore(idToken)
   219→   219→  if (clientResult.success) return clientResult
   220→   220→
   221→   221→  // If client-side Firestore failed AND we have native auth result,
   222→   222→  // use the native result data directly
   223→   223→  if (nativeResult) {
   224→   224→    return verifyWithNativeResult(nativeResult)
   225→   225→  }
   226→   226→
   227→   227→  return clientResult
   228→   228→}

   230→   230→/**
   231→   231→ * Create user profile from native auth result data.
   232→   232→ * Used when both the server API and client-side Firestore are unavailable,

   234→   234→ *
   235→   235→ * This prevents the "Utilizador não autenticado" error on mobile.

   246→   246→
   247→   247→      // Try to read existing profile

   250→   250→
   251→   251→      if (userSnap.exists()) {
   252→   252→        const profile = userSnap.data()

   276→   276→        address: null,
   277→   277→        role: 'user',
   278→   278→        isActive: true,
   279→   279→        emailVerified: nativeResult.emailVerified,
   280→   280→        authProvider,
   281→   281→        isAnonymous: nativeResult.isAnonymous,
   282→   282→        createdAt: new Date(),
   283→   283→        updatedAt: new Date(),
   284→   284→      })
   285→   285→
   286→   286→      const user: User = {
   287→   287→        id: uid,
   288→   288→        name: nativeResult.displayName || 'Utilizador',
   289→   289→        email: nativeResult.email,
   290→   290→        role: 'user',
   291→   291→        avatar: nativeResult.photoUrl || null,
   292→   292→        phone: nativeResult.phoneNumber || null,
   293→   293→        authProvider,
   294→   294→        isAnonymous: nativeResult.isAnonymous,
   295→   295→        emailVerified: nativeResult.emailVerified,
   296→   296→      }
   297→   297→      return { success: true, user }
   298→   298→    }
   299→   299→
   300→   300→    // Firestore not available — use native result data as last resort
   301→   301→    const user: User = {
   302→   302→      id: nativeResult.uid,
   303→   303→      name: nativeResult.displayName || 'Utilizador',
   304→   304→      email: nativeResult.email,
   305→   305→      role: 'user',
   306→   306→      avatar: nativeResult.photoUrl || null,
   307→   307→      phone: nativeResult.phoneNumber || null,
   308→   308→      authProvider: nativeResult.providerId as AuthProvider || 'unknown',
   309→   309→      isAnonymous: nativeResult.isAnonymous,
   310→   310→      emailVerified: nativeResult.emailVerified,
   311→   311→    }
   312→   312→    return { success: true, user }
   313→   313→  } catch (err) {
   314→   314→    console.error('[Auth] Native result fallback error:', err)
   315→   315→    // Last resort: use native result data directly without Firestore
   316→   316→    const user: User = {
   317→   317→      id: nativeResult.uid,
   318→   318→      name: nativeResult.displayName || 'Utilizador',
   319→   319→      email: nativeResult.email,
   320→   320→      role: 'user',
   321→   321→      avatar: nativeResult.photoUrl || null,

   323→   323→      authProvider: nativeResult.providerId as AuthProvider || 'unknown',
   324→   324→      isAnonymous: nativeResult.isAnonymous,
   325→   325→      emailVerified: nativeResult.emailVerified,
   326→   326→    }
   327→   327→    return { success: true, user }
   328→   328→  }

   331→   331→/**
   332→   332→ * Client-side Firestore fallback for authentication.
   333→   333→ * Used when the server API routes are not available (static export / Capacitor).
   334→   334→ * Reads the user profile from Firestore directly, or creates one if it doesn't exist.
   335→   335→ */
   336→   336→async function verifyWithClientFirestore(idToken: string): Promise<{ success: boolean; user?: User; error?: string }> {
   337→   337→  try {
   338→   338→    const { auth, firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   339→   339→
   340→   340→    if (!isFirebaseConfigured() || !firestoreClient) {
   341→   341→      return { success: false, error: 'Firebase não configurado.' }
   342→   342→    }
   343→   343→
   344→   344→    // Decode the ID token to get the Firebase user info
   345→   345→    // We need to get the current Firebase Auth user
   346→   346→    const currentUser = auth.currentUser
   347→   347→    if (!currentUser) {
   348→   348→      return { success: false, error: 'Utilizador não autenticado.' }
   349→   349→    }
   350→   350→
   351→   351→    const uid = currentUser.uid
   352→   352→    const { doc, getDoc, setDoc } = await import('firebase/firestore')
   353→   353→
   354→   354→    // Try to read existing profile from Firestore
   355→   355→    const userRef = doc(firestoreClient, 'users', uid)
   356→   356→    const userSnap = await getDoc(userRef)
   357→   357→
   358→   358→    if (userSnap.exists()) {
   359→   359→      // Profile exists — return it
   360→   360→      const profile = userSnap.data()
   361→   361→      const userRole = mapRole(profile.role)
   362→   362→      const user: User = {
   363→   363→        id: uid,
   364→   364→        name: profile.name || currentUser.displayName || 'Utilizador',
   365→   365→        email: profile.email || currentUser.email || null,
   366→   366→        role: userRole,
   367→   367→        avatar: profile.avatar || currentUser.photoURL || null,
   368→   368→        phone: profile.phone || currentUser.phoneNumber || null,
   369→   369→        company: profile.company || null,
   370→   370→        bio: profile.bio || null,
   371→   371→        address: profile.address || null,
   372→   372→        authProvider: (currentUser.providerData[0]?.providerId as AuthProvider) || 'unknown',
   373→   373→        isAnonymous: currentUser.isAnonymous,
   374→   374→        emailVerified: currentUser.emailVerified,
   375→   375→      }
   376→   376→      return { success: true, user }
   377→   377→    }
   378→   378→
   379→   379→    // Profile doesn't exist — create it (first-time social/anonymous login)
   380→   380→    const providerId = currentUser.providerData[0]?.providerId || 'unknown'
   381→   381→    let authProvider: AuthProvider = 'unknown'
   382→   382→    if (providerId === 'google.com') authProvider = 'google.com'
   383→   383→    else if (providerId === 'github.com') authProvider = 'github.com'
   384→   384→    else if (providerId === 'phone') authProvider = 'phone'
   385→   385→    else if (providerId === 'anonymous') authProvider = 'anonymous'
   386→   386→    else if (providerId === 'password') authProvider = 'email'
   387→   387→
   388→   388→    const newProfile = {
   389→   389→      name: currentUser.displayName || 'Utilizador',
   390→   390→      email: currentUser.email || null,
   391→   391→      phone: currentUser.phoneNumber || null,
   392→   392→      avatar: currentUser.photoURL || null,
   393→   393→      company: null,
   394→   394→      bio: null,
   395→   395→      address: null,
   396→   396→      role: 'user',
   397→   397→      isActive: true,
   398→   398→      emailVerified: currentUser.emailVerified,
   399→   399→      authProvider,
   400→   400→      isAnonymous: currentUser.isAnonymous,
   401→   401→      createdAt: new Date(),
   402→   402→      updatedAt: new Date(),
   403→   403→    }
   404→   404→
   405→   405→    await setDoc(userRef, newProfile)
   406→   406→
   407→   407→    const user: User = {
   408→   408→      id: uid,
   409→   409→      name: newProfile.name,
   410→   410→      email: newProfile.email,
   411→   411→      role: 'user',
   412→   412→      avatar: newProfile.avatar,
   413→   413→      phone: newProfile.phone,
   414→   414→      company: null,
   415→   415→      bio: null,
   416→   416→      address: null,
   417→   417→      authProvider,
   418→   418→      isAnonymous: currentUser.isAnonymous,
   419→   419→      emailVerified: currentUser.emailVerified,
   420→   420→    }
   421→   421→    return { success: true, user }
   422→   422→  } catch (err) {
   423→   423→    console.error('[Auth] Client-side Firestore fallback error:', err)
   424→   424→    return { success: false, error: 'Erro de autenticação. Tente novamente.' }
   425→   425→  }
   426→   426→}

   965→   965→  unreadCount: number
   966→   966→  addNotification: (type: NotificationType, title: string, message: string) => void
   967→   967→  markAsRead: (id: string) => void

   973→   973→export const useNotificationStore = create<NotificationState>((set, get) => ({
   974→   974→  notifications: [],
   975→   975→  unreadCount: 0,
   976→   976→
   977→   977→  addNotification: (type: NotificationType, title: string, message: string) => {
   978→   978→    const id = `notif-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
   979→   979→    const newNotification: Notification = {
   980→   980→      id,
   981→   981→      type,
   982→   982→      title,
   983→   983→      message,
   984→   984→      read: false,
   985→   985→      createdAt: new Date(),
   986→   986→    }
   987→   987→    const updated = [newNotification, ...get().notifications]
   988→   988→    set({
   989→   989→      notifications: updated,
   990→   990→      unreadCount: updated.filter((n) => !n.read).length,
   991→   991→    })
   992→   992→  },
   993→   993→
   994→   994→  markAsRead: (id: string) => {

   996→   996→      n.id === id ? { ...n, read: true } : n
   997→   997→    )
   998→   998→    set({
   999→   999→      notifications: updated,
  1000→  1000→      unreadCount: updated.filter((n) => !n.read).length,
  1001→  1001→    })
  1002→  1002→  },
  1003→  1003→
  1004→  1004→  markAllAsRead: () => {
  1005→  1005→    const updated = get().notifications.map((n) => ({ ...n, read: true }))
  1006→  1006→    set({
  1007→  1007→      notifications: updated,
  1008→  1008→      unreadCount: 0,
  1009→  1009→    })
  1010→  1010→  },
  1011→  1011→
  1012→  1012→  removeNotification: (id: string) => {
  1013→  1013→    const updated = get().notifications.filter((n) => n.id !== id)
  1014→  1014→    set({
  1015→  1015→      notifications: updated,
  1016→  1016→      unreadCount: updated.filter((n) => !n.read).length,
  1017→  1017→    })
  1018→  1018→  },
  1019→  1019→

  1021→  1021→    set({ notifications: [], unreadCount: 0 })

  1023→  1023→}))
