     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { motion } from 'framer-motion';
     5→import {
     6→  FileText,
     7→  CreditCard,
     8→  HeadphonesIcon,
     9→  ArrowRight,
    10→  User,
    11→  Mail,
    12→  Phone,
    13→  MapPin,
    14→  Building2,
    15→  Clock,
    16→  CheckCircle2,
    17→  XCircle,
    18→  AlertCircle,
    19→  TrendingUp,
    20→  Eye,
    21→  Download,
    22→  MessageSquare,
    23→  Star,
    24→  Bell,
    25→  Inbox,
    26→} from 'lucide-react';
    27→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    28→import { Badge } from '@/components/ui/badge';
    29→import { Button } from '@/components/ui/button';
    30→import { Tabs, TabsList, TabsTrigger, TabsContent } from '@/components/ui/tabs';
    31→import {
    32→  Table,
    33→  TableBody,
    34→  TableCell,
    35→  TableHead,
    36→  TableHeader,
    37→  TableRow,
    38→} from '@/components/ui/table';
    39→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    40→import { Separator } from '@/components/ui/separator';
    41→import { Skeleton } from '@/components/ui/skeleton';
    42→import { useAuthStore } from '@/lib/store';
    43→import { useLanguage } from '@/contexts/language-context';
    44→
    45→// ── Animation variants ──
    46→const containerVariants = {
    47→  hidden: { opacity: 0 },
    48→  visible: {
    49→    opacity: 1,
    50→    transition: { staggerChildren: 0.1, delayChildren: 0.05 },
    51→  },
    52→};
    53→
    54→const itemVariants = {
    55→  hidden: { opacity: 0, y: 20 },
    56→  visible: { opacity: 1, y: 0, transition: { duration: 0.5, ease: 'easeOut' } },
    57→};
    58→
    59→// ── Types for API data ──
    60→interface DashboardQuote {
    61→  id: string;
    62→  title: string;
    63→  description: string | null;
    64→  status: string;
    65→  createdAt: string;
    66→  proposals?: { id: string; title: string; totalAmount: number | null; status: string }[];
    67→}
    68→
    69→interface DashboardPayment {
    70→  id: string;
    71→  amount: number;
    72→  method: string;
    73→  status: string;
    74→  createdAt: string;
    75→  proposal?: { id: string; title: string; totalAmount: number | null; status: string };
    76→}
    77→
    78→interface DashboardTicket {
    79→  id: string;
    80→  subject: string;
    81→  status: string;
    82→  priority: string;
    83→  createdAt: string;
    84→  _count?: { replies: number };
    85→}
    86→
    87→interface DashboardNotification {
    88→  id: string;
    89→  title: string;
    90→  message: string;
    91→  type: string;
    92→  isRead: boolean;
    93→  createdAt: string;
    94→}
    95→
    96→interface DashboardData {
    97→  role: string;
    98→  user: {
    99→    id: string;
   100→    name: string;
   101→    email: string;
   102→    avatar: string | null;
   103→    phone: string | null;
   104→  };
   105→  stats: {
   106→    totalQuotes: number;
   107→    totalPayments: number;
   108→    totalTickets: number;
   109→    unreadNotifications: number;
   110→  };
   111→  recentActivity: {
   112→    quotes: DashboardQuote[];
   113→    payments: DashboardPayment[];
   114→    tickets: DashboardTicket[];
   115→    notifications: DashboardNotification[];
   116→  };
   117→}
   118→
   119→export function UserDashboard() {
   120→  const { t, formatCurrency, formatDate } = useLanguage();
   121→  const user = useAuthStore((s) => s.user);
   122→
   123→  const [activeTab, setActiveTab] = useState('overview');
   124→  const [data, setData] = useState<DashboardData | null>(null);
   125→  const [loading, setLoading] = useState(true);
   126→  const [error, setError] = useState<string | null>(null);
   127→
   128→  useEffect(() => {
   129→    if (!user?.id) return;
   130→
   131→    fetch(`/api/dashboard?role=user&userId=${user.id}`)
   132→      .then((res) => {
   133→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   134→        return res.json();
   135→      })
   136→      .then((json) => {
   137→        if (json.success && json.data) {
   138→          setData(json.data as DashboardData);
   139→        } else {
   140→          setError(json.message || 'Failed to load dashboard data');
   141→        }
   142→      })
   143→      .catch((err) => {
   144→        setError(err.message || 'Network error');
   145→      })
   146→      .finally(() => {
   147→        setLoading(false);
   148→      });
   149→  }, [user?.id]);
   150→
   151→  const quotes = data?.recentActivity.quotes ?? [];
   152→  const payments = data?.recentActivity.payments ?? [];
   153→  const tickets = data?.recentActivity.tickets ?? [];
   154→  const notifications = data?.recentActivity.notifications ?? [];
   155→
   156→  const totalQuotes = data?.stats.totalQuotes ?? 0;
   157→  const totalPaymentsCount = data?.stats.totalPayments ?? 0;
   158→  const totalTicketsCount = data?.stats.totalTickets ?? 0;
   159→  const unreadNotifications = data?.stats.unreadNotifications ?? 0;
   160→
   161→  const approvedQuotes = quotes.filter((q) => q.status === 'accepted' || q.status === 'approved').length;
   162→  const confirmedPayments = payments.filter((p) => p.status === 'confirmed').length;
   163→  const openTickets = tickets.filter((tk) => tk.status === 'open' || tk.status === 'in_progress').length;
   164→  const totalPaymentsAmount = payments.reduce((sum, p) => sum + p.amount, 0);
   165→
   166→  const statusBadge = (status: string) => {
   167→    switch (status) {
   168→      case 'accepted':
   169→      case 'approved':
   170→      case 'confirmed':
   171→      case 'resolved':
   172→      case 'closed':
   173→        return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('common.approved')}</Badge>;
   174→      case 'pending':
   175→      case 'open':
   176→      case 'in_progress':
   177→        return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">{t('common.pending')}</Badge>;
   178→      case 'rejected':
   179→      case 'failed':
   180→        return <Badge className="bg-red-100 text-red-700 border-red-200">{t('common.rejected')}</Badge>;
   181→      default:
   182→        return <Badge variant="outline">{status}</Badge>;
   183→    }
   184→  };
   185→
   186→  const priorityBadge = (priority: string) => {
   187→    switch (priority) {
   188→      case 'high':
   189→      case 'urgent':
   190→        return <Badge className="bg-red-100 text-red-700 border-red-200">{t('common.required')}</Badge>;
   191→      case 'medium':
   192→        return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">{priority}</Badge>;
   193→      case 'low':
   194→        return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{priority}</Badge>;
   195→      default:
   196→        return <Badge variant="outline">{priority}</Badge>;
   197→    }
   198→  };
   199→
   200→  const methodLabel = (method: string) => {
   201→    switch (method) {
   202→      case 'mpesa':
   203→        return t('financial.mpesa');
   204→      case 'transfer':
   205→        return t('financial.transfer');
   206→      case 'deposit':
   207→        return 'Deposit';
   208→      default:
   209→        return method;
   210→    }
   211→  };
   212→
   213→  const statsCards = [
   214→    {
   215→      icon: FileText,
   216→      label: t('dashboard.quotes'),
   217→      value: totalQuotes,
   218→      accent: 'emerald',
   219→      description: `${approvedQuotes} ${t('common.approved')}`,
   220→    },
   221→    {
   222→      icon: CreditCard,
   223→      label: t('dashboard.payments'),
   224→      value: formatCurrency(totalPaymentsAmount),
   225→      accent: 'green',
   226→      description: `${confirmedPayments} ${t('financial.paymentSuccess')}`,
   227→    },
   228→    {
   229→      icon: HeadphonesIcon,
   230→      label: t('dashboard.support'),
   231→      value: openTickets,
   232→      accent: 'teal',
   233→      description: `${totalTicketsCount} ${t('common.total')}`,
   234→    },
   235→  ];
   236→
   237→  const quickActions = [
   238→    { icon: FileText, label: t('dashboard.quoteRequest'), color: 'bg-emerald-600 hover:bg-emerald-700' },
   239→    { icon: CreditCard, label: t('dashboard.paymentHistory'), color: 'bg-green-600 hover:bg-green-700' },
   240→    { icon: HeadphonesIcon, label: t('dashboard.supportCreate'), color: 'bg-teal-600 hover:bg-teal-700' },
   241→  ];
   242→
   243→  // ── Loading skeletons ──
   244→  if (loading) {
   245→    return (
   246→      <motion.div
   247→        variants={containerVariants}
   248→        initial="hidden"
   249→        animate="visible"
   250→        className="space-y-6"
   251→      >
   252→        <Card className="bg-gradient-to-r from-emerald-800 via-emerald-700 to-green-600 border-0 text-white overflow-hidden">
   253→          <CardContent className="p-6">
   254→            <div className="flex items-center gap-4">
   255→              <Skeleton className="h-14 w-14 rounded-full bg-white/20" />
   256→              <div className="flex-1">
   257→                <Skeleton className="h-8 w-48 bg-white/20" />
   258→                <Skeleton className="h-4 w-32 bg-white/20 mt-2" />
   259→              </div>
   260→            </div>
   261→          </CardContent>
   262→        </Card>
   263→
   264→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   265→          {[1, 2, 3].map((i) => (
   266→            <Card key={i} className="border-l-4 border-l-emerald-500">
   267→              <CardContent className="p-4 sm:p-6">
   268→                <Skeleton className="h-4 w-20 mb-2" />
   269→                <Skeleton className="h-8 w-24 mb-1" />
   270→                <Skeleton className="h-3 w-16" />
   271→              </CardContent>
   272→            </Card>
   273→          ))}
   274→        </div>
   275→
   276→        <Card>
   277→          <CardContent className="p-6">
   278→            <Skeleton className="h-6 w-40 mb-4" />
   279→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   280→              {[1, 2, 3].map((i) => (
   281→                <Skeleton key={i} className="h-12 rounded-xl" />
   282→              ))}
   283→            </div>
   284→          </CardContent>
   285→        </Card>
   286→
   287→        <Card>
   288→          <CardContent className="p-6">
   289→            <Skeleton className="h-6 w-32 mb-4" />
   290→            <div className="space-y-3">
   291→              {[1, 2, 3].map((i) => (
   292→                <Skeleton key={i} className="h-16 rounded-lg" />
   293→              ))}
   294→            </div>
   295→          </CardContent>
   296→        </Card>
   297→      </motion.div>
   298→    );
   299→  }
   300→
   301→  // ── Error state ──
   302→  if (error) {
   303→    return (
   304→      <motion.div
   305→        variants={containerVariants}
   306→        initial="hidden"
   307→        animate="visible"
   308→        className="space-y-6"
   309→      >
   310→        <Card className="border-l-4 border-l-red-500">
   311→          <CardContent className="p-6">
   312→            <div className="flex items-center gap-3">
   313→              <AlertCircle className="h-6 w-6 text-red-500" />
   314→              <div>
   315→                <h3 className="font-semibold text-red-700">Failed to load dashboard</h3>
   316→                <p className="text-sm text-muted-foreground">{error}</p>
   317→              </div>
   318→            </div>
   319→            <Button
   320→              className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   321→              onClick={() => {
   322→                setError(null);
   323→                setLoading(true);
   324→                fetch(`/api/dashboard?role=user&userId=${user?.id}`)
   325→                  .then((res) => res.json())
   326→                  .then((json) => {
   327→                    if (json.success && json.data) setData(json.data as DashboardData);
   328→                    else setError(json.message || 'Failed to load');
   329→                  })
   330→                  .catch((err) => setError(err.message))
   331→                  .finally(() => setLoading(false));
   332→              }}
   333→            >
   334→              {t('common.retry') || 'Retry'}
   335→            </Button>
   336→          </CardContent>
   337→        </Card>
   338→      </motion.div>
   339→    );
   340→  }
   341→
   342→  return (
   343→    <motion.div
   344→      variants={containerVariants}
   345→      initial="hidden"
   346→      animate="visible"
   347→      className="space-y-6"
   348→    >
   349→      {/* ── Welcome Section ── */}
   350→      <motion.div variants={itemVariants}>
   351→        <Card className="bg-gradient-to-r from-emerald-800 via-emerald-700 to-green-600 border-0 text-white overflow-hidden">
   352→          <CardContent className="p-6">
   353→            <div className="flex items-center gap-4">
   354→              <Avatar className="h-14 w-14 border-2 border-yellow-400 shadow-lg">
   355→                <AvatarFallback className="bg-emerald-600 text-white text-lg font-bold">
   356→                  {data?.user?.name?.charAt(0) || user?.name?.charAt(0) || 'U'}
   357→                </AvatarFallback>
   358→              </Avatar>
   359→              <div className="flex-1">
   360→                <h1 className="text-2xl sm:text-3xl font-bold">
   361→                  {t('dashboard.welcome', { name: data?.user?.name || user?.name || 'Utilizador' })}
   362→                </h1>
   363→                <p className="text-emerald-200 mt-1">
   364→                  {t('dashboard.lastLogin', { date: formatDate(new Date()) })}
   365→                </p>
   366→              </div>
   367→              <div className="hidden sm:flex items-center gap-2">
   368→                <Star className="h-5 w-5 text-yellow-400" />
   369→                <span className="text-sm text-emerald-200">{t('partner.tierSilver')}</span>
   370→              </div>
   371→            </div>
   372→          </CardContent>
   373→        </Card>
   374→      </motion.div>
   375→
   376→      {/* ── Stats Cards ── */}
   377→      <motion.div variants={itemVariants} className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   378→        {statsCards.map((stat) => (
   379→          <motion.div key={stat.label} variants={itemVariants}>
   380→            <Card className="hover:shadow-md transition-shadow border-l-4 border-l-emerald-500">
   381→              <CardContent className="p-4 sm:p-6">
   382→                <div className="flex items-center justify-between">
   383→                  <div>
   384→                    <p className="text-sm text-muted-foreground">{stat.label}</p>
   385→                    <p className="text-2xl font-bold mt-1">{stat.value}</p>
   386→                    <p className="text-xs text-emerald-600 mt-1">{stat.description}</p>
   387→                  </div>
   388→                  <div className="p-3 rounded-xl bg-emerald-50 text-emerald-600">
   389→                    <stat.icon className="h-6 w-6" />
   390→                  </div>
   391→                </div>
   392→              </CardContent>
   393→            </Card>
   394→          </motion.div>
   395→        ))}
   396→      </motion.div>
   397→
   398→      {/* ── Quick Actions ── */}
   399→      <motion.div variants={itemVariants}>
   400→        <Card>
   401→          <CardHeader>
   402→            <CardTitle className="text-lg">{t('dashboard.quickActions')}</CardTitle>
   403→          </CardHeader>
   404→          <CardContent>
   405→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   406→              {quickActions.map((action) => (
   407→                <Button
   408→                  key={action.label}
   409→                  className={`${action.color} text-white rounded-xl h-12 font-semibold shadow-sm transition-all`}
   410→                >
   411→                  <action.icon className="h-5 w-5 mr-2" />
   412→                  {action.label}
   413→                  <ArrowRight className="h-4 w-4 ml-2" />
   414→                </Button>
   415→              ))}
   416→            </div>
   417→          </CardContent>
   418→        </Card>
   419→      </motion.div>
   420→
   421→      {/* ── Main Content Tabs ── */}
   422→      <motion.div variants={itemVariants}>
   423→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   424→          <TabsList className="w-full sm:w-auto bg-emerald-50">
   425→            <TabsTrigger value="overview" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   426→              {t('dashboard.overview')}
   427→            </TabsTrigger>
   428→            <TabsTrigger value="quotes" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   429→              {t('dashboard.quotes')}
   430→            </TabsTrigger>
   431→            <TabsTrigger value="payments" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   432→              {t('dashboard.payments')}
   433→            </TabsTrigger>
   434→            <TabsTrigger value="support" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   435→              {t('dashboard.support')}
   436→            </TabsTrigger>
   437→          </TabsList>
   438→
   439→          {/* ── Overview Tab ── */}
   440→          <TabsContent value="overview" className="space-y-6 mt-4">
   441→            {/* Recent Quotes Summary */}
   442→            <Card>
   443→              <CardHeader>
   444→                <CardTitle className="text-lg flex items-center gap-2">
   445→                  <FileText className="h-5 w-5 text-emerald-600" />
   446→                  {t('dashboard.quotesHistory')}
   447→                </CardTitle>
   448→              </CardHeader>
   449→              <CardContent>
   450→                {quotes.length === 0 ? (
   451→                  <div className="text-center py-8">
   452→                    <Inbox className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   453→                    <p className="text-sm text-muted-foreground">{t('common.noData') || 'No quotes yet'}</p>
   454→                  </div>
   455→                ) : (
   456→                  <div className="space-y-3 max-h-64 overflow-y-auto">
   457→                    {quotes.slice(0, 3).map((quote) => (
   458→                      <div key={quote.id} className="flex items-center justify-between p-3 rounded-lg bg-muted/50 hover:bg-muted/80 transition-colors">
   459→                        <div className="flex items-center gap-3">
   460→                          <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   461→                            <FileText className="h-4 w-4" />
   462→                          </div>
   463→                          <div>
   464→                            <p className="font-medium text-sm">{quote.title}</p>
   465→                            <p className="text-xs text-muted-foreground">{quote.id.slice(0, 8)} · {formatDate(quote.createdAt)}</p>
   466→                          </div>
   467→                        </div>
   468→                        <div className="text-right">
   469→                          {quote.proposals?.[0]?.totalAmount && (
   470→                            <p className="font-semibold text-sm">{formatCurrency(quote.proposals[0].totalAmount)}</p>
   471→                          )}
   472→                          {statusBadge(quote.status)}
   473→                        </div>
   474→                      </div>
   475→                    ))}
   476→                  </div>
   477→                )}
   478→              </CardContent>
   479→            </Card>
   480→
   481→            {/* Recent Payments Summary */}
   482→            <Card>
   483→              <CardHeader>
   484→                <CardTitle className="text-lg flex items-center gap-2">
   485→                  <CreditCard className="h-5 w-5 text-emerald-600" />
   486→                  {t('dashboard.paymentHistory')}
   487→                </CardTitle>
   488→              </CardHeader>
   489→              <CardContent>
   490→                {payments.length === 0 ? (
   491→                  <div className="text-center py-8">
   492→                    <Inbox className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   493→                    <p className="text-sm text-muted-foreground">{t('common.noData') || 'No payments yet'}</p>
   494→                  </div>
   495→                ) : (
   496→                  <div className="space-y-3 max-h-64 overflow-y-auto">
   497→                    {payments.slice(0, 3).map((payment) => (
   498→                      <div key={payment.id} className="flex items-center justify-between p-3 rounded-lg bg-muted/50 hover:bg-muted/80 transition-colors">
   499→                        <div className="flex items-center gap-3">
   500→                          <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   501→                            <CreditCard className="h-4 w-4" />
   502→                          </div>
   503→                          <div>
   504→                            <p className="font-medium text-sm">
   505→                              {payment.proposal?.title || `Payment ${payment.id.slice(0, 8)}`}
   506→                            </p>
   507→                            <p className="text-xs text-muted-foreground">{methodLabel(payment.method)} · {formatDate(payment.createdAt)}</p>
   508→                          </div>
   509→                        </div>
   510→                        <div className="text-right">
   511→                          <p className="font-semibold text-sm">{formatCurrency(payment.amount)}</p>
   512→                          {statusBadge(payment.status)}
   513→                        </div>
   514→                      </div>
   515→                    ))}
   516→                  </div>
   517→                )}
   518→              </CardContent>
   519→            </Card>
   520→
   521→            {/* Profile Summary */}
   522→            <Card className="border-l-4 border-l-emerald-500">
   523→              <CardHeader>
   524→                <CardTitle className="text-lg flex items-center gap-2">
   525→                  <User className="h-5 w-5 text-emerald-600" />
   526→                  {t('dashboard.profile')}
   527→                </CardTitle>
   528→              </CardHeader>
   529→              <CardContent>
   530→                <div className="flex items-start gap-4">
   531→                  <Avatar className="h-16 w-16 border-2 border-emerald-500">
   532→                    <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xl font-bold">
   533→                      {data?.user?.name?.charAt(0) || user?.name?.charAt(0) || 'U'}
   534→                    </AvatarFallback>
   535→                  </Avatar>
   536→                  <div className="flex-1 grid grid-cols-1 sm:grid-cols-2 gap-3">
   537→                    <div className="flex items-center gap-2 text-sm">
   538→                      <User className="h-4 w-4 text-emerald-500" />
   539→                      <span className="text-muted-foreground">{t('auth.fullName')}:</span>
   540→                      <span className="font-medium">{data?.user?.name || user?.name || '—'}</span>
   541→                    </div>
   542→                    <div className="flex items-center gap-2 text-sm">
   543→                      <Mail className="h-4 w-4 text-emerald-500" />
   544→                      <span className="text-muted-foreground">{t('auth.email')}:</span>
