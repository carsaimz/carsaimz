    56→function loadMessages(): ChatMessage[] {
    57→  try {
    58→    const stored = localStorage.getItem(STORAGE_KEY);
    59→    if (stored) {
    60→      const parsed = JSON.parse(stored);
    61→      return parsed.slice(-MAX_MESSAGES);
    62→    }
    63→  } catch {
    64→    // Ignore parse errors
    65→  }
    66→  return [];
    67→}
    68→
    69→function saveMessages(messages: ChatMessage[]) {
    70→  try {
    71→    localStorage.setItem(
    72→      STORAGE_KEY,
    73→      JSON.stringify(messages.slice(-MAX_MESSAGES))
    74→    );
    75→  } catch {
    76→    // Ignore storage errors
    77→  }
    78→}
    79→
    80→function clearMessages() {
    81→  try {
    82→    localStorage.removeItem(STORAGE_KEY);
    83→    localStorage.removeItem(SESSION_KEY);
    84→  } catch {
    85→    // Ignore
    86→  }
    87→}
    88→
    89→function getOrCreateSessionId(): string {
    90→  try {
    91→    const existing = localStorage.getItem(SESSION_KEY);
    92→    if (existing) return existing;
    93→    const newId = `session-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    94→    localStorage.setItem(SESSION_KEY, newId);
    95→    return newId;
    96→  } catch {
    97→    return `session-${Date.now()}`;
    98→  }
    99→}

   101→function loadPosition(): { x: number; y: number } {
   102→  try {
   103→    const stored = localStorage.getItem(POSITION_KEY);
   104→    if (stored) {
   105→      const parsed = JSON.parse(stored);
   106→      if (typeof parsed.x === 'number' && typeof parsed.y === 'number') {
   107→        return parsed;
   108→      }
   109→    }
   110→  } catch {
   111→    // Ignore
   112→  }
   113→  // Default position: bottom-right area
   114→  return { x: 0, y: 0 };
   115→}
   116→
   117→function savePosition(x: number, y: number) {
   118→  try {
   119→    localStorage.setItem(POSITION_KEY, JSON.stringify({ x, y }));
   120→  } catch {
   121→    // Ignore
   122→  }
   123→}

   129→function useDraggable(isEnabled: boolean) {
   130→  const isDragging = useRef(false);
   131→  const dragStart = useRef({ x: 0, y: 0 });
   132→  const containerRef = useRef<HTMLDivElement>(null);
   133→
   134→  // Motion values for smooth drag performance - start at 0, set from effect
   135→  const motionX = useMotionValue(0);
   136→  const motionY = useMotionValue(0);
   137→
   138→  // Store current position for pointer tracking (not accessed during render)
   139→  const currentPosition = useRef({ x: 0, y: 0 });
   140→
   141→  // Load saved position after mount
   142→  useEffect(() => {
   143→    const saved = loadPosition();
   144→    currentPosition.current = saved;
   145→    motionX.set(saved.x);
   146→    motionY.set(saved.y);
   147→  }, [motionX, motionY]);
   148→
   149→  const onPointerDown = useCallback(
   150→    (e: ReactPointerEvent<HTMLDivElement>) => {
   151→      if (!isEnabled) return;
   152→      // Only drag from header, not from buttons
   153→      const target = e.target as HTMLElement;
   154→      if (
   155→        target.closest('button') ||
   156→        target.closest('[data-no-drag]')
   157→      ) {
   158→        return;
   159→      }
   160→
   161→      isDragging.current = true;
   162→      dragStart.current = {
   163→        x: e.clientX - currentPosition.current.x,
   164→        y: e.clientY - currentPosition.current.y,
   165→      };
   166→
   167→      // Capture pointer for reliable tracking
   168→      (e.target as HTMLElement).setPointerCapture(e.pointerId);
   169→    },
   170→    [isEnabled]
   171→  );
   172→
   173→  const onPointerMove = useCallback(
   174→    (e: ReactPointerEvent<HTMLDivElement>) => {
   175→      if (!isDragging.current) return;
   176→
   177→      const newX = e.clientX - dragStart.current.x;
   178→      const newY = e.clientY - dragStart.current.y;
   179→
   180→      // Constrain to viewport bounds
   181→      const container = containerRef.current;
   182→      if (container) {
   183→        const rect = container.getBoundingClientRect();
   184→        const vw = window.innerWidth;
   185→        const vh = window.innerHeight;
   186→
   187→        const constrainedX = Math.max(
   188→          0,
   189→          Math.min(vw - rect.width, newX)
   190→        );
   191→        const constrainedY = Math.max(
   192→          0,
   193→          Math.min(vh - rect.height, newY)
   194→        );
   195→
   196→        currentPosition.current = { x: constrainedX, y: constrainedY };
   197→        motionX.set(constrainedX);
   198→        motionY.set(constrainedY);
   199→      }
   200→    },
   201→    [motionX, motionY]
   202→  );
   203→
   204→  const onPointerUp = useCallback(() => {
   205→    if (isDragging.current) {
   206→      isDragging.current = false;
   207→      savePosition(currentPosition.current.x, currentPosition.current.y);
   208→    }
   209→  }, []);
   210→
   211→  // Reset position when transitioning to fullscreen or when disabled
   212→  useEffect(() => {
   213→    if (!isEnabled) {
   214→      const saved = loadPosition();
   215→      currentPosition.current = saved;
   216→      motionX.set(saved.x);
   217→      motionY.set(saved.y);
   218→    }
   219→  }, [isEnabled, motionX, motionY]);
   220→
   221→  return {
   222→    containerRef,
   223→    motionX,
   224→    motionY,
   225→    onPointerDown,
   226→    onPointerMove,
   227→    onPointerUp,
   228→    isDragging: isDragging,
   229→  };
   230→}

   384→// ──────────────────────────────────────────────
   385→// Enhanced AI Chat Assistant Component
   386→// ──────────────────────────────────────────────
   387→
   388→export function AiChatAssistant() {
   389→  const { t } = useLanguage();
   390→  const [windowState, setWindowState] = useState<WindowState>('closed');
   391→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   392→  const [input, setInput] = useState('');
   393→  const [isLoading, setIsLoading] = useState(false);
   394→  const [sessionId, setSessionId] = useState('');
   395→  const [hasError, setHasError] = useState(false);
   396→  const [hasUnread, setHasUnread] = useState(false);
   397→  const scrollRef = useRef<HTMLDivElement>(null);
   398→  const inputRef = useRef<HTMLInputElement>(null);
   399→  const prevMessageCountRef = useRef(0);
   400→  const prevWindowStateRef = useRef<WindowState>('closed');
   401→
   402→  // ── Dragging (enabled only in 'normal' state on desktop) ──
   403→  const isDesktop = useRef(true);
   404→
   405→  useEffect(() => {
   406→    const checkDesktop = () => {
   407→      isDesktop.current = window.innerWidth >= 1024;
   408→    };
   409→    checkDesktop();
   410→    window.addEventListener('resize', checkDesktop);
   411→    return () => window.removeEventListener('resize', checkDesktop);
   412→  }, []);
   413→
   414→  const draggable = useDraggable(
   415→    windowState === 'normal' && isDesktop.current
   416→  );
   417→
   418→  // ── Quick questions ──
   419→  const quickQuestions = [
   420→    t('chat.quickQuestion1'),
   421→    t('chat.quickQuestion2'),
   422→    t('chat.quickQuestion3'),
   423→    t('chat.quickQuestion4'),
   424→  ];
   425→
   426→  // ── Initialize session and load memory ──
   427→  useEffect(() => {
   428→    setSessionId(getOrCreateSessionId());
   429→    const stored = loadMessages();
   430→    if (stored.length > 0) {
   431→      setMessages(stored);
   432→    }
   433→  }, []);
   434→
   435→  // ── Save messages to localStorage on change ──
   436→  useEffect(() => {
   437→    if (messages.length > 0) {
   438→      saveMessages(messages);
   439→    }
   440→  }, [messages]);
   441→
   442→  // ── Track unread notifications ──
   443→  useEffect(() => {
   444→    const prevState = prevWindowStateRef.current;
   445→    const prevCount = prevMessageCountRef.current;
   446→
   447→    // If new assistant messages arrived while window is closed or minimized
   448→    if (
   449→      (prevState === 'closed' || prevState === 'minimized') &&
   450→      messages.length > prevCount &&
   451→      messages[messages.length - 1]?.role === 'assistant'
   452→    ) {
   453→      setHasUnread(true);
   454→    }
   455→
   456→    prevMessageCountRef.current = messages.length;
   457→  }, [messages]);
   458→
   459→  // ── Clear unread when window is opened normally or fullscreen ──
   460→  useEffect(() => {
   461→    if (windowState === 'normal' || windowState === 'fullscreen') {
   462→      setHasUnread(false);
   463→    }
   464→    prevWindowStateRef.current = windowState;
   465→  }, [windowState]);
   466→
   467→  // ── Auto-scroll to bottom ──
   468→  const scrollToBottom = useCallback(() => {
   469→    if (scrollRef.current) {
   470→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   471→    }
   472→  }, []);
   473→
   474→  useEffect(() => {
   475→    scrollToBottom();
   476→  }, [messages, isLoading, scrollToBottom]);
   477→
   478→  // ── Focus input on open and after response received ──
   479→  const prevLoadingRef = useRef(false);
   480→  const prevWindowStateRef2 = useRef<WindowState>('closed');
   481→  useEffect(() => {
   482→    const isOpen = windowState === 'normal' || windowState === 'fullscreen';
   483→    const justOpened = prevWindowStateRef2.current !== windowState && isOpen;
   484→    const justFinishedLoading = prevLoadingRef.current && !isLoading;
   485→
   486→    // Update refs after computing conditions
   487→    prevWindowStateRef2.current = windowState;
   488→    prevLoadingRef.current = isLoading;
   489→
   490→    // Focus on open OR when response arrives (loading finishes)
   491→    if (isOpen && (justOpened || justFinishedLoading)) {
   492→      // Use double requestAnimationFrame for reliable focus on mobile
   493→      requestAnimationFrame(() => {
   494→        requestAnimationFrame(() => {
   495→          inputRef.current?.focus();
   496→        });
   497→      });
   498→    }
   499→  }, [windowState, isLoading]);
   500→
   501→  // ── Initialize greeting if no messages ──
   502→  useEffect(() => {
   503→    if (
   504→      (windowState === 'normal' || windowState === 'fullscreen') &&
   505→      messages.length === 0
   506→    ) {
   507→      const greeting: ChatMessage = {
   508→        id: 'greeting',
   509→        role: 'assistant',
   510→        content: t('chat.greeting'),
   511→        timestamp: Date.now(),
   512→      };
   513→      setMessages([greeting]);
   514→    }
   515→  }, [windowState, messages.length, t]);
   516→
   517→  // ── ESC key closes the chat ──
   518→  useEffect(() => {
   519→    const handleKeyDown = (e: KeyboardEvent) => {
   520→      if (e.key === 'Escape') {
   521→        if (windowState === 'fullscreen') {
   522→          setWindowState('normal');
   523→        } else if (windowState === 'normal' || windowState === 'minimized') {
   524→          setWindowState('closed');
   525→        }
   526→      }
   527→    };
   528→    window.addEventListener('keydown', handleKeyDown);
   529→    return () => window.removeEventListener('keydown', handleKeyDown);
   530→  }, [windowState]);
   531→
   532→  // ── Send message to API ──
   533→  const sendMessage = async (content: string) => {
   534→    if (!content.trim() || isLoading) return;
   535→
   536→    const userMessage: ChatMessage = {
   537→      id: `user-${Date.now()}`,
   538→      role: 'user',
   539→      content: content.trim(),
   540→      timestamp: Date.now(),
   541→    };
   542→
   543→    setMessages((prev) => [...prev, userMessage]);
   544→    setInput('');
   545→    setIsLoading(true);
   546→    setHasError(false);
   547→
   548→    try {
   549→      // Include up to 10 recent messages for context/memory
   550→      const contextMessages = messages.slice(-10).map((m) => ({
   551→        role: m.role,
   552→        content: m.content,
   553→      }));
   554→
   555→      const response = await fetch(buildApiUrl('/api/chat'), {
   556→        method: 'POST',
   557→        headers: { 'Content-Type': 'application/json' },
   558→        body: JSON.stringify({
   559→          message: content.trim(),
   560→          context: contextMessages,
   561→          sessionId,
   562→        }),
   563→      });
   564→
   565→      const data = await response.json();
   566→
   567→      if (data.success && data.response) {
   568→        const assistantMessage: ChatMessage = {
   569→          id: `assistant-${Date.now()}`,
   570→          role: 'assistant',
   571→          content: data.response,
   572→          timestamp: Date.now(),
   573→        };
   574→        setMessages((prev) => [...prev, assistantMessage]);
   575→
   576→        // Update session ID if server provides one
   577→        if (data.sessionId) {
   578→          setSessionId(data.sessionId);
   579→          try {
   580→            localStorage.setItem(SESSION_KEY, data.sessionId);
   581→          } catch {
   582→            /* ignore */
   583→          }
   584→        }
   585→      } else {
   586→        setHasError(true);
   587→        const errorMessage: ChatMessage = {
   588→          id: `error-${Date.now()}`,
   589→          role: 'assistant',
   590→          content: t('chat.error'),
   591→          timestamp: Date.now(),
   592→        };
   593→        setMessages((prev) => [...prev, errorMessage]);
   594→      }
   595→    } catch {
   596→      setHasError(true);
   597→      const errorMessage: ChatMessage = {
   598→        id: `error-${Date.now()}`,
   599→        role: 'assistant',
   600→        content: t('chat.error'),
   601→        timestamp: Date.now(),
   602→      };
   603→      setMessages((prev) => [...prev, errorMessage]);
   604→    } finally {
   605→      setIsLoading(false);
   606→    }
   607→  };
   608→
   609→  // ── Handle submit ──
   610→  const handleSubmit = (e: React.FormEvent) => {
   611→    e.preventDefault();
   612→    sendMessage(input);
   613→  };
   614→
   615→  // ── Handle quick question click ──
   616→  const handleQuickQuestion = (question: string) => {
   617→    sendMessage(question);
   618→  };
   619→
   620→  // ── Clear chat history ──
   621→  const handleClearHistory = () => {
   622→    setMessages([]);
   623→    clearMessages();
   624→    setSessionId(getOrCreateSessionId());
   625→    // Add fresh greeting
   626→    const greeting: ChatMessage = {
   627→      id: 'greeting-new',
   628→      role: 'assistant',
   629→      content: t('chat.greeting'),
   630→      timestamp: Date.now(),
   631→    };
   632→    setMessages([greeting]);
   633→  };
   634→
   635→  // ── Retry last failed message ──
   636→  const handleRetry = () => {
   637→    const lastUserMsg = [...messages]
   638→      .reverse()
   639→      .find((m) => m.role === 'user');
   640→    if (lastUserMsg) {
   641→      // Remove error messages after the last user message
   642→      setMessages(
   643→        (prev) => prev.filter((m) => m.timestamp <= lastUserMsg.timestamp)
   644→      );
   645→      sendMessage(lastUserMsg.content);
   646→    }
   647→  };
   648→
   649→  // ── Format timestamp ──
   650→  const formatTime = (timestamp: number) => {
   651→    const date = new Date(timestamp);
   652→    return date.toLocaleTimeString([], {
   653→      hour: '2-digit',
   654→      minute: '2-digit',
   655→    });
   656→  };
   657→
   658→  // ── Window state transitions ──
   659→  const openNormal = () => setWindowState('normal');
   660→  const minimize = () => setWindowState('minimized');
   661→  const maximize = () => setWindowState('fullscreen');
