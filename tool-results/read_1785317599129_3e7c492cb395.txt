     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { motion } from 'framer-motion';
     5→import {
     6→  Users, AlertCircle, RefreshCw, Search, Plus, Pencil, Trash2,
     7→  Loader2, ChevronLeft, ChevronRight, UserPlus,
     8→} from 'lucide-react';
     9→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Button } from '@/components/ui/button';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { Input } from '@/components/ui/input';
    14→import { Label } from '@/components/ui/label';
    15→import { Separator } from '@/components/ui/separator';
    16→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    17→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    18→import {
    19→  Dialog, DialogContent, DialogDescription, DialogFooter,
    20→  DialogHeader, DialogTitle, DialogTrigger,
    21→} from '@/components/ui/dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  AlertDialog, AlertDialogAction, AlertDialogCancel,
    27→  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
    28→  AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger,
    29→} from '@/components/ui/alert-dialog';
    30→import { useLanguage } from '@/contexts/language-context';
    31→import { useAuthStore } from '@/lib/store';
    32→import { apiFetch } from '@/lib/api-fetch';
    33→import { useToast } from '@/hooks/use-toast';
    34→
    35→const containerVariants = { hidden: { opacity: 0 }, visible: { opacity: 1, transition: { staggerChildren: 0.08 } } };
    36→const itemVariants = { hidden: { opacity: 0, y: 20 }, visible: { opacity: 1, y: 0, transition: { duration: 0.5 } } };
    37→
    38→interface UserData {
    39→  id: string;
    40→  name: string;
    41→  email: string;
    42→  phone?: string;
    43→  role: string;
    44→  isActive?: boolean;
    45→  createdAt: string;
    46→}
    47→
    48→interface CreateUserData {
    49→  name: string;
    50→  email: string;
    51→  password: string;
    52→  role: string;
    53→  phone: string;
    54→}
    55→
    56→interface EditUserData {
    57→  name: string;
    58→  email: string;
    59→  phone: string;
    60→  role: string;
    61→  isActive: boolean;
    62→  bio: string;
    63→  company: string;
    64→  address: string;
    65→}
    66→
    67→const PAGE_SIZE = 10;
    68→
    69→export function AdminUsers() {
    70→  const { t, formatDate } = useLanguage();
    71→  const { user: currentUser } = useAuthStore();
    72→  const { toast } = useToast();
    73→
    74→  // ── State ──
    75→  const [users, setUsers] = useState<UserData[]>([]);
    76→  const [loading, setLoading] = useState(true);
    77→  const [error, setError] = useState<string | null>(null);
    78→  const [searchQuery, setSearchQuery] = useState('');
    79→  const [currentPage, setCurrentPage] = useState(1);
    80→  const [totalPages, setTotalPages] = useState(1);
    81→  const [total, setTotal] = useState(0);
    82→
    83→  // ── Create dialog ──
    84→  const [createOpen, setCreateOpen] = useState(false);
    85→  const [createData, setCreateData] = useState<CreateUserData>({
    86→    name: '', email: '', password: '', role: 'user', phone: '',
    87→  });
    88→  const [creating, setCreating] = useState(false);
    89→
    90→  // ── Edit dialog ──
    91→  const [editOpen, setEditOpen] = useState(false);
    92→  const [editUser, setEditUser] = useState<UserData | null>(null);
    93→  const [editData, setEditData] = useState<EditUserData>({
    94→    name: '', email: '', phone: '', role: 'user', isActive: true, bio: '', company: '', address: '',
    95→  });
    96→  const [editing, setEditing] = useState(false);
    97→
    98→  // ── Deactivate ──
    99→  const [deactivatingId, setDeactivatingId] = useState<string | null>(null);
   100→
   101→  // ── Fetch users ──
   102→  const fetchUsers = useCallback(async (page: number, search: string) => {
   103→    setLoading(true);
   104→    setError(null);
   105→    try {
   106→      const params = new URLSearchParams({
   107→        limit: String(PAGE_SIZE),
   108→        page: String(page),
   109→      });
   110→      if (search) params.set('search', search);
   111→
   112→      const res = await apiFetch(`/api/admin/users?${params.toString()}`);
   113→      const data = await res.json();
   114→      if (data.success) {
   115→        setUsers(data.data || []);
   116→        setTotal(data.meta?.total || 0);
   117→        setTotalPages(data.meta?.totalPages || 1);
   118→      } else {
   119→        setError(data.message || 'Failed to load users');
   120→      }
   121→    } catch (err: any) {
   122→      setError(err.message);
   123→    } finally {
   124→      setLoading(false);
   125→    }
   126→  }, []);
   127→
   128→  useEffect(() => {
   129→    fetchUsers(currentPage, searchQuery);
   130→  }, [currentPage, searchQuery, fetchUsers, currentUser?.id]);
   131→
   132→  // ── Search debounce ──
   133→  const [searchInput, setSearchInput] = useState('');
   134→  useEffect(() => {
   135→    const timer = setTimeout(() => {
   136→      setSearchQuery(searchInput);
   137→      setCurrentPage(1);
   138→    }, 300);
   139→    return () => clearTimeout(timer);
   140→  }, [searchInput]);
   141→
   142→  // ── Create user ──
   143→  const handleCreate = async () => {
   144→    setCreating(true);
   145→    try {
   146→      const res = await apiFetch('/api/admin/users', {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify(createData),
   150→      });
   151→      const data = await res.json();
   152→      if (data.success) {
   153→        toast({ title: t('admin.userCreated'), description: t('admin.createUserDesc') });
   154→        setCreateOpen(false);
   155→        setCreateData({ name: '', email: '', password: '', role: 'user', phone: '' });
   156→        fetchUsers(currentPage, searchQuery);
   157→      } else {
   158→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   159→      }
   160→    } catch (err: any) {
   161→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   162→    } finally {
   163→      setCreating(false);
   164→    }
   165→  };
   166→
   167→  // ── Edit user ──
   168→  const handleEdit = async () => {
   169→    if (!editUser) return;
   170→    setEditing(true);
   171→    try {
   172→      const res = await apiFetch(`/api/admin/users?id=${editUser.id}`, {
   173→        method: 'PUT',
   174→        headers: { 'Content-Type': 'application/json' },
   175→        body: JSON.stringify(editData),
   176→      });
   177→      const data = await res.json();
   178→      if (data.success) {
   179→        toast({ title: t('admin.userUpdated'), description: t('admin.editUserDesc') });
   180→        setEditOpen(false);
   181→        setEditUser(null);
   182→        fetchUsers(currentPage, searchQuery);
   183→      } else {
   184→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   185→      }
   186→    } catch (err: any) {
   187→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   188→    } finally {
   189→      setEditing(false);
   190→    }
   191→  };
   192→
   193→  // ── Deactivate user ──
   194→  const handleDeactivate = async (userId: string) => {
   195→    setDeactivatingId(userId);
   196→    try {
   197→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   198→        method: 'DELETE',
   199→      });
   200→      const data = await res.json();
   201→      if (data.success) {
   202→        toast({ title: t('admin.userDeactivated') });
   203→        fetchUsers(currentPage, searchQuery);
   204→      } else {
   205→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   206→      }
   207→    } catch (err: any) {
   208→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   209→    } finally {
   210→      setDeactivatingId(null);
   211→    }
   212→  };
   213→
   214→  // ── Activate user (re-enable via PUT) ──
   215→  const handleActivate = async (userId: string) => {
   216→    setDeactivatingId(userId);
   217→    try {
   218→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   219→        method: 'PUT',
   220→        headers: { 'Content-Type': 'application/json' },
   221→        body: JSON.stringify({ isActive: true }),
   222→      });
   223→      const data = await res.json();
   224→      if (data.success) {
   225→        toast({ title: t('admin.userUpdated') });
   226→        fetchUsers(currentPage, searchQuery);
   227→      } else {
   228→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   229→      }
   230→    } catch (err: any) {
   231→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   232→    } finally {
   233→      setDeactivatingId(null);
   234→    }
   235→  };
   236→
   237→  // ── Role badge ──
   238→  const roleBadge = (role: string) => {
   239→    switch (role) {
   240→      case 'super_admin': return <Badge className="bg-purple-100 text-purple-700 border-purple-200">Super Admin</Badge>;
   241→      case 'admin': return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">Admin</Badge>;
   242→      case 'partner': return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">Partner</Badge>;
   243→      case 'user': return <Badge className="bg-blue-100 text-blue-700 border-blue-200">User</Badge>;
   244→      default: return <Badge variant="outline">{role}</Badge>;
   245→    }
   246→  };
   247→
   248→  // ── Status badge ──
   249→  const statusBadge = (isActive?: boolean) => {
   250→    if (isActive === false) {
   251→      return <Badge className="bg-red-100 text-red-700 border-red-200">{t('admin.userInactive') || 'Inactivo'}</Badge>;
   252→    }
   253→    return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('admin.userActive') || 'Activo'}</Badge>;
   254→  };
   255→
   256→  // ── Open edit dialog ──
   257→  const openEdit = (u: UserData) => {
   258→    setEditUser(u);
   259→    setEditData({
   260→      name: u.name,
   261→      email: u.email,
   262→      phone: u.phone || '',
   263→      role: u.role,
   264→      isActive: u.isActive !== false,
   265→      bio: (u as any).bio || '',
   266→      company: (u as any).company || '',
   267→      address: (u as any).address || '',
   268→    });
   269→    setEditOpen(true);
   270→  };
   271→
   272→  // ── Error state ──
   273→  if (error && !loading) {
   274→    return (
   275→      <motion.div variants={containerVariants} initial="hidden" animate="visible">
   276→        <Card className="border-l-4 border-l-red-500">
   277→          <CardContent className="p-6">
   278→            <div className="flex items-center gap-3">
   279→              <AlertCircle className="h-6 w-6 text-red-500" />
   280→              <p>{error}</p>
   281→            </div>
   282→            <Button className="mt-4" onClick={() => fetchUsers(currentPage, searchQuery)}>
   283→              <RefreshCw className="mr-2 h-4 w-4" />
   284→              {t('common.retry') || 'Retry'}
   285→            </Button>
   286→          </CardContent>
   287→        </Card>
   288→      </motion.div>
   289→    );
   290→  }
   291→
   292→  return (
   293→    <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-6">
   294→      {/* ── Header ── */}
   295→      <motion.div variants={itemVariants}>
   296→        <div className="flex items-center justify-between">
   297→          <div>
   298→            <h2 className="text-2xl font-bold flex items-center gap-2">
   299→              <Users className="h-6 w-6 text-emerald-600" />
   300→              {t('admin.users') || 'Users Management'}
   301→            </h2>
   302→            <p className="text-muted-foreground mt-1">
   303→              {total} {t('admin.usersTotal') || 'total users'}
   304→            </p>
   305→          </div>
   306→
   307→          {/* ── Create user button ── */}
   308→          <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   309→            <DialogTrigger asChild>
   310→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">
   311→                <UserPlus className="h-4 w-4" />
   312→                {t('admin.createUser') || 'Create User'}
   313→              </Button>
   314→            </DialogTrigger>
   315→            <DialogContent className="sm:max-w-[425px]">
   316→              <DialogHeader>
   317→                <DialogTitle>{t('admin.createUser') || 'Create User'}</DialogTitle>
   318→                <DialogDescription>{t('admin.createUserDesc') || 'Create a new platform user'}</DialogDescription>
   319→              </DialogHeader>
   320→              <div className="grid gap-4 py-4">
   321→                <div className="grid gap-2">
   322→                  <Label htmlFor="create-name">{t('admin.userName') || 'Name'}</Label>
   323→                  <Input
   324→                    id="create-name"
   325→                    value={createData.name}
   326→                    onChange={(e) => setCreateData(prev => ({ ...prev, name: e.target.value }))}
   327→                    className="focus-visible:ring-emerald-500"
   328→                  />
   329→                </div>
   330→                <div className="grid gap-2">
   331→                  <Label htmlFor="create-email">{t('admin.userEmail') || 'Email'}</Label>
   332→                  <Input
   333→                    id="create-email"
   334→                    type="email"
   335→                    value={createData.email}
   336→                    onChange={(e) => setCreateData(prev => ({ ...prev, email: e.target.value }))}
   337→                    className="focus-visible:ring-emerald-500"
   338→                  />
   339→                </div>
   340→                <div className="grid gap-2">
   341→                  <Label htmlFor="create-password">{t('admin.userPassword') || 'Password'}</Label>
   342→                  <Input
   343→                    id="create-password"
   344→                    type="password"
   345→                    value={createData.password}
   346→                    onChange={(e) => setCreateData(prev => ({ ...prev, password: e.target.value }))}
   347→                    className="focus-visible:ring-emerald-500"
   348→                  />
   349→                </div>
   350→                <div className="grid gap-2">
   351→                  <Label htmlFor="create-phone">{t('admin.userPhone') || 'Phone'}</Label>
   352→                  <Input
   353→                    id="create-phone"
   354→                    value={createData.phone}
   355→                    onChange={(e) => setCreateData(prev => ({ ...prev, phone: e.target.value }))}
   356→                    className="focus-visible:ring-emerald-500"
   357→                  />
   358→                </div>
   359→                <div className="grid gap-2">
   360→                  <Label>{t('admin.userRole') || 'Role'}</Label>
   361→                  <Select
   362→                    value={createData.role}
   363→                    onValueChange={(val) => setCreateData(prev => ({ ...prev, role: val }))}
   364→                  >
   365→                    <SelectTrigger className="focus:ring-emerald-500">
   366→                      <SelectValue />
   367→                    </SelectTrigger>
   368→                    <SelectContent>
   369→                      <SelectItem value="user">User</SelectItem>
   370→                      <SelectItem value="partner">Partner</SelectItem>
   371→                      <SelectItem value="admin">Admin</SelectItem>
   372→                      <SelectItem value="super_admin">Super Admin</SelectItem>
   373→                    </SelectContent>
   374→                  </Select>
   375→                </div>
   376→              </div>
   377→              <DialogFooter>
   378→                <Button variant="outline" onClick={() => setCreateOpen(false)}>
   379→                  {t('admin.cancel') || 'Cancel'}
   380→                </Button>
   381→                <Button
   382→                  className="bg-emerald-600 hover:bg-emerald-700 text-white"
   383→                  onClick={handleCreate}
   384→                  disabled={creating || !createData.name || !createData.email || !createData.password}
   385→                >
   386→                  {creating ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Plus className="h-4 w-4 mr-2" />}
   387→                  {t('admin.createUser') || 'Create'}
   388→                </Button>
   389→              </DialogFooter>
   390→            </DialogContent>
   391→          </Dialog>
   392→        </div>
   393→      </motion.div>
   394→
   395→      {/* ── Search ── */}
   396→      <motion.div variants={itemVariants}>
   397→        <div className="relative">
   398→          <Search className="absolute left-3 top-1/2 h-4 w-4 text-muted-foreground" />
   399→          <Input
   400→            placeholder={t('admin.searchUsers') || 'Search users...'}
   401→            value={searchInput}
   402→            onChange={(e) => setSearchInput(e.target.value)}
   403→            className="pl-10 focus-visible:ring-emerald-500"
   404→          />
   405→        </div>
   406→      </motion.div>
   407→
   408→      {/* ── Users table ── */}
   409→      <motion.div variants={itemVariants}>
   410→        <Card>
   411→          <CardContent className="p-0">
   412→            {loading ? (
   413→              <div className="p-6 space-y-4">
   414→                {[1, 2, 3].map(i => (
   415→                  <Skeleton key={i} className="h-12 w-full rounded" />
   416→                ))}
   417→              </div>
   418→            ) : users.length === 0 ? (
   419→              <div className="text-center py-12">
   420→                <Users className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   421→                <p className="text-muted-foreground">{t('common.noData') || 'No users found'}</p>
   422→              </div>
   423→            ) : (
   424→              <Table>
   425→                <TableHeader>
   426→                  <TableRow className="bg-emerald-50/50">
   427→                    <TableHead>{t('admin.userName') || 'Name'}</TableHead>
   428→                    <TableHead>{t('admin.userEmail') || 'Email'}</TableHead>
   429→                    <TableHead>{t('admin.userPhone') || 'Phone'}</TableHead>
   430→                    <TableHead>{t('admin.userRole') || 'Role'}</TableHead>
   431→                    <TableHead>{t('admin.userStatus') || 'Status'}</TableHead>
   432→                    <TableHead>{t('admin.date') || 'Date'}</TableHead>
   433→                    <TableHead className="text-right">{t('admin.actions') || 'Actions'}</TableHead>
   434→                  </TableRow>
   435→                </TableHeader>
   436→                <TableBody>
   437→                  {users.map((u) => (
   438→                    <TableRow key={u.id}>
   439→                      <TableCell>
   440→                        <div className="flex items-center gap-2">
   441→                          <Avatar className="size-6">
   442→                            <AvatarFallback className="text-xs">
   443→                              {u.name.split(' ').map(n => n[0]).join('').slice(0, 2)}
   444→                            </AvatarFallback>
   445→                          </Avatar>
   446→                          <span className="font-medium">{u.name}</span>
   447→                        </div>
   448→                      </TableCell>
   449→                      <TableCell className="text-muted-foreground">{u.email}</TableCell>
   450→                      <TableCell className="text-muted-foreground">{u.phone || '—'}</TableCell>
   451→                      <TableCell>{roleBadge(u.role)}</TableCell>
   452→                      <TableCell>{statusBadge(u.isActive)}</TableCell>
   453→                      <TableCell className="text-muted-foreground">{formatDate(u.createdAt)}</TableCell>
   454→                      <TableCell className="text-right">
   455→                        <div className="flex items-center justify-end gap-1">
   456→                          {/* Edit button */}
   457→                          <Button
   458→                            variant="ghost"
   459→                            size="icon"
   460→                            className="h-8 w-8"
   461→                            onClick={() => openEdit(u)}
   462→                          >
   463→                            <Pencil className="h-4 w-4 text-emerald-600" />
   464→                          </Button>
   465→
   466→                          {/* Deactivate / Activate button */}
   467→                          {u.isActive === false ? (
   468→                            <Button
   469→                              variant="ghost"
   470→                              size="icon"
   471→                              className="h-8 w-8"
   472→                              onClick={() => handleActivate(u.id)}
   473→                              disabled={deactivatingId === u.id}
   474→                            >
   475→                              {deactivatingId === u.id ? (
   476→                                <Loader2 className="h-4 w-4 animate-spin" />
   477→                              ) : (
   478→                                <RefreshCw className="h-4 w-4 text-emerald-600" />
   479→                              )}
   480→                            </Button>
   481→                          ) : (
   482→                            <AlertDialog>
   483→                              <AlertDialogTrigger asChild>
   484→                                <Button variant="ghost" size="icon" className="h-8 w-8">
   485→                                  <Trash2 className="h-4 w-4 text-red-600" />
   486→                                </Button>
   487→                              </AlertDialogTrigger>
   488→                              <AlertDialogContent>
   489→                                <AlertDialogHeader>
   490→                                  <AlertDialogTitle>{t('admin.deactivateUser') || 'Deactivate User'}</AlertDialogTitle>
   491→                                  <AlertDialogDescription>
   492→                                    {t('admin.confirmDeactivate') || 'Are you sure you want to deactivate this user?'}
   493→                                  </AlertDialogDescription>
   494→                                </AlertDialogHeader>
   495→                                <AlertDialogFooter>
   496→                                  <AlertDialogCancel>{t('admin.cancel') || 'Cancel'}</AlertDialogCancel>
   497→                                  <AlertDialogAction
   498→                                    className="bg-red-600 hover:bg-red-700 text-white"
   499→                                    onClick={() => handleDeactivate(u.id)}
   500→                                  >
   501→                                    {t('admin.deactivateUser') || 'Deactivate'}
   502→                                  </AlertDialogAction>
   503→                                </AlertDialogFooter>
   504→                              </AlertDialogContent>
   505→                            </AlertDialog>
   506→                          )}
   507→                        </div>
   508→                      </TableCell>
   509→                    </TableRow>
   510→                  ))}
   511→                </TableBody>
   512→              </Table>
   513→            )}
   514→          </CardContent>
   515→        </Card>
   516→      </motion.div>
   517→
   518→      {/* ── Pagination ── */}
   519→      {totalPages > 1 && (
   520→        <motion.div variants={itemVariants}>
   521→          <div className="flex items-center justify-center gap-2">
   522→            <Button
   523→              variant="outline"
   524→              size="icon"
   525→              disabled={currentPage <= 1 || loading}
   526→              onClick={() => setCurrentPage(prev => prev - 1)}
   527→            >
   528→              <ChevronLeft className="h-4 w-4" />
   529→            </Button>
   530→            <span className="text-sm text-muted-foreground">
   531→              {currentPage} / {totalPages}
   532→            </span>
   533→            <Button
   534→              variant="outline"
   535→              size="icon"
   536→              disabled={currentPage >= totalPages || loading}
   537→              onClick={() => setCurrentPage(prev => prev + 1)}
   538→            >
