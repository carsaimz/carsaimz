    52→    52→// Auth Store (Firebase Auth — client-side authentication)
    53→    53→//
    54→    54→// Flow:
    55→    55→// 1. Client authenticates with Firebase Auth directly (email/password, Google, phone, anonymous)
    56→    56→// 2. Firebase Auth returns a UserCredential with an ID token
    57→    57→// 3. Client sends ID token to /api/auth/login, /api/auth/social, /api/auth/anonymous, or /api/auth/verify
    58→    58→// 4. Server verifies token, returns Firestore profile (role, isActive, etc.)
    59→    59→// 5. Store saves the combined profile
    60→    60→//
    61→    61→// Phone Auth Flow:
    62→    62→// 1. Client calls loginWithPhone(phoneNumber, recaptchaContainerId)
    63→    63→// 2. Firebase sends SMS OTP to the phone number
    64→    64→// 3. User enters the OTP code
    65→    65→// 4. Client calls verifyPhoneCode(verificationId, otpCode)
    66→    66→// 5. Firebase verifies OTP and returns credential with ID token
    67→    67→// 6. Client sends ID token to /api/auth/social (phone treated as social provider)
    68→    68→// 7. Server verifies token and returns Firestore profile

   124→   124→/**
   125→   125→ * Verify authentication with server API route.
   126→   126→ * If server is unavailable (static export / Capacitor), falls back to
   127→   127→ * client-side Firestore profile lookup/creation.
   128→   128→ */
   129→   129→async function verifyWithServer(idToken: string, endpoint: string): Promise<{ success: boolean; user?: User; error?: string }> {
   130→   130→  try {
   131→   131→    const res = await fetch(buildApiUrl(endpoint), {
   132→   132→      method: 'POST',
   133→   133→      headers: { 'Content-Type': 'application/json' },
   134→   134→      body: JSON.stringify({ idToken }),
   135→   135→    })
   136→   136→
   137→   137→    // If the server returned a valid response (not 404), use it
   138→   138→    if (res.ok) {
   139→   139→      const data = await res.json()
   140→   140→
   141→   141→      if (data.user) {
   142→   142→        const userRole = mapRole(data.user.role)
   143→   143→        const user: User = {
   144→   144→          id: data.user.id,
   145→   145→          name: data.user.name,
   146→   146→          email: data.user.email,
   147→   147→          role: userRole,
   148→   148→          avatar: data.user.avatar || null,
   149→   149→          phone: data.user.phone || null,
   150→   150→          company: data.user.company || null,
   151→   151→          bio: data.user.bio || null,
   152→   152→          address: data.user.address || null,
   153→   153→          authProvider: data.user.authProvider as AuthProvider || 'unknown',
   154→   154→          isAnonymous: data.user.isAnonymous || false,
   155→   155→          emailVerified: data.user.emailVerified || false,
   156→   156→        }
   157→   157→        return { success: true, user }
   158→   158→      } else {
   159→   159→        return { success: false, error: data.error || 'Autenticação falhou.' }
   160→   160→      }
   161→   161→    }
   162→   162→
   163→   163→    // Server returned 404 or other error — fall back to client-side Firestore
   164→   164→    console.warn('[Auth] Server API unavailable, falling back to client-side Firestore')
   165→   165→    return await verifyWithClientFirestore(idToken)
   166→   166→  } catch (err) {
   167→   167→    // Network error — server not available (static export / Capacitor)
   168→   168→    console.warn('[Auth] Server API not reachable, falling back to client-side Firestore:', err)
   169→   169→    return await verifyWithClientFirestore(idToken)
   170→   170→  }
   171→   171→}

   173→   173→/**
   174→   174→ * Client-side Firestore fallback for authentication.
   175→   175→ * Used when the server API routes are not available (static export / Capacitor).
   176→   176→ * Reads the user profile from Firestore directly, or creates one if it doesn't exist.
   177→   177→ */
   178→   178→async function verifyWithClientFirestore(idToken: string): Promise<{ success: boolean; user?: User; error?: string }> {
   179→   179→  try {
   180→   180→    const { auth, firestoreClient, isFirebaseConfigured } = await import('@/lib/firebase-client')
   181→   181→
   182→   182→    if (!isFirebaseConfigured() || !firestoreClient) {
   183→   183→      return { success: false, error: 'Firebase não configurado.' }
   184→   184→    }
   185→   185→
   186→   186→    // Decode the ID token to get the Firebase user info
   187→   187→    // We need to get the current Firebase Auth user
   188→   188→    const currentUser = auth.currentUser
   189→   189→    if (!currentUser) {
   190→   190→      return { success: false, error: 'Utilizador não autenticado.' }
   191→   191→    }
   192→   192→
   193→   193→    const uid = currentUser.uid
   194→   194→    const { doc, getDoc, setDoc } = await import('firebase/firestore')
   195→   195→
   196→   196→    // Try to read existing profile from Firestore
   197→   197→    const userRef = doc(firestoreClient, 'users', uid)
   198→   198→    const userSnap = await getDoc(userRef)
   199→   199→
   200→   200→    if (userSnap.exists()) {
   201→   201→      // Profile exists — return it
   202→   202→      const profile = userSnap.data()
   203→   203→      const userRole = mapRole(profile.role)
   204→   204→      const user: User = {
   205→   205→        id: uid,
   206→   206→        name: profile.name || currentUser.displayName || 'Utilizador',
   207→   207→        email: profile.email || currentUser.email || null,
   208→   208→        role: userRole,
   209→   209→        avatar: profile.avatar || currentUser.photoURL || null,
   210→   210→        phone: profile.phone || currentUser.phoneNumber || null,
   211→   211→        company: profile.company || null,
   212→   212→        bio: profile.bio || null,
   213→   213→        address: profile.address || null,
   214→   214→        authProvider: (currentUser.providerData[0]?.providerId as AuthProvider) || 'unknown',
   215→   215→        isAnonymous: currentUser.isAnonymous,
   216→   216→        emailVerified: currentUser.emailVerified,
   217→   217→      }
   218→   218→      return { success: true, user }
   219→   219→    }
   220→   220→
   221→   221→    // Profile doesn't exist — create it (first-time social/anonymous login)
   222→   222→    const providerId = currentUser.providerData[0]?.providerId || 'unknown'
   223→   223→    let authProvider: AuthProvider = 'unknown'
   224→   224→    if (providerId === 'google.com') authProvider = 'google.com'
   225→   225→    else if (providerId === 'phone') authProvider = 'phone'
   226→   226→    else if (providerId === 'anonymous') authProvider = 'anonymous'
   227→   227→    else if (providerId === 'password') authProvider = 'email'
   228→   228→
   229→   229→    const newProfile = {
   230→   230→      name: currentUser.displayName || 'Utilizador',
   231→   231→      email: currentUser.email || null,
   232→   232→      phone: currentUser.phoneNumber || null,
   233→   233→      avatar: currentUser.photoURL || null,
   234→   234→      company: null,
   235→   235→      bio: null,
   236→   236→      address: null,
   237→   237→      role: 'user',
   238→   238→      isActive: true,
   239→   239→      emailVerified: currentUser.emailVerified,
   240→   240→      authProvider,
   241→   241→      isAnonymous: currentUser.isAnonymous,
   242→   242→      createdAt: new Date(),
   243→   243→      updatedAt: new Date(),
   244→   244→    }
   245→   245→
   246→   246→    await setDoc(userRef, newProfile)
   247→   247→
   248→   248→    const user: User = {
   249→   249→      id: uid,
   250→   250→      name: newProfile.name,
   251→   251→      email: newProfile.email,
   252→   252→      role: 'user',
   253→   253→      avatar: newProfile.avatar,
   254→   254→      phone: newProfile.phone,
   255→   255→      company: null,
   256→   256→      bio: null,
   257→   257→      address: null,
   258→   258→      authProvider,
   259→   259→      isAnonymous: currentUser.isAnonymous,
   260→   260→      emailVerified: currentUser.emailVerified,
   261→   261→    }
   262→   262→    return { success: true, user }
   263→   263→  } catch (err) {
   264→   264→    console.error('[Auth] Client-side Firestore fallback error:', err)
   265→   265→    return { success: false, error: 'Erro de autenticação. Tente novamente.' }
   266→   266→  }
   267→   267→}

   334→   334→            set({ idToken })
   335→   335→            return { success: true, user: result.user }
   336→   336→          } else {
   337→   337→            set({ isLoading: false, lastLoginError: result.error })
   338→   338→            return { success: false, error: result.error }
   339→   339→          }
   340→   340→        } catch (err: any) {

   342→   342→          let errorMsg = 'Credenciais inválidas.'
   343→   343→

   352→   352→          } else if (err.message?.includes('not configured')) {
   353→   353→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   354→   354→          }

   458→   458→          } else if (err.code === 'auth/account-exists-with-different-credential') {
   459→   459→            errorMsg = 'Conta já existe com outro método de login. Use o mesmo método que registou.'
   460→   460→          }
   461→   461→
   462→   462→          set({ isLoading: false, lastLoginError: errorMsg })
   463→   463→          return { success: false, error: errorMsg }
   464→   464→        }
   465→   465→      },
   466→   466→
   467→   467→      // ── Phone Authentication (Step 1: Send SMS OTP) ──
   468→   468→      loginWithPhone: async (phoneNumber: string, recaptchaContainerId: string): Promise<{ verificationId: string } | null> => {
   469→   469→        set({ isLoading: true, lastLoginError: null })
   470→   470→
   471→   471→        try {
   472→   472→          const fb = await getFirebaseAuth()
   473→   473→          const { auth, PhoneAuthProvider, createRecaptchaVerifier } = fb
   474→   474→
   475→   475→          // Create RecaptchaVerifier for the given container
   476→   476→          const recaptchaVerifier = createRecaptchaVerifier(recaptchaContainerId, 'invisible')
   477→   477→
   478→   478→          // Send SMS verification code
   479→   479→          const phoneProvider = new PhoneAuthProvider(auth)
   480→   480→          const verificationId = await phoneProvider.verifyPhoneNumber(
   481→   481→            phoneNumber,
   482→   482→            recaptchaVerifier
   483→   483→          )
   484→   484→
   485→   485→          set({ phoneVerificationId: verificationId, isLoading: false })
   486→   486→          return { verificationId }
   487→   487→        } catch (err: any) {
   488→   488→          console.error('Phone auth error:', err)
   489→   489→          let errorMsg = 'Falha no login por telefone.'
   490→   490→
   491→   491→          if (err.code === 'auth/invalid-phone-number') {
   492→   492→            errorMsg = 'Número de telefone inválido. Use formato +258XXXXXXXXX.'
   493→   493→          } else if (err.code === 'auth/quota-exceeded') {
   494→   494→            errorMsg = 'Limite de SMS excedido. Tente novamente mais tarde.'
   495→   495→          } else if (err.code === 'auth/too-many-requests') {
   496→   496→            errorMsg = 'Muitas tentativas. Tente novamente mais tarde.'
   497→   497→          }
   498→   498→
   499→   499→          set({ isLoading: false, lastLoginError: errorMsg })
   500→   500→          return null
   501→   501→        }
   502→   502→      },
   503→   503→
   504→   504→      // ── Phone Authentication (Step 2: Verify OTP code) ──
   505→   505→      verifyPhoneCode: async (verificationId: string, verificationCode: string): Promise<AuthResult> => {
   506→   506→        set({ isLoading: true, lastLoginError: null })
   507→   507→
   508→   508→        try {
   509→   509→          const fb = await getFirebaseAuth()
   510→   510→          const { auth, PhoneAuthProvider, signInWithCredential } = fb
   511→   511→
   512→   512→          // Create credential from verification ID + OTP code
   513→   513→          const credential = PhoneAuthProvider.credential(verificationId, verificationCode)
   514→   514→          const userCredential = await signInWithCredential(auth, credential)
   515→   515→          const idToken = await userCredential.user.getIdToken()
   516→   516→
   517→   517→          // Verify with server to get Firestore profile
   518→   518→          const result = await verifyWithServer(idToken, '/api/auth/social')
   519→   519→
   520→   520→          if (result.success && result.user) {
   521→   521→            setUserInStore(set, result.user)
   522→   522→            set({ idToken, phoneVerificationId: null })
   523→   523→            return { success: true, user: result.user }
   524→   524→          } else {
   525→   525→            set({ isLoading: false, lastLoginError: result.error, phoneVerificationId: null })
   526→   526→            return { success: false, error: result.error }
   527→   527→          }
   528→   528→        } catch (err: any) {
   529→   529→          console.error('Phone verify error:', err)
   530→   530→          let errorMsg = 'Código de verificação inválido.'
   531→   531→
   532→   532→          if (err.code === 'auth/invalid-verification-code') {
   533→   533→            errorMsg = 'Código SMS incorreto. Verifique e tente novamente.'
   534→   534→          } else if (err.code === 'auth/invalid-verification-id') {
   535→   535→            errorMsg = 'Sessão de verificação expirada. Tente novamente.'
   536→   536→          } else if (err.code === 'auth/code-expired') {
   537→   537→            errorMsg = 'Código expirado. Solicite um novo código SMS.'
   538→   538→          }
   539→   539→
   540→   540→          set({ isLoading: false, lastLoginError: errorMsg, phoneVerificationId: null })
   541→   541→          return { success: false, error: errorMsg }
   542→   542→        }
   543→   543→      },
   544→   544→
   545→   545→      // ── Anonymous Sign-In ──
   546→   546→      loginAnonymously: async (): Promise<AuthResult> => {
   547→   547→        set({ isLoading: true, lastLoginError: null })
   548→   548→
   549→   549→        try {
   550→   550→          let idToken: string
   551→   551→
   552→   552→          if (shouldUseNativeAuth()) {
   553→   553→            // ── Native path: @capacitor-firebase/authentication ──
   554→   554→            const nativeResult = await nativeSignInAnonymously()
   555→   555→            idToken = nativeResult.idToken
   556→   556→          } else {
   557→   557→            // ── Web path: Firebase Web SDK ──
   558→   558→            const { signInAnonymously, auth } = await getFirebaseAuth()
   559→   559→            const credential = await signInAnonymously(auth)
   560→   560→            idToken = await credential.user.getIdToken()
   561→   561→          }
   562→   562→
   563→   563→          const result = await verifyWithServer(idToken, '/api/auth/anonymous')
   564→   564→
   565→   565→          if (result.success && result.user) {
   566→   566→            setUserInStore(set, result.user)
   567→   567→            set({ idToken })
   568→   568→            return { success: true, user: result.user }
   569→   569→          } else {
   570→   570→            set({ isLoading: false, lastLoginError: result.error })
   571→   571→            return { success: false, error: result.error }
   572→   572→          }
   573→   573→        } catch (err: any) {
   574→   574→          console.error('Anonymous login error:', err)
   575→   575→          set({ isLoading: false, lastLoginError: 'Falha no login como visitante.' })
   576→   576→          return { success: false, error: 'Falha no login como visitante.' }
   577→   577→        }
   578→   578→      },
   579→   579→
   580→   580→      // ── Email/password registration ──
   581→   581→      register: async (name: string, email: string, password: string, phone?: string): Promise<AuthResult> => {
   582→   582→        set({ isLoading: true, lastRegisterError: null })
   583→   583→
   584→   584→        try {
   585→   585→          const { createUserWithEmailAndPassword, auth, updateProfile } = await getFirebaseAuth()
   586→   586→          const credential = await createUserWithEmailAndPassword(auth, email, password)
   587→   587→
   588→   588→          // Update display name in Firebase Auth
   589→   589→          await updateProfile(credential.user, { displayName: name })
   590→   590→
   591→   591→          const idToken = await credential.user.getIdToken()
   592→   592→
   593→   593→          // Send to server to create Firestore profile
   594→   594→          const res = await fetch(buildApiUrl('/api/auth/register'), {
   595→   595→            method: 'POST',
   596→   596→            headers: { 'Content-Type': 'application/json' },
   597→   597→            body: JSON.stringify({ name, email, phone }),
   598→   598→          })
   599→   599→
   600→   600→          const data = await res.json()
   601→   601→
   602→   602→          if (data.user) {
   603→   603→            const userRole = mapRole(data.user.role)
   604→   604→            const user: User = {
   605→   605→              id: data.user.id || credential.user.uid,
   606→   606→              name: data.user.name || name,
   607→   607→              email: data.user.email || email,
   608→   608→              role: userRole,
   609→   609→              avatar: null,
   610→   610→              phone: data.user.phone || phone || null,
   611→   611→              authProvider: 'email',
   612→   612→              isAnonymous: false,
   613→   613→              emailVerified: false,
   614→   614→            }
   615→   615→            setUserInStore(set, user)
   616→   616→            set({ idToken, lastRegisterError: null })
   617→   617→            return { success: true, user }
   618→   618→          } else {
   619→   619→            // Firebase Auth user was created but Firestore profile failed
   620→   620→            // Still set the user with basic info
   621→   621→            const user: User = {
   622→   622→              id: credential.user.uid,
   623→   623→              name: name,
   624→   624→              email: email,
   625→   625→              role: 'user',
   626→   626→              avatar: null,
   627→   627→              authProvider: 'email',
   628→   628→              isAnonymous: false,
   629→   629→              emailVerified: false,
   630→   630→            }
   631→   631→            setUserInStore(set, user)
   632→   632→            set({ idToken })
   633→   633→            return { success: true, user }
   634→   634→          }
   635→   635→        } catch (err: any) {
   636→   636→          console.error('Register error:', err)
   637→   637→          let errorMsg = 'Falha ao criar conta.'
   638→   638→
   639→   639→          if (err.code === 'auth/email-already-in-use') {
   640→   640→            errorMsg = 'Este email já está registado.'
   641→   641→          } else if (err.code === 'auth/weak-password') {
   642→   642→            errorMsg = 'Senha demasiado fraca. Use pelo menos 6 caracteres.'
   643→   643→          } else if (err.code === 'auth/invalid-email') {
   644→   644→            errorMsg = 'Email inválido.'
   645→   645→          } else if (err.message?.includes('not configured')) {
   646→   646→            errorMsg = 'Firebase não configurado. Contacte o administrador.'
   647→   647→          }
   648→   648→
   649→   649→          set({ isLoading: false, lastRegisterError: errorMsg })
   650→   650→          return { success: false, error: errorMsg }
   651→   651→        }
   652→   652→      },
   653→   653→
   654→   654→      logout: async () => {
   655→   655→        try {
   656→   656→          if (shouldUseNativeAuth()) {
   657→   657→            await nativeSignOut()
   658→   658→          } else {
   659→   659→            const { signOut, auth } = await getFirebaseAuth()
   660→   660→            await signOut(auth)
   661→   661→          }
   662→   662→        } catch (err) {
   663→   663→          console.warn('Firebase signOut failed (non-critical):', err)
   664→   664→        }
   665→   665→
   666→   666→        set({
   667→   667→          user: null,
   668→   668→          isAuthenticated: false,
   669→   669→          isAdmin: false,
   670→   670→          isPartner: false,
   671→   671→          isUser: false,
   672→   672→          isSuperAdmin: false,
   673→   673→          idToken: null,
   674→   674→          phoneVerificationId: null,
   675→   675→          lastLoginError: null,
   676→   676→          lastRegisterError: null,
   677→   677→        })
   678→   678→      },
   679→   679→
   680→   680→      setUser: (user: User) => {
   681→   681→        setUserInStore(set, user)
   682→   682→      },
   683→   683→
   684→   684→      updateAvatar: (avatar: string) => {
   685→   685→        const currentUser = get().user
   686→   686→        if (currentUser) {
   687→   687→          set({ user: { ...currentUser, avatar } })
   688→   688→        }
   689→   689→      },
   690→   690→
   691→   691→      setIdToken: (token: string) => {
   692→   692→        set({ idToken: token })
   693→   693→      },
   694→   694→    }),
   695→   695→    {
   696→   696→      name: 'carsai-auth',
   697→   697→      partialize: (state) => ({

   719→   719→interface NotificationState {
   720→   720→  notifications: Notification[]
   721→   721→  unreadCount: number
   722→   722→  addNotification: (type: NotificationType, title: string, message: string) => void
   723→   723→  markAsRead: (id: string) => void
   724→   724→  markAllAsRead: () => void
   725→   725→  removeNotification: (id: string) => void
   726→   726→  clearAll: () => void
   727→   727→}
   728→   728→
   729→   729→export const useNotificationStore = create<NotificationState>((set, get) => ({
   730→   730→  notifications: [],
   731→   731→  unreadCount: 0,
   732→   732→
   733→   733→  addNotification: (type: NotificationType, title: string, message: string) => {
   734→   734→    const id = `notif-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`
   735→   735→    const newNotification: Notification = {
   736→   736→      id,
   737→   737→      type,
   738→   738→      title,
   739→   739→      message,
   740→   740→      read: false,
   741→   741→      createdAt: new Date(),
   742→   742→    }
   743→   743→    const updated = [newNotification, ...get().notifications]
   744→   744→    set({
   745→   745→      notifications: updated,
   746→   746→      unreadCount: updated.filter((n) => !n.read).length,
   747→   747→    })
   748→   748→  },
   749→   749→
   750→   750→  markAsRead: (id: string) => {
   751→   751→    const updated = get().notifications.map((n) =>
   752→   752→      n.id === id ? { ...n, read: true } : n
   753→   753→    )
   754→   754→    set({
   755→   755→      notifications: updated,
   756→   756→      unreadCount: updated.filter((n) => !n.read).length,
   757→   757→    })
   758→   758→  },
   759→   759→
   760→   760→  markAllAsRead: () => {
   761→   761→    const updated = get().notifications.map((n) => ({ ...n, read: true }))
   762→   762→    set({
   763→   763→      notifications: updated,
   764→   764→      unreadCount: 0,
   765→   765→    })
   766→   766→  },
   767→   767→
   768→   768→  removeNotification: (id: string) => {
   769→   769→    const updated = get().notifications.filter((n) => n.id !== id)
   770→   770→    set({
   771→   771→      notifications: updated,
   772→   772→      unreadCount: updated.filter((n) => !n.read).length,
   773→   773→    })
   774→   774→  },
   775→   775→
   776→   776→  clearAll: () => {
   777→   777→    set({ notifications: [], unreadCount: 0 })
   778→   778→  },
   779→   779→}))
