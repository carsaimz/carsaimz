    46→const STORAGE_KEY = 'carsai_chat_messages';

    55→function loadMessages(): ChatMessage[] {
    56→  try {
    57→    const stored = localStorage.getItem(STORAGE_KEY);
    58→    if (stored) {
    59→      const parsed = JSON.parse(stored);
    60→      return parsed.slice(-MAX_MESSAGES);
    61→    }
    62→  } catch {
    63→    // Ignore parse errors
    64→  }
    65→  return [];
    66→}
    67→
    68→function saveMessages(messages: ChatMessage[]) {
    69→  try {
    70→    localStorage.setItem(
    71→      STORAGE_KEY,
    72→      JSON.stringify(messages.slice(-MAX_MESSAGES))
    73→    );
    74→  } catch {
    75→    // Ignore storage errors
    76→  }
    77→}
    78→
    79→function clearMessages() {
    80→  try {
    81→    localStorage.removeItem(STORAGE_KEY);
    82→    localStorage.removeItem(SESSION_KEY);
    83→  } catch {
    84→    // Ignore
    85→  }
    86→}
    87→
    88→function getOrCreateSessionId(): string {
    89→  try {
    90→    const existing = localStorage.getItem(SESSION_KEY);
    91→    if (existing) return existing;
    92→    const newId = `session-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    93→    localStorage.setItem(SESSION_KEY, newId);
    94→    return newId;
    95→  } catch {
    96→    return `session-${Date.now()}`;
    97→  }
    98→}
    99→
   100→function loadPosition(): { x: number; y: number } {
   101→  try {
   102→    const stored = localStorage.getItem(POSITION_KEY);
   103→    if (stored) {
   104→      const parsed = JSON.parse(stored);
   105→      if (typeof parsed.x === 'number' && typeof parsed.y === 'number') {
   106→        return parsed;
   107→      }
   108→    }
   109→  } catch {
   110→    // Ignore
   111→  }
   112→  // Default position: bottom-right area
   113→  return { x: 0, y: 0 };
   114→}
   115→
   116→function savePosition(x: number, y: number) {
   117→  try {
   118→    localStorage.setItem(POSITION_KEY, JSON.stringify({ x, y }));
   119→  } catch {
   120→    // Ignore
   121→  }
   122→}

   124→// ──────────────────────────────────────────────
   125→// Hook: useDraggable
   126→// ──────────────────────────────────────────────
   127→
   128→function useDraggable(isEnabled: boolean) {
   129→  const isDragging = useRef(false);
   130→  const dragStart = useRef({ x: 0, y: 0 });
   131→  const containerRef = useRef<HTMLDivElement>(null);
   132→
   133→  // Motion values for smooth drag performance - start at 0, set from effect
   134→  const motionX = useMotionValue(0);
   135→  const motionY = useMotionValue(0);
   136→
   137→  // Store current position for pointer tracking (not accessed during render)
   138→  const currentPosition = useRef({ x: 0, y: 0 });
   139→
   140→  // Load saved position after mount
   141→  useEffect(() => {
   142→    const saved = loadPosition();
   143→    currentPosition.current = saved;
   144→    motionX.set(saved.x);
   145→    motionY.set(saved.y);
   146→  }, [motionX, motionY]);
   147→
   148→  const onPointerDown = useCallback(
   149→    (e: ReactPointerEvent<HTMLDivElement>) => {
   150→      if (!isEnabled) return;
   151→      // Only drag from header, not from buttons
   152→      const target = e.target as HTMLElement;
   153→      if (
   154→        target.closest('button') ||
   155→        target.closest('[data-no-drag]')
   156→      ) {
   157→        return;
   158→      }
   159→
   160→      isDragging.current = true;
   161→      dragStart.current = {
   162→        x: e.clientX - currentPosition.current.x,
   163→        y: e.clientY - currentPosition.current.y,
   164→      };
   165→
   166→      // Capture pointer for reliable tracking
   167→      (e.target as HTMLElement).setPointerCapture(e.pointerId);
   168→    },
   169→    [isEnabled]
   170→  );
   171→
   172→  const onPointerMove = useCallback(
   173→    (e: ReactPointerEvent<HTMLDivElement>) => {
   174→      if (!isDragging.current) return;
   175→
   176→      const newX = e.clientX - dragStart.current.x;
   177→      const newY = e.clientY - dragStart.current.y;
   178→
   179→      // Constrain to viewport bounds
   180→      const container = containerRef.current;
   181→      if (container) {
   182→        const rect = container.getBoundingClientRect();
   183→        const vw = window.innerWidth;
   184→        const vh = window.innerHeight;
   185→
   186→        const constrainedX = Math.max(
   187→          0,
   188→          Math.min(vw - rect.width, newX)
   189→        );
   190→        const constrainedY = Math.max(
   191→          0,
   192→          Math.min(vh - rect.height, newY)
   193→        );
   194→
   195→        currentPosition.current = { x: constrainedX, y: constrainedY };
   196→        motionX.set(constrainedX);
   197→        motionY.set(constrainedY);
   198→      }
   199→    },
   200→    [motionX, motionY]
   201→  );
   202→
   203→  const onPointerUp = useCallback(() => {
   204→    if (isDragging.current) {
   205→      isDragging.current = false;
   206→      savePosition(currentPosition.current.x, currentPosition.current.y);
   207→    }
   208→  }, []);
   209→
   210→  // Reset position when transitioning to fullscreen or when disabled
   211→  useEffect(() => {
   212→    if (!isEnabled) {
   213→      const saved = loadPosition();
   214→      currentPosition.current = saved;
   215→      motionX.set(saved.x);
   216→      motionY.set(saved.y);
   217→    }
   218→  }, [isEnabled, motionX, motionY]);
   219→
   220→  return {
   221→    containerRef,
   222→    motionX,
   223→    motionY,
   224→    onPointerDown,
   225→    onPointerMove,
   226→    onPointerUp,
   227→    isDragging: isDragging,
   228→  };
   229→}

   331→// ──────────────────────────────────────────────
   332→// Enhanced AI Chat Assistant Component
   333→// ──────────────────────────────────────────────
   334→
   335→export function AiChatAssistant() {
   336→  const { t } = useLanguage();
   337→  const [windowState, setWindowState] = useState<WindowState>('closed');
   338→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   339→  const [input, setInput] = useState('');
   340→  const [isLoading, setIsLoading] = useState(false);
   341→  const [sessionId, setSessionId] = useState('');
   342→  const [hasError, setHasError] = useState(false);
   343→  const [hasUnread, setHasUnread] = useState(false);
   344→  const scrollRef = useRef<HTMLDivElement>(null);
   345→  const inputRef = useRef<HTMLInputElement>(null);
   346→  const prevMessageCountRef = useRef(0);
   347→  const prevWindowStateRef = useRef<WindowState>('closed');
   348→
   349→  // ── Dragging (enabled only in 'normal' state on desktop) ──
   350→  const isDesktop = useRef(true);
   351→
   352→  useEffect(() => {
   353→    const checkDesktop = () => {
   354→      isDesktop.current = window.innerWidth >= 1024;
   355→    };
   356→    checkDesktop();
   357→    window.addEventListener('resize', checkDesktop);
   358→    return () => window.removeEventListener('resize', checkDesktop);
   359→  }, []);
   360→
   361→  const draggable = useDraggable(
   362→    windowState === 'normal' && isDesktop.current
   363→  );
   364→
   365→  // ── Quick questions ──
   366→  const quickQuestions = [
   367→    t('chat.quickQuestion1'),
   368→    t('chat.quickQuestion2'),
   369→    t('chat.quickQuestion3'),
   370→    t('chat.quickQuestion4'),
   371→  ];
   372→
   373→  // ── Initialize session and load memory ──
   374→  useEffect(() => {
   375→    setSessionId(getOrCreateSessionId());
   376→    const stored = loadMessages();
   377→    if (stored.length > 0) {
   378→      setMessages(stored);
   379→    }
   380→  }, []);
   381→
   382→  // ── Save messages to localStorage on change ──
   383→  useEffect(() => {
   384→    if (messages.length > 0) {
   385→      saveMessages(messages);
   386→    }
   387→  }, [messages]);
   388→
   389→  // ── Track unread notifications ──
   390→  useEffect(() => {
   391→    const prevState = prevWindowStateRef.current;
   392→    const prevCount = prevMessageCountRef.current;
   393→
   394→    // If new assistant messages arrived while window is closed or minimized
   395→    if (
   396→      (prevState === 'closed' || prevState === 'minimized') &&
   397→      messages.length > prevCount &&
   398→      messages[messages.length - 1]?.role === 'assistant'
   399→    ) {
   400→      setHasUnread(true);
   401→    }
   402→
   403→    prevMessageCountRef.current = messages.length;
   404→  }, [messages]);
   405→
   406→  // ── Clear unread when window is opened normally or fullscreen ──
   407→  useEffect(() => {
   408→    if (windowState === 'normal' || windowState === 'fullscreen') {
   409→      setHasUnread(false);
   410→    }
   411→    prevWindowStateRef.current = windowState;
   412→  }, [windowState]);
   413→
   414→  // ── Auto-scroll to bottom ──
   415→  const scrollToBottom = useCallback(() => {
   416→    if (scrollRef.current) {
   417→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   418→    }
   419→  }, []);
   420→
   421→  useEffect(() => {
   422→    scrollToBottom();
   423→  }, [messages, isLoading, scrollToBottom]);
   424→
   425→  // ── Focus input on open ──
   426→  useEffect(() => {
   427→    if (
   428→      (windowState === 'normal' || windowState === 'fullscreen') &&
   429→      inputRef.current
   430→    ) {
   431→      setTimeout(() => inputRef.current?.focus(), 150);
   432→    }
   433→  }, [windowState]);
   434→
   435→  // ── Initialize greeting if no messages ──
   436→  useEffect(() => {
   437→    if (
   438→      (windowState === 'normal' || windowState === 'fullscreen') &&
   439→      messages.length === 0
   440→    ) {
   441→      const greeting: ChatMessage = {
   442→        id: 'greeting',
   443→        role: 'assistant',
   444→        content: t('chat.greeting'),
   445→        timestamp: Date.now(),
   446→      };
   447→      setMessages([greeting]);
   448→    }
   449→  }, [windowState, messages.length, t]);
   450→
   451→  // ── ESC key closes the chat ──
   452→  useEffect(() => {
   453→    const handleKeyDown = (e: KeyboardEvent) => {
   454→      if (e.key === 'Escape') {
   455→        if (windowState === 'fullscreen') {
   456→          setWindowState('normal');
   457→        } else if (windowState === 'normal' || windowState === 'minimized') {
   458→          setWindowState('closed');
   459→        }
   460→      }
   461→    };
   462→    window.addEventListener('keydown', handleKeyDown);
   463→    return () => window.removeEventListener('keydown', handleKeyDown);
   464→  }, [windowState]);
   465→
   466→  // ── Send message to API ──
   467→  const sendMessage = async (content: string) => {
   468→    if (!content.trim() || isLoading) return;
   469→
   470→    const userMessage: ChatMessage = {
   471→      id: `user-${Date.now()}`,
   472→      role: 'user',
   473→      content: content.trim(),
   474→      timestamp: Date.now(),
   475→    };
   476→
   477→    setMessages((prev) => [...prev, userMessage]);
   478→    setInput('');
   479→    setIsLoading(true);
   480→    setHasError(false);
   481→
   482→    try {
   483→      // Include up to 10 recent messages for context/memory
   484→      const contextMessages = messages.slice(-10).map((m) => ({
   485→        role: m.role,
   486→        content: m.content,
   487→      }));
   488→
   489→      const response = await fetch('/api/chat', {
   490→        method: 'POST',
   491→        headers: { 'Content-Type': 'application/json' },
   492→        body: JSON.stringify({
   493→          message: content.trim(),
   494→          context: contextMessages,
   495→          sessionId,
   496→        }),
   497→      });
   498→
   499→      const data = await response.json();
   500→
   501→      if (data.success && data.response) {
   502→        const assistantMessage: ChatMessage = {
   503→          id: `assistant-${Date.now()}`,
   504→          role: 'assistant',
   505→          content: data.response,
   506→          timestamp: Date.now(),
   507→        };
   508→        setMessages((prev) => [...prev, assistantMessage]);
   509→
   510→        // Update session ID if server provides one
   511→        if (data.sessionId) {
   512→          setSessionId(data.sessionId);
   513→          try {
   514→            localStorage.setItem(SESSION_KEY, data.sessionId);
   515→          } catch {
   516→            /* ignore */
   517→          }
   518→        }
   519→      } else {
   520→        setHasError(true);
   521→        const errorMessage: ChatMessage = {
   522→          id: `error-${Date.now()}`,
   523→          role: 'assistant',
   524→          content: t('chat.error'),
   525→          timestamp: Date.now(),
   526→        };
   527→        setMessages((prev) => [...prev, errorMessage]);
   528→      }
   529→    } catch {
   530→      setHasError(true);
   531→      const errorMessage: ChatMessage = {
   532→        id: `error-${Date.now()}`,
   533→        role: 'assistant',
   534→        content: t('chat.error'),
   535→        timestamp: Date.now(),
   536→      };
   537→      setMessages((prev) => [...prev, errorMessage]);
   538→    } finally {
   539→      setIsLoading(false);
   540→    }
   541→  };
   542→
   543→  // ── Handle submit ──
   544→  const handleSubmit = (e: React.FormEvent) => {
   545→    e.preventDefault();
   546→    sendMessage(input);
   547→  };
   548→
   549→  // ── Handle quick question click ──
   550→  const handleQuickQuestion = (question: string) => {
   551→    sendMessage(question);
   552→  };
   553→
   554→  // ── Clear chat history ──
   555→  const handleClearHistory = () => {
   556→    setMessages([]);
   557→    clearMessages();
   558→    setSessionId(getOrCreateSessionId());
   559→    // Add fresh greeting
   560→    const greeting: ChatMessage = {
   561→      id: 'greeting-new',
   562→      role: 'assistant',
   563→      content: t('chat.greeting'),
   564→      timestamp: Date.now(),
   565→    };
   566→    setMessages([greeting]);
   567→  };
   568→
   569→  // ── Retry last failed message ──
   570→  const handleRetry = () => {
   571→    const lastUserMsg = [...messages]
   572→      .reverse()
   573→      .find((m) => m.role === 'user');
   574→    if (lastUserMsg) {
   575→      // Remove error messages after the last user message
   576→      setMessages(
   577→        (prev) => prev.filter((m) => m.timestamp <= lastUserMsg.timestamp)
   578→      );
   579→      sendMessage(lastUserMsg.content);
   580→    }
   581→  };
   582→
   583→  // ── Format timestamp ──
   584→  const formatTime = (timestamp: number) => {
   585→    const date = new Date(timestamp);
   586→    return date.toLocaleTimeString([], {
   587→      hour: '2-digit',
   588→      minute: '2-digit',
   589→    });
   590→  };
   591→
   592→  // ── Window state transitions ──
   593→  const openNormal = () => setWindowState('normal');
   594→  const minimize = () => setWindowState('minimized');
   595→  const maximize = () => setWindowState('fullscreen');
   596→  const restoreFromFullscreen = () => setWindowState('normal');
   597→  const restoreFromMinimized = () => setWindowState('normal');
   598→  const closeChat = () => setWindowState('closed');
   599→
   600→  // ── Determine sizing classes based on window state ──
   601→  const getSizeClasses = () => {
   602→    if (windowState === 'fullscreen') {
   603→      return 'w-[calc(100vw-16px)] h-[calc(100vh-16px)] sm:w-[calc(100vw-24px)] sm:h-[calc(100vh-24px)]';
   604→    }
   605→    if (windowState === 'minimized') {
   606→      return 'w-[calc(100vw-32px)] sm:w-[320px] h-auto';
   607→    }
   608→    // Normal state - responsive sizing
   609→    return 'w-[calc(100vw-32px)] sm:w-[60vw] md:w-[480px] lg:w-[400px] h-[70vh] lg:h-[600px]';
   610→  };
   611→
   612→  const getPositionClasses = () => {
   613→    if (windowState === 'fullscreen') {
   614→      return 'top-2 left-2 sm:top-3 sm:left-3';
   615→    }
   616→    if (windowState === 'minimized') {
   617→      return 'bottom-4 left-1/2 -translate-x-1/2 sm:left-auto sm:translate-x-0 sm:right-6';
   618→    }
   619→    // Normal: positioned by drag on desktop, fixed bottom-right on mobile
   620→    return '';
   621→  };
   622→
   623→  // ── Render Header ──
   624→  const renderHeader = (compact: boolean = false) => (
   625→    <CardHeader
   626→      className={`pb-2 bg-gradient-to-r from-emerald-600 via-emerald-700 to-green-600 text-white rounded-t-2xl border-b border-emerald-500/30 select-none ${
   627→        compact ? 'py-2 px-3' : ''
   628→      }`}
   629→    >
   630→      <div className="flex items-center justify-between">
   631→        <div className="flex items-center gap-2.5 min-w-0">
   632→          <div
   633→            className={`shrink-0 rounded-full bg-white/20 flex items-center justify-center ${
   634→              compact ? 'size-6' : 'size-8'
   635→            }`}
   636→          >
   637→            <Bot className={`${compact ? 'size-3' : 'size-4'} text-white`} />
   638→          </div>
   639→          <div className="min-w-0">
   640→            <CardTitle
   641→              className={`font-semibold truncate ${
   642→                compact ? 'text-xs' : 'text-sm'
   643→              }`}
   644→            >
   645→              {t('chat.title')}
   646→            </CardTitle>
   647→            {!compact && (
   648→              <p className="text-xs text-emerald-200/80 truncate">
   649→                {t('chat.subtitle')}
   650→              </p>
   651→            )}
   652→          </div>
