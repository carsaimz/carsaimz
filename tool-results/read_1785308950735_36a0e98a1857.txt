    54→// ── Animation variants ──
    55→const containerVariants = {
    56→  hidden: { opacity: 0 },
    57→  visible: {
    58→    opacity: 1,
    59→    transition: { staggerChildren: 0.08, delayChildren: 0.05 },

    61→};
    62→
    63→const itemVariants = {
    64→  hidden: { opacity: 0, y: 20 },
    65→  visible: { opacity: 1, y: 0, transition: { duration: 0.5, ease: 'easeOut' } },
    66→};
    67→
    68→// ── Chart Colors (Emerald theme) ──
    69→const CHART_COLORS = ['#059669', '#10b981', '#34d399', '#6ee7b7', '#a7f3d0', '#d1fae5'];
    70→
    71→// ── Types ──

    73→  overview: {
    74→    totalUsers: number;
    75→    totalPosts: number;
    76→    totalProjects: number;
    77→    totalServices: number;
    78→    totalForumTopics: number;
    79→    totalRevenue: number;
    80→    confirmedRevenue: number;
    81→  };
    82→  users: {
    83→    total: number;
    84→    active: number;
    85→    admins: number;
    86→    partners: number;
    87→    regular: number;
    88→  };
    89→  payments: {
    90→    total: number;
    91→    totalRevenue: number;
    92→    confirmedRevenue: number;
    93→    mpesa: number;
    94→    transfer: number;
    95→    deposit: number;
    96→  };
    97→}
    98→
    99→interface HistoryData {
   100→  revenue: { month: string; revenue: number; target: number }[];
   101→  usersGrowth: { month: string; newUsers: number; totalUsers: number }[];
   102→  postsByCategory: { name: string; value: number }[];
   103→}
   104→

   106→  id: string;
   107→  name: string;
   108→  email: string;
   109→  createdAt: string;
   110→  isActive: boolean;

   112→
   113→interface RecentQuote {
   114→  id: string;
   115→  title: string;
   116→  status: string;
   117→  createdAt: string;
   118→  user: { id: string; name: string; email: string };

   120→
   121→interface RecentPayment {
   122→  id: string;
   123→  amount: number;
   124→  method: string;
   125→  status: string;
   126→  createdAt: string;
   127→  user: { id: string; name: string; email: string };
   128→  proposal: { id: string; title: string } | null;
   129→}
   130→
   131→interface RecentTicket {
   132→  id: string;
   133→  subject: string;
   134→  status: string;
   135→  createdAt: string;
   136→  user: { id: string; name: string; email: string };
   137→}
   138→
   139→interface RecentPost {
   140→  id: string;
   141→  title: string;
   142→  createdAt: string;

   144→  _count: { comments: number };
   145→}
   146→
   147→interface DashboardData {
   148→  role: string;
   149→  stats: {
   150→    totalUsers: number;
   151→    totalPosts: number;
   152→    totalQuotes: number;
   153→    totalPayments: number;
   154→    totalTickets: number;
   155→    totalForumTopics: number;
   156→    totalRevenue: number;
   157→  };
   158→  breakdowns: {
   159→    quotes: { status: string; count: number }[];
   160→    payments: { status: string; count: number; total: number }[];
   161→    tickets: { status: string; count: number }[];
   162→  };
   163→  recentActivity: {
   164→    users: RecentUser[];
   165→    quotes: RecentQuote[];
   166→    payments: RecentPayment[];
   167→    tickets: RecentTicket[];
   168→    posts: RecentPost[];
   169→  };
   170→}
   171→
   172→export function AdminDashboard() {
   173→  const { t, formatCurrency, formatDate } = useLanguage();
   174→  const user = useAuthStore((s) => s.user);
   175→
   176→  const [stats, setStats] = useState<StatsData | null>(null);
   177→  const [history, setHistory] = useState<HistoryData | null>(null);
   178→  const [dashboardData, setDashboardData] = useState<DashboardData | null>(null);
   179→  const [activeTab, setActiveTab] = useState('overview');
   180→  const [loading, setLoading] = useState(true);
   181→  const [error, setError] = useState<string | null>(null);
   182→
   183→  useEffect(() => {
   184→    if (!user?.id) return;
   185→
   186→    Promise.all([
   187→      apiFetch('/api/stats').then((res) => {
   188→        if (!res.ok) throw new Error(`Stats: HTTP ${res.status}`);
   189→        return res.json();
   190→      }),
   191→      apiFetch('/api/stats/history').then((res) => {
   192→        if (!res.ok) throw new Error(`History: HTTP ${res.status}`);
   193→        return res.json();
   194→      }),
   195→      apiFetch(`/api/dashboard?role=${user?.role || 'admin'}&userId=${user.id}`).then((res) => {
   196→        if (!res.ok) throw new Error(`Dashboard: HTTP ${res.status}`);
   197→        return res.json();
   198→      }),
   199→    ])
   200→      .then(([statsJson, historyJson, dashboardJson]) => {
   201→        if (statsJson.success && statsJson.data) {
   202→          setStats(statsJson.data as StatsData);
   203→        }
   204→        if (historyJson.success && historyJson.data) {
   205→          setHistory(historyJson.data as HistoryData);
   206→        }
   207→        if (dashboardJson.success && dashboardJson.data) {
   208→          setDashboardData(dashboardJson.data as DashboardData);
   209→        }
   210→      })
   211→      .catch((err) => {
   212→        setError(err.message || 'Network error');
   213→      })
   214→      .finally(() => {
   215→        setLoading(false);
   216→      });
   217→  }, [user?.id]);
   218→
   219→  const totalUsers = stats?.users.total ?? 0;
   220→  const totalPosts = stats?.overview.totalPosts ?? 0;
   221→  const totalProjects = stats?.overview.totalProjects ?? 0;
   222→  const totalRevenue = stats?.overview.totalRevenue ?? 0;
   223→  const totalServices = stats?.overview.totalServices ?? 0;
   224→  const confirmedRevenue = stats?.overview.confirmedRevenue ?? 0;
   225→
   226→  const revenueData = history?.revenue ?? [];
   227→  const usersGrowthData = history?.usersGrowth ?? [];
   228→  const postsByCategoryData = history?.postsByCategory ?? [];
   229→
   230→  // Build recent activity items from dashboard data
   231→  const recentActivityItems: {
   232→    id: string;
   233→    type: string;
   234→    message: string;
   235→    time: string;
   236→    icon: typeof Users;
   237→  }[] = [];
   238→
   239→  if (dashboardData?.recentActivity) {
   240→    const { users: recentUsers, quotes: recentQuotes, payments: recentPayments, tickets: recentTickets, posts: recentPosts } = dashboardData.recentActivity;
   241→
   242→    recentUsers?.forEach((u) => {
   243→      recentActivityItems.push({
   244→        id: u.id,
   245→        type: 'user',
   246→        message: `New user registered: ${u.name}`,
   247→        time: formatDate(u.createdAt),
   248→        icon: Users,
   249→      });
   250→    });
   251→
   252→    recentQuotes?.forEach((q) => {
   253→      recentActivityItems.push({
   254→        id: q.id,
   255→        type: 'quote',
   256→        message: `Quote "${q.title}" — ${q.status}`,
   257→        time: formatDate(q.createdAt),
   258→        icon: FileText,
   259→      });
   260→    });
   261→
   262→    recentPayments?.forEach((p) => {
   263→      recentActivityItems.push({
   264→        id: p.id,
   265→        type: 'payment',
   266→        message: `Payment ${formatCurrency(p.amount)} via ${p.method} — ${p.status}`,
   267→        time: formatDate(p.createdAt),
   268→        icon: DollarSign,
   269→      });
   270→    });
   271→
   272→    recentTickets?.forEach((tk) => {
   273→      recentActivityItems.push({
   274→        id: tk.id,
   275→        type: 'ticket',
   276→        message: `Support ticket: "${tk.subject}" — ${tk.status}`,
   277→        time: formatDate(tk.createdAt),
   278→        icon: CheckCircle2,
   279→      });
   280→    });
   281→
   282→    recentPosts?.forEach((p) => {
   283→      recentActivityItems.push({
   284→        id: p.id,
   285→        type: 'post',
   286→        message: `Blog post published: "${p.title}"`,
   287→        time: formatDate(p.createdAt),
   288→        icon: BookOpen,
   289→      });
   290→    });
   291→
   292→    // Sort by most recent
   293→    recentActivityItems.sort((a, b) => {
   294→      // Simple string comparison on date strings
   295→      return a.time.localeCompare(b.time) * -1;
   296→    });
   297→  }
   298→
   299→  const overviewCards = [
   300→    {
   301→      icon: Users,
   302→      label: t('admin.usersTotal'),
   303→      value: totalUsers.toLocaleString(),
   304→      change: `${stats?.users.active ?? 0} active`,
   305→      up: true,
   306→      color: 'emerald',
   307→    },
   308→    {
   309→      icon: FileText,
   310→      label: 'Posts',
   311→      value: totalPosts.toLocaleString(),
   312→      change: `${stats?.content?.publishedPosts ?? 0} published`,
   313→      up: true,
   314→      color: 'green',
   315→    },
   316→    {
   317→      icon: Briefcase,
   318→      label: t('projects.title'),
   319→      value: totalProjects.toLocaleString(),
   320→      change: `${stats?.projects?.featured ?? 0} featured`,
   321→      up: true,
   322→      color: 'teal',
   323→    },
   324→    {
   325→      icon: DollarSign,
   326→      label: t('admin.totalRevenue'),
   327→      value: formatCurrency(totalRevenue),
   328→      change: `${formatCurrency(confirmedRevenue)} confirmed`,
   329→      up: true,
   330→      color: 'yellow',
   331→    },
   332→  ];
   333→
   334→  const managementSections = [
   335→    {
   336→      icon: Users,
   337→      title: t('admin.users'),
   338→      description: t('admin.usersList'),
   339→      count: totalUsers,
   340→      color: 'emerald',
   341→    },
   342→    {
   343→      icon: Layers,
   344→      title: t('services.title'),
   345→      description: t('admin.moderationContent'),
   346→      count: totalServices,
   347→      color: 'green',
   348→    },
   349→    {
   350→      icon: Briefcase,
   351→      title: t('projects.title'),
   352→      description: 'Manage platform projects',
   353→      count: totalProjects,
   354→      color: 'teal',
   355→    },
   356→    {
   357→      icon: BookOpen,
   358→      title: t('blog.title'),
   359→      description: t('admin.moderationContent'),
   360→      count: totalPosts,
   361→      color: 'yellow',
   362→    },
   363→  ];
   364→
   365→  // ── Loading skeletons ──
   366→  if (loading) {
   367→    return (
   368→      <motion.div
   369→        variants={containerVariants}
   370→        initial="hidden"
   371→        animate="visible"
   372→        className="space-y-6"
   373→      >
   374→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   375→          {[1, 2, 3, 4].map((i) => (
   376→            <Card key={i} className="border-l-4 border-l-emerald-500">
   377→              <CardContent className="p-4 sm:p-6">
   378→                <Skeleton className="h-4 w-20 mb-2" />
   379→                <Skeleton className="h-8 w-24 mb-2" />
   380→                <Skeleton className="h-3 w-16" />
   381→              </CardContent>
   382→            </Card>
   383→          ))}
   384→        </div>
   385→
   386→        <Card>
   387→          <CardContent className="p-6">
   388→            <Skeleton className="h-6 w-32 mb-4" />
   389→            <Skeleton className="h-[300px] w-full rounded-lg" />
   390→          </CardContent>
   391→        </Card>
   392→
   393→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   394→          <Card>
   395→            <CardContent className="p-6">
   396→              <Skeleton className="h-6 w-32 mb-4" />
   397→              <Skeleton className="h-[250px] w-full rounded-lg" />
   398→            </CardContent>
   399→          </Card>
   400→          <Card>
   401→            <CardContent className="p-6">
   402→              <Skeleton className="h-6 w-32 mb-4" />
   403→              <Skeleton className="h-[250px] w-full rounded-lg" />
   404→            </CardContent>
   405→          </Card>
   406→        </div>
   407→      </motion.div>
   408→    );
   409→  }
   410→
   411→  // ── Error state ──
   412→  if (error) {
   413→    return (
   414→      <motion.div
   415→        variants={containerVariants}
   416→        initial="hidden"
   417→        animate="visible"
   418→        className="space-y-6"
   419→      >
   420→        <Card className="border-l-4 border-l-red-500">
   421→          <CardContent className="p-6">
   422→            <div className="flex items-center gap-3">
   423→              <AlertCircle className="h-6 w-6 text-red-500" />
   424→              <div>
   425→                <h3 className="font-semibold text-red-700">Failed to load admin dashboard</h3>
   426→                <p className="text-sm text-muted-foreground">{error}</p>
   427→              </div>
   428→            </div>
   429→            <Button
   430→              className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   431→              onClick={() => window.location.reload()}
   432→            >
   433→              {t('common.retry') || 'Retry'}
   434→            </Button>
   435→          </CardContent>
   436→        </Card>
   437→      </motion.div>
   438→    );
   439→  }
   440→
   441→  return (
   442→    <motion.div
   443→      variants={containerVariants}
   444→      initial="hidden"
   445→      animate="visible"
   446→      className="space-y-6"
   447→    >
   448→      {/* ── Stats Overview Cards ── */}
   449→      <motion.div variants={itemVariants} className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   450→        {overviewCards.map((card) => (
   451→          <motion.div key={card.label} variants={itemVariants}>
   452→            <Card className="hover:shadow-md transition-all border-l-4 border-l-emerald-500">
   453→              <CardContent className="p-4 sm:p-6">
   454→                <div className="flex items-start justify-between">
   455→                  <div className="space-y-2">
   456→                    <p className="text-sm text-muted-foreground">{card.label}</p>
   457→                    <p className="text-2xl font-bold">{card.value}</p>
   458→                    <div className="flex items-center gap-1">
   459→                      {card.up ? (
   460→                        <ArrowUpRight className="h-3.5 w-3.5 text-emerald-600" />
   461→                      ) : (
   462→                        <ArrowDownRight className="h-3.5 w-3.5 text-red-500" />
   463→                      )}
   464→                      <span className={`text-xs font-medium ${card.up ? 'text-emerald-600' : 'text-red-500'}`}>
   465→                        {card.change}
   466→                      </span>
   467→                    </div>
   468→                  </div>
   469→                  <div className="p-3 rounded-xl bg-emerald-50 text-emerald-600">
   470→                    <card.icon className="h-5 w-5" />
   471→                  </div>
   472→                </div>
   473→              </CardContent>
   474→            </Card>
   475→          </motion.div>
   476→        ))}
   477→      </motion.div>
   478→
   479→      {/* ── Charts Section ── */}
   480→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   481→        <TabsList className="bg-emerald-50">
   482→          <TabsTrigger value="overview" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   483→            {t('common.overview')}
   484→          </TabsTrigger>
   485→          <TabsTrigger value="analytics" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   486→            {t('admin.analytics')}
   487→          </TabsTrigger>
   488→          <TabsTrigger value="management" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white">
   489→            {t('admin.moderation')}
   490→          </TabsTrigger>
   491→        </TabsList>
   492→
   493→        {/* ── Overview Tab ── */}
   494→        <TabsContent value="overview" className="space-y-6 mt-4">
   495→          {/* Revenue Chart */}
   496→          <motion.div variants={itemVariants}>
   497→            <Card>
   498→              <CardHeader>
   499→                <CardTitle className="text-lg flex items-center gap-2">
   500→                  <TrendingUp className="h-5 w-5 text-emerald-600" />
   501→                  {t('admin.analyticsRevenue')}
   502→                </CardTitle>
   503→                <CardDescription>{t('admin.monthlyRevenue')}</CardDescription>
   504→              </CardHeader>
   505→              <CardContent>
   506→                {revenueData.length === 0 ? (
   507→                  <div className="text-center py-12">
   508→                    <BarChart3 className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   509→                    <p className="text-muted-foreground">{t('common.noData') || 'No revenue data available'}</p>
   510→                  </div>
   511→                ) : (
   512→                  <ResponsiveContainer width="100%" height={300}>
   513→                    <LineChart data={revenueData}>
   514→                      <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
   515→                      <XAxis dataKey="month" tick={{ fontSize: 12 }} />
   516→                      <YAxis tick={{ fontSize: 12 }} />
   517→                      <Tooltip
   518→                        contentStyle={{ borderRadius: '8px', border: '1px solid #d1fae5' }}
   519→                        formatter={(value: number) => formatCurrency(value)}
   520→                      />
   521→                      <Legend />
   522→                      <Line
   523→                        type="monotone"
   524→                        dataKey="revenue"
   525→                        name="Revenue"
   526→                        stroke="#059669"
   527→                        strokeWidth={3}
   528→                        dot={{ fill: '#059669', strokeWidth: 2 }}
   529→                        activeDot={{ r: 6 }}
   530→                      />
   531→                      <Line
   532→                        type="monotone"
   533→                        dataKey="target"
   534→                        name="Target"
   535→                        stroke="#6ee7b7"
   536→                        strokeWidth={2}
   537→                        strokeDasharray="5 5"
   538→                      />
   539→                    </LineChart>
   540→                  </ResponsiveContainer>
   541→                )}
   542→              </CardContent>
   543→            </Card>
   544→          </motion.div>
   545→
   546→          {/* Two-column charts */}
   547→          <motion.div variants={itemVariants} className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   548→            {/* Users Growth */}
   549→            <Card>
   550→              <CardHeader>
   551→                <CardTitle className="text-lg flex items-center gap-2">
   552→                  <BarChart3 className="h-5 w-5 text-emerald-600" />
   553→                  {t('admin.analyticsUsers')}
   554→                </CardTitle>
   555→              </CardHeader>
   556→              <CardContent>
   557→                {usersGrowthData.length === 0 ? (
   558→                  <div className="text-center py-12">
   559→                    <BarChart3 className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   560→                    <p className="text-muted-foreground">{t('common.noData') || 'No user data available'}</p>
   561→                  </div>
   562→                ) : (
   563→                  <ResponsiveContainer width="100%" height={250}>
   564→                    <BarChart data={usersGrowthData}>
   565→                      <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
   566→                      <XAxis dataKey="month" tick={{ fontSize: 11 }} />
   567→                      <YAxis tick={{ fontSize: 11 }} />
   568→                      <Tooltip
   569→                        contentStyle={{ borderRadius: '8px', border: '1px solid #d1fae5' }}
   570→                      />
   571→                      <Legend />
   572→                      <Bar dataKey="newUsers" name="New Users" fill="#059669" radius={[4, 4, 0, 0]} />
   573→                      <Bar dataKey="totalUsers" name="Total Users" fill="#6ee7b7" radius={[4, 4, 0, 0]} />
   574→                    </BarChart>
   575→                  </ResponsiveContainer>
   576→                )}
   577→              </CardContent>
   578→            </Card>
   579→
   580→            {/* Posts by Category */}
   581→            <Card>
   582→              <CardHeader>
   583→                <CardTitle className="text-lg flex items-center gap-2">
   584→                  <PieChartIcon className="h-5 w-5 text-emerald-600" />
   585→                  Posts by Category
   586→                </CardTitle>
   587→              </CardHeader>
   588→              <CardContent>
   589→                {postsByCategoryData.length === 0 ? (
   590→                  <div className="text-center py-12">
   591→                    <PieChartIcon className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   592→                    <p className="text-muted-foreground">{t('common.noData') || 'No category data available'}</p>
   593→                  </div>
   594→                ) : (
   595→                  <ResponsiveContainer width="100%" height={250}>
   596→                    <PieChart>
   597→                      <Pie
   598→                        data={postsByCategoryData}
   599→                        cx="50%"
   600→                        cy="50%"
   601→                        innerRadius={60}
   602→                        outerRadius={90}
   603→                        paddingAngle={3}
   604→                        dataKey="value"
   605→                        label={({ name, value }) => `${name}: ${value}`}
   606→                      >
