     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { motion } from 'framer-motion';
     5→import {
     6→  Users, AlertCircle, RefreshCw, Search, Plus, Pencil, Trash2,
     7→  Loader2, ChevronLeft, ChevronRight, UserPlus,
     8→} from 'lucide-react';
     9→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    10→import { Badge } from '@/components/ui/badge';
    11→import { Button } from '@/components/ui/button';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { Input } from '@/components/ui/input';
    14→import { Label } from '@/components/ui/label';
    15→import { Separator } from '@/components/ui/separator';
    16→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table';
    17→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    18→import {
    19→  Dialog, DialogContent, DialogDescription, DialogFooter,
    20→  DialogHeader, DialogTitle, DialogTrigger,
    21→} from '@/components/ui/dialog';
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select';
    25→import {
    26→  AlertDialog, AlertDialogAction, AlertDialogCancel,
    27→  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
    28→  AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger,
    29→} from '@/components/ui/alert-dialog';
    30→import { useLanguage } from '@/contexts/language-context';
    31→import { useAuthStore } from '@/lib/store';
    32→import { apiFetch, safeJson } from '@/lib/api-fetch';
    33→import { useToast } from '@/hooks/use-toast';
    34→
    35→const containerVariants = { hidden: { opacity: 0 }, visible: { opacity: 1, transition: { staggerChildren: 0.08 } } };
    36→const itemVariants = { hidden: { opacity: 0, y: 20 }, visible: { opacity: 1, y: 0, transition: { duration: 0.5 } } };
    37→
    38→interface UserData {
    39→  id: string;
    40→  name: string;
    41→  email: string;
    42→  phone?: string;
    43→  role: string;
    44→  isActive?: boolean;
    45→  createdAt: string;
    46→}
    47→
    48→interface CreateUserData {
    49→  name: string;
    50→  email: string;
    51→  password: string;
    52→  role: string;
    53→  phone: string;
    54→}
    55→
    56→interface EditUserData {
    57→  name: string;
    58→  email: string;
    59→  phone: string;
    60→  role: string;
    61→  isActive: boolean;
    62→  bio: string;
    63→  company: string;
    64→  address: string;
    65→}
    66→
    67→const PAGE_SIZE = 10;
    68→
    69→export function AdminUsers() {
    70→  const { t, formatDate } = useLanguage();
    71→  const { user: currentUser } = useAuthStore();
    72→  const { toast } = useToast();
    73→
    74→  // ── State ──
    75→  const [users, setUsers] = useState<UserData[]>([]);
    76→  const [loading, setLoading] = useState(true);
    77→  const [error, setError] = useState<string | null>(null);
    78→  const [searchQuery, setSearchQuery] = useState('');
    79→  const [currentPage, setCurrentPage] = useState(1);
    80→  const [totalPages, setTotalPages] = useState(1);
    81→  const [total, setTotal] = useState(0);
    82→
    83→  // ── Create dialog ──
    84→  const [createOpen, setCreateOpen] = useState(false);
    85→  const [createData, setCreateData] = useState<CreateUserData>({
    86→    name: '', email: '', password: '', role: 'user', phone: '',
    87→  });
    88→  const [creating, setCreating] = useState(false);
    89→
    90→  // ── Edit dialog ──
    91→  const [editOpen, setEditOpen] = useState(false);
    92→  const [editUser, setEditUser] = useState<UserData | null>(null);
    93→  const [editData, setEditData] = useState<EditUserData>({
    94→    name: '', email: '', phone: '', role: 'user', isActive: true, bio: '', company: '', address: '',
    95→  });
    96→  const [editing, setEditing] = useState(false);
    97→
    98→  // ── Deactivate ──
    99→  const [deactivatingId, setDeactivatingId] = useState<string | null>(null);
   100→
   101→  // ── Fetch users ──
   102→  const fetchUsers = useCallback(async (page: number, search: string) => {
   103→    setLoading(true);
   104→    setError(null);
   105→    try {
   106→      const params = new URLSearchParams({
   107→        limit: String(PAGE_SIZE),
   108→        page: String(page),
   109→      });
   110→      if (search) params.set('search', search);
   111→
   112→      const res = await apiFetch(`/api/admin/users?${params.toString()}`);
   113→      const data = await safeJson(res);
   114→      if (!data) { setError('Server returned non-JSON response'); return; }
   115→      if (data.success) {
   116→        setUsers(data.data || []);
   117→        setTotal(data.meta?.total || 0);
   118→        setTotalPages(data.meta?.totalPages || 1);
   119→      } else {
   120→        setError(data.message || 'Failed to load users');
   121→      }
   122→    } catch (err: any) {
   123→      setError(err.message);
   124→    } finally {
   125→      setLoading(false);
   126→    }
   127→  }, []);
   128→
   129→  useEffect(() => {
   130→    fetchUsers(currentPage, searchQuery);
   131→  }, [currentPage, searchQuery, fetchUsers, currentUser?.id]);
   132→
   133→  // ── Search debounce ──
   134→  const [searchInput, setSearchInput] = useState('');
   135→  useEffect(() => {
   136→    const timer = setTimeout(() => {
   137→      setSearchQuery(searchInput);
   138→      setCurrentPage(1);
   139→    }, 300);
   140→    return () => clearTimeout(timer);
   141→  }, [searchInput]);
   142→
   143→  // ── Create user ──
   144→  const handleCreate = async () => {
   145→    setCreating(true);
   146→    try {
   147→      const res = await apiFetch('/api/admin/users', {
   148→        method: 'POST',
   149→        headers: { 'Content-Type': 'application/json' },
   150→        body: JSON.stringify(createData),
   151→      });
   152→      const data = await safeJson(res);
   153→      if (!data) { toast({ title: t('admin.error') || 'Error', description: 'Server returned non-JSON response', variant: 'destructive' }); return; }
   154→      if (data.success) {
   155→        toast({ title: t('admin.userCreated'), description: t('admin.createUserDesc') });
   156→        setCreateOpen(false);
   157→        setCreateData({ name: '', email: '', password: '', role: 'user', phone: '' });
   158→        fetchUsers(currentPage, searchQuery);
   159→      } else {
   160→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   161→      }
   162→    } catch (err: any) {
   163→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   164→    } finally {
   165→      setCreating(false);
   166→    }
   167→  };
   168→
   169→  // ── Edit user ──
   170→  const handleEdit = async () => {
   171→    if (!editUser) return;
   172→    setEditing(true);
   173→    try {
   174→      const res = await apiFetch(`/api/admin/users?id=${editUser.id}`, {
   175→        method: 'PUT',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify(editData),
   178→      });
   179→      const data = await safeJson(res);
   180→      if (!data) { toast({ title: t('admin.error') || 'Error', description: 'Server returned non-JSON response', variant: 'destructive' }); return; }
   181→      if (data.success) {
   182→        toast({ title: t('admin.userUpdated'), description: t('admin.editUserDesc') });
   183→        setEditOpen(false);
   184→        setEditUser(null);
   185→        fetchUsers(currentPage, searchQuery);
   186→      } else {
   187→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   188→      }
   189→    } catch (err: any) {
   190→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   191→    } finally {
   192→      setEditing(false);
   193→    }
   194→  };
   195→
   196→  // ── Deactivate user ──
   197→  const handleDeactivate = async (userId: string) => {
   198→    setDeactivatingId(userId);
   199→    try {
   200→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   201→        method: 'DELETE',
   202→      });
   203→      const data = await safeJson(res);
   204→      if (!data) { toast({ title: t('admin.error') || 'Error', description: 'Server returned non-JSON response', variant: 'destructive' }); return; }
   205→      if (data.success) {
   206→        toast({ title: t('admin.userDeactivated') });
   207→        fetchUsers(currentPage, searchQuery);
   208→      } else {
   209→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   210→      }
   211→    } catch (err: any) {
   212→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   213→    } finally {
   214→      setDeactivatingId(null);
   215→    }
   216→  };
   217→
   218→  // ── Activate user (re-enable via PUT) ──
   219→  const handleActivate = async (userId: string) => {
   220→    setDeactivatingId(userId);
   221→    try {
   222→      const res = await apiFetch(`/api/admin/users?id=${userId}`, {
   223→        method: 'PUT',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({ isActive: true }),
   226→      });
   227→      const data = await safeJson(res);
   228→      if (!data) { toast({ title: t('admin.error') || 'Error', description: 'Server returned non-JSON response', variant: 'destructive' }); return; }
   229→      if (data.success) {
   230→        toast({ title: t('admin.userUpdated') });
   231→        fetchUsers(currentPage, searchQuery);
   232→      } else {
   233→        toast({ title: t('admin.error') || 'Error', description: data.message, variant: 'destructive' });
   234→      }
   235→    } catch (err: any) {
   236→      toast({ title: t('admin.error') || 'Error', description: err.message, variant: 'destructive' });
   237→    } finally {
   238→      setDeactivatingId(null);
   239→    }
   240→  };
   241→
   242→  // ── Role badge ──
   243→  const roleBadge = (role: string) => {
   244→    switch (role) {
   245→      case 'super_admin': return <Badge className="bg-purple-100 text-purple-700 border-purple-200">Super Admin</Badge>;
   246→      case 'admin': return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">Admin</Badge>;
   247→      case 'partner': return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">Partner</Badge>;
   248→      case 'user': return <Badge className="bg-blue-100 text-blue-700 border-blue-200">User</Badge>;
   249→      default: return <Badge variant="outline">{role}</Badge>;
   250→    }
   251→  };
   252→
   253→  // ── Status badge ──
   254→  const statusBadge = (isActive?: boolean) => {
   255→    if (isActive === false) {
   256→      return <Badge className="bg-red-100 text-red-700 border-red-200">{t('admin.userInactive') || 'Inactivo'}</Badge>;
   257→    }
   258→    return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('admin.userActive') || 'Activo'}</Badge>;
   259→  };
   260→
   261→  // ── Open edit dialog ──
   262→  const openEdit = (u: UserData) => {
   263→    setEditUser(u);
   264→    setEditData({
   265→      name: u.name,
   266→      email: u.email,
   267→      phone: u.phone || '',
   268→      role: u.role,
   269→      isActive: u.isActive !== false,
   270→      bio: (u as any).bio || '',
   271→      company: (u as any).company || '',
   272→      address: (u as any).address || '',
   273→    });
   274→    setEditOpen(true);
   275→  };
   276→
   277→  // ── Error state ──
   278→  if (error && !loading) {
   279→    return (
   280→      <motion.div variants={containerVariants} initial="hidden" animate="visible">
   281→        <Card className="border-l-4 border-l-red-500">
   282→          <CardContent className="p-6">
   283→            <div className="flex items-center gap-3">
   284→              <AlertCircle className="h-6 w-6 text-red-500" />
   285→              <p>{error}</p>
   286→            </div>
   287→            <Button className="mt-4" onClick={() => fetchUsers(currentPage, searchQuery)}>
   288→              <RefreshCw className="mr-2 h-4 w-4" />
   289→              {t('common.retry') || 'Retry'}
   290→            </Button>
   291→          </CardContent>
   292→        </Card>
   293→      </motion.div>
   294→    );
   295→  }
   296→
   297→  return (
   298→    <motion.div variants={containerVariants} initial="hidden" animate="visible" className="space-y-6">
   299→      {/* ── Header ── */}
   300→      <motion.div variants={itemVariants}>
   301→        <div className="flex items-center justify-between">
   302→          <div>
   303→            <h2 className="text-2xl font-bold flex items-center gap-2">
   304→              <Users className="h-6 w-6 text-emerald-600" />
   305→              {t('admin.users') || 'Users Management'}
   306→            </h2>
   307→            <p className="text-muted-foreground mt-1">
   308→              {total} {t('admin.usersTotal') || 'total users'}
   309→            </p>
   310→          </div>
   311→
   312→          {/* ── Create user button ── */}
   313→          <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   314→            <DialogTrigger asChild>
   315→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">
   316→                <UserPlus className="h-4 w-4" />
   317→                {t('admin.createUser') || 'Create User'}
   318→              </Button>
   319→            </DialogTrigger>
   320→            <DialogContent className="sm:max-w-[425px]">
   321→              <DialogHeader>
   322→                <DialogTitle>{t('admin.createUser') || 'Create User'}</DialogTitle>
   323→                <DialogDescription>{t('admin.createUserDesc') || 'Create a new platform user'}</DialogDescription>
   324→              </DialogHeader>
   325→              <div className="grid gap-4 py-4">
   326→                <div className="grid gap-2">
   327→                  <Label htmlFor="create-name">{t('admin.userName') || 'Name'}</Label>
   328→                  <Input
   329→                    id="create-name"
   330→                    value={createData.name}
   331→                    onChange={(e) => setCreateData(prev => ({ ...prev, name: e.target.value }))}
   332→                    className="focus-visible:ring-emerald-500"
   333→                  />
   334→                </div>
   335→                <div className="grid gap-2">
   336→                  <Label htmlFor="create-email">{t('admin.userEmail') || 'Email'}</Label>
   337→                  <Input
   338→                    id="create-email"
   339→                    type="email"
   340→                    value={createData.email}
   341→                    onChange={(e) => setCreateData(prev => ({ ...prev, email: e.target.value }))}
   342→                    className="focus-visible:ring-emerald-500"
   343→                  />
   344→                </div>
   345→                <div className="grid gap-2">
   346→                  <Label htmlFor="create-password">{t('admin.userPassword') || 'Password'}</Label>
   347→                  <Input
   348→                    id="create-password"
   349→                    type="password"
   350→                    value={createData.password}
   351→                    onChange={(e) => setCreateData(prev => ({ ...prev, password: e.target.value }))}
   352→                    className="focus-visible:ring-emerald-500"
   353→                  />
   354→                </div>
   355→                <div className="grid gap-2">
   356→                  <Label htmlFor="create-phone">{t('admin.userPhone') || 'Phone'}</Label>
   357→                  <Input
   358→                    id="create-phone"
   359→                    value={createData.phone}
   360→                    onChange={(e) => setCreateData(prev => ({ ...prev, phone: e.target.value }))}
   361→                    className="focus-visible:ring-emerald-500"
   362→                  />
   363→                </div>
   364→                <div className="grid gap-2">
   365→                  <Label>{t('admin.userRole') || 'Role'}</Label>
   366→                  <Select
   367→                    value={createData.role}
   368→                    onValueChange={(val) => setCreateData(prev => ({ ...prev, role: val }))}
   369→                  >
   370→                    <SelectTrigger className="focus:ring-emerald-500">
   371→                      <SelectValue />
   372→                    </SelectTrigger>
   373→                    <SelectContent>
   374→                      <SelectItem value="user">User</SelectItem>
   375→                      <SelectItem value="partner">Partner</SelectItem>
   376→                      <SelectItem value="admin">Admin</SelectItem>
   377→                      <SelectItem value="super_admin">Super Admin</SelectItem>
   378→                    </SelectContent>
   379→                  </Select>
   380→                </div>
   381→              </div>
   382→              <DialogFooter>
   383→                <Button variant="outline" onClick={() => setCreateOpen(false)}>
   384→                  {t('admin.cancel') || 'Cancel'}
   385→                </Button>
   386→                <Button
   387→                  className="bg-emerald-600 hover:bg-emerald-700 text-white"
   388→                  onClick={handleCreate}
   389→                  disabled={creating || !createData.name || !createData.email || !createData.password}
   390→                >
   391→                  {creating ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Plus className="h-4 w-4 mr-2" />}
   392→                  {t('admin.createUser') || 'Create'}
   393→                </Button>
   394→              </DialogFooter>
   395→            </DialogContent>
   396→          </Dialog>
   397→        </div>
   398→      </motion.div>
   399→
   400→      {/* ── Search ── */}
   401→      <motion.div variants={itemVariants}>
   402→        <div className="relative">
   403→          <Search className="absolute left-3 top-1/2 h-4 w-4 text-muted-foreground" />
   404→          <Input
   405→            placeholder={t('admin.searchUsers') || 'Search users...'}
   406→            value={searchInput}
   407→            onChange={(e) => setSearchInput(e.target.value)}
   408→            className="pl-10 focus-visible:ring-emerald-500"
   409→          />
   410→        </div>
   411→      </motion.div>
   412→
   413→      {/* ── Users table ── */}
   414→      <motion.div variants={itemVariants}>
   415→        <Card>
   416→          <CardContent className="p-0">
   417→            {loading ? (
   418→              <div className="p-6 space-y-4">
   419→                {[1, 2, 3].map(i => (
   420→                  <Skeleton key={i} className="h-12 w-full rounded" />
   421→                ))}
   422→              </div>
   423→            ) : users.length === 0 ? (
   424→              <div className="text-center py-12">
   425→                <Users className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   426→                <p className="text-muted-foreground">{t('common.noData') || 'No users found'}</p>
   427→              </div>
   428→            ) : (
   429→              <Table>
   430→                <TableHeader>
   431→                  <TableRow className="bg-emerald-50/50">
   432→                    <TableHead>{t('admin.userName') || 'Name'}</TableHead>
   433→                    <TableHead>{t('admin.userEmail') || 'Email'}</TableHead>
   434→                    <TableHead>{t('admin.userPhone') || 'Phone'}</TableHead>
   435→                    <TableHead>{t('admin.userRole') || 'Role'}</TableHead>
   436→                    <TableHead>{t('admin.userStatus') || 'Status'}</TableHead>
   437→                    <TableHead>{t('admin.date') || 'Date'}</TableHead>
   438→                    <TableHead className="text-right">{t('admin.actions') || 'Actions'}</TableHead>
   439→                  </TableRow>
   440→                </TableHeader>
   441→                <TableBody>
   442→                  {users.map((u) => (
   443→                    <TableRow key={u.id}>
   444→                      <TableCell>
   445→                        <div className="flex items-center gap-2">
   446→                          <Avatar className="size-6">
   447→                            <AvatarFallback className="text-xs">
   448→                              {u.name.split(' ').map(n => n[0]).join('').slice(0, 2)}
   449→                            </AvatarFallback>
   450→                          </Avatar>
   451→                          <span className="font-medium">{u.name}</span>
   452→                        </div>
   453→                      </TableCell>
   454→                      <TableCell className="text-muted-foreground">{u.email}</TableCell>
   455→                      <TableCell className="text-muted-foreground">{u.phone || '—'}</TableCell>
   456→                      <TableCell>{roleBadge(u.role)}</TableCell>
   457→                      <TableCell>{statusBadge(u.isActive)}</TableCell>
   458→                      <TableCell className="text-muted-foreground">{formatDate(u.createdAt)}</TableCell>
   459→                      <TableCell className="text-right">
   460→                        <div className="flex items-center justify-end gap-1">
   461→                          {/* Edit button */}
   462→                          <Button
   463→                            variant="ghost"
   464→                            size="icon"
   465→                            className="h-8 w-8"
   466→                            onClick={() => openEdit(u)}
   467→                          >
   468→                            <Pencil className="h-4 w-4 text-emerald-600" />
   469→                          </Button>
   470→
   471→                          {/* Deactivate / Activate button */}
   472→                          {u.isActive === false ? (
   473→                            <Button
   474→                              variant="ghost"
   475→                              size="icon"
   476→                              className="h-8 w-8"
   477→                              onClick={() => handleActivate(u.id)}
   478→                              disabled={deactivatingId === u.id}
   479→                            >
   480→                              {deactivatingId === u.id ? (
   481→                                <Loader2 className="h-4 w-4 animate-spin" />
   482→                              ) : (
   483→                                <RefreshCw className="h-4 w-4 text-emerald-600" />
   484→                              )}
   485→                            </Button>
   486→                          ) : (
   487→                            <AlertDialog>
   488→                              <AlertDialogTrigger asChild>
   489→                                <Button variant="ghost" size="icon" className="h-8 w-8">
   490→                                  <Trash2 className="h-4 w-4 text-red-600" />
   491→                                </Button>
   492→                              </AlertDialogTrigger>
   493→                              <AlertDialogContent>
   494→                                <AlertDialogHeader>
   495→                                  <AlertDialogTitle>{t('admin.deactivateUser') || 'Deactivate User'}</AlertDialogTitle>
   496→                                  <AlertDialogDescription>
   497→                                    {t('admin.confirmDeactivate') || 'Are you sure you want to deactivate this user?'}
   498→                                  </AlertDialogDescription>
   499→                                </AlertDialogHeader>
   500→                                <AlertDialogFooter>
   501→                                  <AlertDialogCancel>{t('admin.cancel') || 'Cancel'}</AlertDialogCancel>
   502→                                  <AlertDialogAction
   503→                                    className="bg-red-600 hover:bg-red-700 text-white"
   504→                                    onClick={() => handleDeactivate(u.id)}
   505→                                  >
   506→                                    {t('admin.deactivateUser') || 'Deactivate'}
   507→                                  </AlertDialogAction>
   508→                                </AlertDialogFooter>
   509→                              </AlertDialogContent>
   510→                            </AlertDialog>
   511→                          )}
   512→                        </div>
   513→                      </TableCell>
   514→                    </TableRow>
   515→                  ))}
   516→                </TableBody>
   517→              </Table>
   518→            )}
   519→          </CardContent>
   520→        </Card>
   521→      </motion.div>
   522→
   523→      {/* ── Pagination ── */}
   524→      {totalPages > 1 && (
   525→        <motion.div variants={itemVariants}>
   526→          <div className="flex items-center justify-center gap-2">
   527→            <Button
   528→              variant="outline"
