    79→  async function checkFirestoreStatus() {
    80→    setChecking(true);
    81→    try {
    82→      const { isFirebaseConfigured } = await import('@/lib/client-config');
    83→      const configured = isFirebaseConfigured();
    84→
    85→      if (!configured) {
    86→        setStatus({
    87→          configured: false,
    88→          connected: false,
    89→          hasUsers: false,
    90→          hasAdmin: false,
    91→          collections: [],
    92→        });
    93→        setChecking(false);
    94→        return;
    95→      }
    96→
    97→      // Try to connect to Firestore and check for existing data
    98→      const { firestoreClient } = await import('@/lib/firebase-client');
    99→
   100→      if (!firestoreClient) {
   101→        setStatus(prev => ({ ...prev, configured, connected: false }));
   102→        setChecking(false);
   103→        return;
   104→      }
   105→
   106→      const { collection, getDocs, query, limit, where, getDoc, doc } = await import('firebase/firestore');
   107→
   108→      // Check collections
   109→      const collectionsToCheck = ['users', 'services', 'categories', 'blog_posts', 'settings'];
   110→      const existingCollections: string[] = [];
   111→
   112→      for (const colName of collectionsToCheck) {
   113→        try {
   114→          const colRef = collection(firestoreClient, colName);
   115→          const snap = await getDocs(query(colRef, limit(1)));
   116→          if (!snap.empty) {
   117→            existingCollections.push(colName);
   118→          }
   119→        } catch {
   120→          // Collection doesn't exist yet
   121→        }
   122→      }
   123→
   124→      // Check for admin user
   125→      let hasAdmin = false;
   126→      let hasUsers = false;
   127→      try {
   128→        const usersRef = collection(firestoreClient, 'users');
   129→        const usersSnap = await getDocs(query(usersRef, limit(1)));
   130→        hasUsers = !usersSnap.empty;
   131→
   132→        if (hasUsers) {
   133→          const adminSnap = await getDocs(query(usersRef, where('role', 'in', ['admin', 'super_admin']), limit(1)));
   134→          hasAdmin = !adminSnap.empty;
   135→        }
   136→      } catch {
   137→        // Firestore rules might block — that's OK, means no admin yet
   138→      }
   139→
   140→      // Check setup lock
   141→      let isLocked = false;
   142→      try {
   143→        const settingsRef = doc(firestoreClient, 'settings', 'app');
   144→        const settingsSnap = await getDoc(settingsRef);
   145→        if (settingsSnap.exists() && settingsSnap.data()?.setupComplete) {
   146→          isLocked = true;
   147→        }
   148→      } catch {
   149→        // Settings doc doesn't exist yet
   150→      }
   151→
   152→      setStatus({
   153→        configured,
   154→        connected: true,
   155→        hasUsers,
   156→        hasAdmin,
   157→        collections: existingCollections,
   158→      });
   159→
   160→      // If setup is already done and admin exists, go to done
   161→      if (hasAdmin) {
   162→        setStep('done');
   163→      } else {
   164→        setStep('admin');
   165→      }
   166→    } catch (err) {
   167→      console.error('Status check error:', err);
   168→      setStatus(prev => ({ ...prev, configured: true, connected: false }));
   169→      setStep('admin');
   170→    }
   171→    setChecking(false);
   172→  }

   190→      const credential = await createUserWithEmailAndPassword(auth, adminEmail, adminPassword);
   191→
   192→      // Update display name
   193→      await updateProfile(credential.user, { displayName: adminName });
   194→
   195→      // Get ID token
   196→      const idToken = await credential.user.getIdToken();
   197→
   198→      // Create Firestore profile with super_admin role
   199→      const { firestoreClient } = await import('@/lib/firebase-client');
   200→      const { doc, setDoc } = await import('firebase/firestore');
   201→
   202→      if (firestoreClient) {
   203→        const uid = credential.user.uid;
   204→        await setDoc(doc(firestoreClient, 'users', uid), {
   205→          name: adminName,
   206→          email: adminEmail,
   207→          role: 'super_admin',
   208→          avatar: null,
   209→          phone: null,
   210→          company: null,
   211→          bio: null,
   212→          address: null,
   213→          isActive: true,
   214→          emailVerified: false,
   215→          authProvider: 'email',
   216→          isAnonymous: false,
   217→          createdAt: new Date(),
   218→          updatedAt: new Date(),
   219→        });
   220→      }
   221→
   222→      // Mark setup as complete
   223→      if (firestoreClient) {
   224→        const { doc, setDoc } = await import('firebase/firestore');
   225→        await setDoc(doc(firestoreClient, 'settings', 'app'), {
   226→          setupComplete: true,
   227→          setupDate: new Date(),
   228→          version: '1.0.0',
   229→          appName: 'Carsai Mozambique',
   230→        }, { merge: true });
   231→      }
   232→
   233→      // Sign out the admin (they'll log in normally)
   234→      const { signOut } = await import('@/lib/firebase-client');
   235→      await signOut(auth);
   236→
   237→      setStep('seed');
   238→    } catch (err: any) {
   239→      console.error('Admin creation error:', err);
   240→      let errorMsg = 'Erro ao criar administrador.';
   241→      if (err.code === 'auth/email-already-in-use') {
   242→        errorMsg = 'Este email já está registado no Firebase Auth.';
   243→      } else if (err.code === 'auth/weak-password') {
   244→        errorMsg = 'Senha demasiado fraca. Use pelo menos 8 caracteres.';
   245→      } else if (err.code === 'auth/invalid-email') {
   246→        errorMsg = 'Email inválido.';
   247→      }
   248→      setAdminError(errorMsg);
   249→    }
   250→    setAdminCreating(false);
   251→  }

   595→            </CardContent>

   598→
   599→        {/* Step 4: Done */}
   600→        {step === 'done' && (
   601→          <Card>

   611→            <CardContent className="space-y-4">
   612→              <div className="space-y-2">
   613→                <div className="flex items-center gap-2 text-sm">
   614→                  <CheckCircle2 className="size-4 text-green-500" />

   649→            </CardContent>
   650→          </Card>
   651→        )}
   652→      </div>
   653→    </div>
   654→  );
   655→}
