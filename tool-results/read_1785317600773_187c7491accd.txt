    98→function slugify(text: string): string {
    99→  return text
   100→    .toLowerCase()
   101→    .normalize('NFD')
   102→    .replace(/[\u0300-\u036f]/g, '')
   103→    .replace(/[^\w\s-]/g, '')
   104→    .replace(/\s+/g, '-')
   105→    .replace(/-+/g, '-')
   106→    .trim();
   107→}
   108→
   109→function parseI18n(jsonStr: string | null | undefined): Record<string, string> {
   110→  if (!jsonStr) return {};
   111→  try {
   112→    return JSON.parse(jsonStr);
   113→  } catch {
   114→    return {};
   115→  }
   116→}
   117→
   118→function stringifyI18n(obj: Record<string, string>): string | null {
   119→  const filtered = Object.fromEntries(
   120→    Object.entries(obj).filter(([_, v]) => v && v.trim())
   121→  );
   122→  if (Object.keys(filtered).length === 0) return null;
   123→  return JSON.stringify(filtered);
   124→}
   125→
   126→function getApiEndpoint(type: ContentType): string {
   127→  return `/api/admin/${type}`;
   128→}
