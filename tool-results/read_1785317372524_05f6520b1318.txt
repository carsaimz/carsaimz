     9→  Download,

    29→import { Button } from '@/components/ui/button';
    30→import { Input } from '@/components/ui/input';
    31→import { Label } from '@/components/ui/label';
    32→import { Textarea } from '@/components/ui/textarea';
    33→import { Tabs, TabsList, TabsTrigger, TabsContent } from '@/components/ui/tabs';
    34→import {

    40→  TableRow,
    41→} from '@/components/ui/table';
    42→import {

    44→  SelectContent,
    45→  SelectItem,
    46→  SelectTrigger,
    47→  SelectValue,
    48→} from '@/components/ui/select';
    49→import {
    50→  Dialog,
    51→  DialogContent,
    52→  DialogDescription,
    53→  DialogFooter,
    54→  DialogHeader,
    55→  DialogTitle,
    56→  DialogTrigger,
    57→} from '@/components/ui/dialog';
    58→import { Separator } from '@/components/ui/separator';
    59→import { Skeleton } from '@/components/ui/skeleton';
    60→import { useAuthStore } from '@/lib/store';

    63→
    64→// ── Animation variants ──
    65→const containerVariants = {
    66→  hidden: { opacity: 0 },
    67→  visible: {
    68→    opacity: 1,
    69→    transition: { staggerChildren: 0.08, delayChildren: 0.05 },
    70→  },
    71→};
    72→
    73→const itemVariants = {
    74→  hidden: { opacity: 0, y: 20 },
    75→  visible: { opacity: 1, y: 0, transition: { duration: 0.5, ease: 'easeOut' } },
    76→};
    77→
    78→// ── Types for API data ──
    79→interface ApiQuote {
    80→  id: string;
    81→  title: string;
    82→  description: string | null;
    83→  status: string;
    84→  createdAt: string;
    85→  proposals: {
    86→    id: string;
    87→    title: string;
    88→    description: string | null;
    89→    totalAmount: number | null;
    90→    status: string;
    91→    validUntil: string | null;
    92→    createdAt: string;
    93→  }[];
    94→}
    95→
    96→interface ApiProposal {
    97→  id: string;
    98→  title: string;
    99→  description: string | null;
   100→  totalAmount: number | null;
   101→  status: string;
   102→  validUntil: string | null;
   103→  createdAt: string;
   104→  quote: {
   105→    id: string;
   106→    title: string;
   107→    userId: string;
   108→  };
   109→}
   110→
   111→interface ApiPayment {
   112→  id: string;
   113→  amount: number;
   114→  method: string;
   115→  status: string;
   116→  createdAt: string;
   117→  proposal: {
   118→    id: string;
   119→    title: string;
   120→    description: string | null;
   121→    totalAmount: number | null;
   122→    status: string;
   123→    createdAt: string;
   124→    quote: {
   125→      id: string;
   126→      title: string;
   127→      status: string;
   128→    };
   129→  } | null;
   130→}
   131→
   132→interface ApiInvoice {
   133→  id: string;
   134→  number: string;
   135→  totalAmount: number;
   136→  status: string;
   137→  dueDate: string | null;
   138→  createdAt: string;
   139→  proposal: {
   140→    id: string;
   141→    title: string;
   142→    quote: {
   143→      id: string;
   144→      title: string;
   145→      userId: string;
   146→    };
   147→  };
   148→  items: {
   149→    id: string;
   150→    description: string;
   151→    quantity: number;
   152→    unitPrice: number;
   153→    total: number;
   154→  }[];
   155→}
   156→
   157→export function FinancialSection() {
   158→  const { t, formatCurrency, formatDate } = useLanguage();
   159→  const user = useAuthStore((s) => s.user);
   160→
   161→  const [activeTab, setActiveTab] = useState('quotes');
   162→  const [quoteDialogOpen, setQuoteDialogOpen] = useState(false);
   163→  const [newQuoteService, setNewQuoteService] = useState('');
   164→  const [newQuoteDescription, setNewQuoteDescription] = useState('');
   165→  const [newQuoteBudget, setNewQuoteBudget] = useState('');
   166→
   167→  const [quotes, setQuotes] = useState<ApiQuote[]>([]);
   168→  const [payments, setPayments] = useState<ApiPayment[]>([]);
   169→  const [invoices, setInvoices] = useState<ApiInvoice[]>([]);
   170→  const [loading, setLoading] = useState(true);
   171→  const [error, setError] = useState<string | null>(null);
   172→  const [submittingQuote, setSubmittingQuote] = useState(false);
   173→
   174→  useEffect(() => {
   175→    if (!user?.id) return;
   176→
   177→    Promise.all([
   178→      apiFetch(`/api/quotes?userId=${user.id}`).then((res) => {
   179→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   180→        return res.json();
   181→      }),
   182→      apiFetch(`/api/payments?userId=${user.id}`).then((res) => {
   183→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   184→        return res.json();
   185→      }),
   186→      apiFetch(`/api/invoices?userId=${user.id}`).then((res) => {
   187→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   188→        return res.json();
   189→      }),
   190→    ])
   191→      .then(([quotesJson, paymentsJson, invoicesJson]) => {
   192→        if (quotesJson.success) setQuotes(quotesJson.data || []);
   193→        if (paymentsJson.success) setPayments(paymentsJson.data || []);
   194→        if (invoicesJson.success) setInvoices(invoicesJson.data || []);
   195→      })
   196→      .catch((err) => {
   197→        setError(err.message || 'Network error');
   198→      })
   199→      .finally(() => {
   200→        setLoading(false);
   201→      });
   202→  }, [user?.id]);
   203→
   204→  // Derive proposals from quotes
   205→  const proposals: ApiProposal[] = quotes.flatMap((q) =>
   206→    q.proposals.map((p) => ({
   207→      ...p,
   208→      quote: { id: q.id, title: q.title, userId: q.id },
   209→    }))
   210→  );
   211→
   212→  const statusBadge = (status: string) => {
   213→    switch (status) {
   214→      case 'approved':
   215→      case 'accepted':
   216→      case 'confirmed':
   217→      case 'paid':
   218→      case 'resolved':
   219→      case 'closed':
   220→        return <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">{t('common.approved')}</Badge>;
   221→      case 'pending':
   222→      case 'sent':
   223→      case 'open':
   224→      case 'in_progress':
   225→      case 'issued':
   226→      case 'unpaid':
   227→        return <Badge className="bg-yellow-100 text-yellow-700 border-yellow-200">{t('common.pending')}</Badge>;
   228→      case 'rejected':
   229→      case 'failed':
   230→        return <Badge className="bg-red-100 text-red-700 border-red-200">{t('common.rejected')}</Badge>;
   231→      case 'overdue':
   232→        return <Badge className="bg-red-100 text-red-800 border-red-200">{t('financial.invoiceOverdue')}</Badge>;
   233→      default:
   234→        return <Badge variant="outline">{status}</Badge>;
   235→    }
   236→  };
   237→
   238→  const methodBadge = (method: string) => {
   239→    switch (method) {
   240→      case 'mpesa':
   241→        return (
   242→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   243→            <Phone className="h-3 w-3 mr-1" />
   244→            {t('financial.mpesa')}
   245→          </Badge>
   246→        );
   247→      case 'transfer':
   248→        return (
   249→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   250→            <Building2 className="h-3 w-3 mr-1" />
   251→            {t('financial.transfer')}
   252→          </Badge>
   253→        );
   254→      case 'deposit':
   255→        return (
   256→          <Badge variant="outline" className="text-emerald-600 border-emerald-200">
   257→            <BanknoteIcon className="h-3 w-3 mr-1" />
   258→            Deposit
   259→          </Badge>
   260→        );
   261→      default:
   262→        return <Badge variant="outline">{method}</Badge>;
   263→    }
   264→  };
   265→
   266→  const handleSubmitQuote = async () => {
   267→    if (!user?.id || !newQuoteService) return;
   268→
   269→    setSubmittingQuote(true);
   270→    try {
   271→      const res = await apiFetch('/api/quotes', {
   272→        method: 'POST',
   273→        headers: { 'Content-Type': 'application/json' },
   274→        body: JSON.stringify({
   275→          userId: user.id,
   276→          title: newQuoteService,
   277→          description: newQuoteDescription,
   278→        }),
   279→      });
   280→
   281→      const json = await res.json();
   282→      if (json.success) {
   283→        // Refresh quotes list
   284→        const quotesRes = await apiFetch(`/api/quotes?userId=${user.id}`);
   285→        const quotesJson = await quotesRes.json();
   286→        if (quotesJson.success) setQuotes(quotesJson.data || []);
   287→
   288→        setQuoteDialogOpen(false);
   289→        setNewQuoteService('');
   290→        setNewQuoteDescription('');
   291→        setNewQuoteBudget('');
   292→      }
   293→    } catch (err) {
   294→      console.error('Failed to submit quote:', err);
   295→    } finally {
   296→      setSubmittingQuote(false);
   297→    }
   298→  };
   299→
   300→  // Summary stats from real data
   301→  const totalQuotesValue = proposals.reduce((s, p) => s + (p.totalAmount || 0), 0);
   302→  const totalPaymentsValue = payments.reduce((s, p) => s + p.amount, 0);
   303→  const totalInvoicesValue = invoices.reduce((s, i) => s + i.totalAmount, 0);
   304→  const overdueInvoices = invoices.filter((i) => i.status === 'overdue').length;
   305→  const acceptedProposals = proposals.filter((p) => p.status === 'accepted').length;
   306→  const confirmedPayments = payments.filter((p) => p.status === 'confirmed').length;
   307→
   308→  // ── Loading skeletons ──
   309→  if (loading) {
   310→    return (
   311→      <motion.div
   312→        variants={containerVariants}
   313→        initial="hidden"
   314→        animate="visible"
   315→        className="space-y-6"
   316→      >
   317→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   318→          {[1, 2, 3, 4].map((i) => (
   319→            <Card key={i} className="border-l-4 border-l-emerald-500">
   320→              <CardContent className="p-4 sm:p-5">
   321→                <Skeleton className="h-4 w-20 mb-2" />
   322→                <Skeleton className="h-8 w-24 mb-1" />
   323→                <Skeleton className="h-3 w-16" />
   324→              </CardContent>
   325→            </Card>
   326→          ))}
   327→        </div>
   328→
   329→        <Card>
   330→          <CardContent className="p-6">
   331→            <Skeleton className="h-10 w-full mb-4" />
   332→            <div className="space-y-3">
   333→              {[1, 2, 3, 4, 5].map((i) => (
   334→                <Skeleton key={i} className="h-14 rounded-lg" />
   335→              ))}
   336→            </div>
   337→          </CardContent>
   338→        </Card>
   339→      </motion.div>
   340→    );
   341→  }
   342→
   343→  // ── Error state ──
   344→  if (error) {
   345→    return (
   346→      <motion.div
   347→        variants={containerVariants}
   348→        initial="hidden"
   349→        animate="visible"
   350→        className="space-y-6"
   351→      >
   352→        <Card className="border-l-4 border-l-red-500">
   353→          <CardContent className="p-6">
   354→            <div className="flex items-center gap-3">
   355→              <AlertCircle className="h-6 w-6 text-red-500" />
   356→              <div>
   357→                <h3 className="font-semibold text-red-700">Failed to load financial data</h3>
   358→                <p className="text-sm text-muted-foreground">{error}</p>
   359→              </div>
   360→            </div>
   361→            <Button
   362→              className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   363→              onClick={() => {
   364→                setError(null);
   365→                setLoading(true);
   366→                Promise.all([
   367→                  apiFetch(`/api/quotes?userId=${user?.id}`).then((r) => r.json()),
   368→                  apiFetch(`/api/payments?userId=${user?.id}`).then((r) => r.json()),
   369→                  apiFetch(`/api/invoices?userId=${user?.id}`).then((r) => r.json()),
   370→                ])
   371→                  .then(([q, p, i]) => {
   372→                    if (q.success) setQuotes(q.data || []);
   373→                    if (p.success) setPayments(p.data || []);
   374→                    if (i.success) setInvoices(i.data || []);
   375→                  })
   376→                  .catch((err) => setError(err.message))
   377→                  .finally(() => setLoading(false));
   378→              }}
   379→            >
   380→              {t('common.retry') || 'Retry'}
   381→            </Button>
   382→          </CardContent>
   383→        </Card>
   384→      </motion.div>
   385→    );
   386→  }
   387→
   388→  return (
   389→    <motion.div
   390→      variants={containerVariants}
   391→      initial="hidden"
   392→      animate="visible"
   393→      className="space-y-6"
   394→    >
   395→      {/* ── Summary Stats ── */}
   396→      <motion.div variants={itemVariants} className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   397→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   398→          <CardContent className="p-4 sm:p-5">
   399→            <div className="flex items-start justify-between">
   400→              <div>
   401→                <p className="text-sm text-muted-foreground">{t('dashboard.quotes')}</p>
   402→                <p className="text-2xl font-bold">{quotes.length}</p>
   403→                <p className="text-xs text-emerald-600 mt-1">
   404→                  {totalQuotesValue > 0 ? `Value: ${formatCurrency(totalQuotesValue)}` : 'No proposals yet'}
   405→                </p>
   406→              </div>
   407→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   408→                <FileText className="h-5 w-5" />
   409→              </div>
   410→            </div>
   411→          </CardContent>
   412→        </Card>
   413→
   414→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   415→          <CardContent className="p-4 sm:p-5">
   416→            <div className="flex items-start justify-between">
   417→              <div>
   418→                <p className="text-sm text-muted-foreground">{t('financial.proposal')}</p>
   419→                <p className="text-2xl font-bold">{proposals.length}</p>
   420→                <p className="text-xs text-emerald-600 mt-1">{acceptedProposals} accepted</p>
   421→              </div>
   422→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   423→                <Send className="h-5 w-5" />
   424→              </div>
   425→            </div>
   426→          </CardContent>
   427→        </Card>
   428→
   429→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   430→          <CardContent className="p-4 sm:p-5">
   431→            <div className="flex items-start justify-between">
   432→              <div>
   433→                <p className="text-sm text-muted-foreground">{t('financial.payment')}</p>
   434→                <p className="text-2xl font-bold">{formatCurrency(totalPaymentsValue)}</p>
   435→                <p className="text-xs text-emerald-600 mt-1">{confirmedPayments} confirmed</p>
   436→              </div>
   437→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   438→                <CreditCard className="h-5 w-5" />
   439→              </div>
   440→            </div>
   441→          </CardContent>
   442→        </Card>
   443→
   444→        <Card className="border-l-4 border-l-emerald-500 hover:shadow-md transition-shadow">
   445→          <CardContent className="p-4 sm:p-5">
   446→            <div className="flex items-start justify-between">
   447→              <div>
   448→                <p className="text-sm text-muted-foreground">{t('financial.invoice')}</p>
   449→                <p className="text-2xl font-bold">{invoices.length}</p>
   450→                <p className="text-xs text-red-600 mt-1">{overdueInvoices} overdue</p>
   451→              </div>
   452→              <div className="p-2 rounded-lg bg-emerald-50 text-emerald-600">
   453→                <Receipt className="h-5 w-5" />
   454→              </div>
   455→            </div>
   456→          </CardContent>
   457→        </Card>
   458→      </motion.div>
   459→
   460→      {/* ── Main Tabs ── */}
   461→      <motion.div variants={itemVariants}>
   462→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   463→          <TabsList className="bg-emerald-50 w-full sm:w-auto">
   464→            <TabsTrigger value="quotes" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   465→              {t('dashboard.quotes')}
   466→            </TabsTrigger>
   467→            <TabsTrigger value="proposals" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   468→              {t('financial.proposal')}
   469→            </TabsTrigger>
   470→            <TabsTrigger value="payments" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   471→              {t('financial.payment')}
   472→            </TabsTrigger>
   473→            <TabsTrigger value="invoices" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white flex-1 sm:flex-none">
   474→              {t('financial.invoice')}
   475→            </TabsTrigger>
   476→          </TabsList>
   477→
   478→          {/* ── Quotes Tab ── */}
   479→          <TabsContent value="quotes" className="space-y-4 mt-4">
   480→            {/* Request New Quote */}
   481→            <Card>
   482→              <CardHeader>
   483→                <div className="flex items-center justify-between">
   484→                  <CardTitle className="text-lg flex items-center gap-2">
   485→                    <Plus className="h-5 w-5 text-emerald-600" />
   486→                    {t('financial.quoteRequestTitle')}
   487→                  </CardTitle>
   488→                </div>
   489→                <CardDescription>{t('financial.quoteRequestDesc')}</CardDescription>
   490→              </CardHeader>
   491→              <CardContent>
   492→                <Dialog open={quoteDialogOpen} onOpenChange={setQuoteDialogOpen}>
   493→                  <DialogTrigger asChild>
   494→                    <Button className="bg-emerald-600 hover:bg-emerald-700 text-white font-semibold">
   495→                      <Plus className="h-4 w-4 mr-2" />
   496→                      {t('dashboard.quoteRequest')}
   497→                    </Button>
   498→                  </DialogTrigger>
   499→                  <DialogContent className="sm:max-w-lg">
   500→                    <DialogHeader>
   501→                      <DialogTitle>{t('financial.quoteRequestTitle')}</DialogTitle>
   502→                      <DialogDescription>{t('financial.quoteRequestDesc')}</DialogDescription>
   503→                    </DialogHeader>
   504→                    <div className="space-y-4">
   505→                      <div className="space-y-2">
   506→                        <Label>{t('services.title')}</Label>
   507→                        <Select value={newQuoteService} onValueChange={setNewQuoteService}>
   508→                          <SelectTrigger className="w-full">
   509→                            <SelectValue placeholder="Select a service" />
   510→                          </SelectTrigger>
   511→                          <SelectContent>
   512→                            <SelectItem value="web-dev">{t('services.webDev')}</SelectItem>
   513→                            <SelectItem value="mobile-dev">{t('services.mobileDev')}</SelectItem>
   514→                            <SelectItem value="cloud">{t('services.cloud')}</SelectItem>
   515→                            <SelectItem value="ai">{t('services.ai')}</SelectItem>
   516→                            <SelectItem value="consulting">{t('services.consulting')}</SelectItem>
   517→                            <SelectItem value="design">{t('services.design')}</SelectItem>
   518→                            <SelectItem value="seo">{t('services.seo')}</SelectItem>
   519→                            <SelectItem value="maintenance">{t('services.maintenance')}</SelectItem>
   520→                          </SelectContent>
   521→                        </Select>
   522→                      </div>
   523→                      <div className="space-y-2">
   524→                        <Label>{t('financial.description')}</Label>
   525→                        <Textarea
   526→                          value={newQuoteDescription}
   527→                          onChange={(e) => setNewQuoteDescription(e.target.value)}
   528→                          placeholder="Describe your requirements..."
   529→                        />
   530→                      </div>
   531→                      <div className="space-y-2">
   532→                        <Label>{t('financial.amount')} ({t('financial.currency')})</Label>
   533→                        <Input
   534→                          type="number"
   535→                          value={newQuoteBudget}
   536→                          onChange={(e) => setNewQuoteBudget(e.target.value)}
   537→                          placeholder="Estimated budget in MT"
   538→                        />
   539→                      </div>
   540→                    </div>
   541→                    <DialogFooter className="mt-4">
   542→                      <Button variant="outline" onClick={() => setQuoteDialogOpen(false)}>
   543→                        {t('common.cancel')}
   544→                      </Button>
   545→                      <Button
   546→                        className="bg-emerald-600 hover:bg-emerald-700 text-white"
   547→                        onClick={handleSubmitQuote}
   548→                        disabled={submittingQuote || !newQuoteService}
   549→                      >
   550→                        <Send className="h-4 w-4 mr-2" />
   551→                        {submittingQuote ? 'Submitting...' : t('common.confirm')}
   552→                      </Button>
   553→                    </DialogFooter>
   554→                  </DialogContent>
   555→                </Dialog>
   556→              </CardContent>
   557→            </Card>
   558→
   559→            {/* Existing Quotes List */}
   560→            <Card>
   561→              <CardHeader>
